    91→    91→// Exportar transacciones a CSV
    92→    92→function exportCSV(transactions: Transaction[], filename: string) {
    93→    93→  const headers = ['Tipo', 'Fecha', 'Contraparte', 'Activo', 'Cantidad', 'Tasa', 'Total Bruto', 'Moneda', 'Comisión', 'Total Neto', 'Exchange', 'Banco Origen', 'Banco Destino', 'Referencia', 'Estado', 'Notas']
    94→    94→  const rows = transactions.map((t) => [
    95→    95→    t.type === 'compra' ? 'Compra' : 'Venta',
    96→    96→    formatDate(t.date, true),
    97→    97→    t.counterparty,
    98→    98→    t.asset,
    99→    99→    t.amount.toString(),
   100→   100→    t.rate.toString(),
   101→   101→    t.total.toString(),
   102→   102→    t.currency,
   103→   103→    t.fee.toString(),
   104→   104→    (t.netTotal ?? t.total - t.fee).toString(),
   105→   105→    t.exchange?.name ?? '',
   106→   106→    t.fromBank?.name ?? '',
   107→   107→    t.toBank?.name ?? '',
   108→   108→    t.reference ?? '',
   109→   109→    STATUS_LABELS[t.status],
   110→   110→    (t.notes ?? '').replace(/"/g, '""'),
   111→   111→  ])
   112→   112→  const csvContent = [headers, ...rows]
   113→   113→    .map((row) => row.map((cell) => `"${cell}"`).join(','))
   114→   114→    .join('\n')
   115→   115→  const blob = new Blob(['\uFEFF' + csvContent], { type: 'text/csv;charset=utf-8;' })
   116→   116→  const url = URL.createObjectURL(blob)
   117→   117→  const a = document.createElement('a')
   118→   118→  a.href = url
   119→   119→  a.download = `${filename}.csv`
   120→   120→  a.click()
   121→   121→  URL.revokeObjectURL(url)
   122→   122→  toast({ title: 'CSV exportado', description: `${transactions.length} transacciones exportadas.` })
   123→   123→}

   125→   125→// Exportar transacciones a PDF (formato simple con texto)
   126→   126→function exportPDF(transactions: Transaction[], filename: string) {
   127→   127→  // Generar un PDF simple usando jsPDF-like approach con canvas
   128→   128→  const lines: string[] = []
   129→   129→  lines.push('P2P Ledger - Reporte de Transacciones')
   130→   130→  lines.push(`Generado: ${formatDate(new Date(), true)}`)
   131→   131→  lines.push(`Total: ${transactions.length} operaciones`)
   132→   132→  lines.push('─'.repeat(80))
   133→   133→  lines.push('')
   134→   134→
   135→   135→  for (const t of transactions) {
   136→   136→    const tipo = t.type === 'compra' ? 'COMPRA' : 'VENTA'
   137→   137→    lines.push(`[${tipo}] ${formatDate(t.date, true)} | ${t.counterparty}`)
   138→   138→    lines.push(`  ${formatNumber(t.amount, 2)} ${t.asset} @ ${formatNumber(t.rate, 2)} = ${formatCurrency(t.total, t.currency)}`)
   139→   139→    if (t.fee > 0) {
   140→   140→      lines.push(`  Comisión: ${formatCurrency(t.fee, t.currency)} | Neto: ${formatCurrency(t.netTotal ?? t.total - t.fee, t.currency)}`)
   141→   141→    }
   142→   142→    if (t.exchange) lines.push(`  Exchange: ${t.exchange.name}`)
   143→   143→    if (t.reference) lines.push(`  Ref: ${t.reference}`)
   144→   144→    lines.push(`  Estado: ${STATUS_LABELS[t.status]}`)
   145→   145→    lines.push('')
   146→   146→  }
   147→   147→
   148→   148→  const textContent = lines.join('\n')
   149→   149→  const blob = new Blob([textContent], { type: 'text/plain;charset=utf-8;' })
   150→   150→  const url = URL.createObjectURL(blob)
   151→   151→  const a = document.createElement('a')
   152→   152→  a.href = url
   153→   153→  a.download = `${filename}.txt`
   154→   154→  a.click()
   155→   155→  URL.revokeObjectURL(url)
   156→   156→  toast({ title: 'Reporte exportado', description: `${transactions.length} transacciones exportadas como texto.` })
   157→   157→}
