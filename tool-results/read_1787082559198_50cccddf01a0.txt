    90→function formatMonthLabel(monthKey: string): string {
    91→  const [year, month] = monthKey.split('-')
    92→  const monthNames = [
    93→    'Enero', 'Febrero', 'Marzo', 'Abril', 'Mayo', 'Junio',
    94→    'Julio', 'Agosto', 'Septiembre', 'Octubre', 'Noviembre', 'Diciembre',
    95→  ]
    96→  return `${monthNames[parseInt(month) - 1]} ${year}`
    97→}

    99→// Helper: detectar si estamos en Capacitor (Android/iOS)
   100→function isCapacitorNative(): boolean {
   101→  return typeof window !== 'undefined' && !!(window as unknown as { Capacitor?: unknown }).Capacitor?.isNativePlatform?.() || !!(window as unknown as { Capacitor?: { platform?: string } }).Capacitor?.platform
   102→}

   104→// Helper: guardar archivo preguntando al usuario dónde guardar
   105→// En Android (Capacitor): guarda en Downloads y ofrece compartir
   106→// En Chrome/Edge desktop: diálogo nativo "Guardar como"
   107→// Fallback: descarga automática
   108→async function saveFile(blob: Blob, defaultName: string, description: string): Promise<boolean> {
   109→  // 1) Capacitor nativo (Android/iOS)
   110→  if (isCapacitorNative()) {
   111→    try {
   112→      const { Filesystem, Directory, Encoding } = await import('@capacitor/filesystem')
   113→      const { Share } = await import('@capacitor/share')
   114→
   115→      // Convertir blob a base64
   116→      const base64 = await new Promise<string>((resolve) => {
   117→        const reader = new FileReader()
   118→        reader.onloadend = () => {
   119→          const dataUrl = reader.result as string
   120→          resolve(dataUrl.split(',')[1])
   121→        }
   122→        reader.readAsDataURL(blob)
   123→      })
   124→
   125→      // Guardar en la carpeta Downloads del dispositivo
   126→      const result = await Filesystem.writeFile({
   127→        path: `Download/${defaultName}`,
   128→        data: base64,
   129→        directory: Directory.ExternalStorage,
   130→        encoding: Encoding.Base64,
   131→        recursive: true,
   132→      })
   133→
   134→      // Ofrecer compartir el archivo (el usuario puede elegir "Guardar en..." o enviarlo)
   135→      const canShare = await Share.canShare()
   136→      if (canShare.value) {
   137→        await Share.share({
   138→          title: defaultName,
   139→          text: description,
   140→          url: result.uri,
   141→        })
   142→      }
   143→
   144→      return true
   145→    } catch (err) {
   146→      // Si share fue cancelado por el usuario, no es error
   147→      if (err instanceof Error && err.message?.includes('cancel')) return true
   148→      // Si filesystem falla (permisos), fallback a descarga web
   149→      console.warn('Capacitor save failed, falling back:', err)
   150→    }
   151→  }
   152→
   153→  // 2) showSaveFilePicker (Chrome, Edge, Opera desktop)
   154→  if ('showSaveFilePicker' in window) {
   155→    try {
   156→      const ext = defaultName.split('.').pop() ?? 'txt'
   157→      const mimeMap: Record<string, string> = {
   158→        csv: 'text/csv',
   159→        pdf: 'application/pdf',
   160→      }
   161→      const handle = await (window as unknown as { showSaveFilePicker: (opts: unknown) => Promise<FileSystemFileHandle> }).showSaveFilePicker({
   162→        suggestedName: defaultName,
   163→        types: [{
   164→          description: description,
   165→          accept: { [mimeMap[ext] ?? 'application/octet-stream']: [`.${ext}`] },
   166→        }],
   167→      })
   168→      const writable = await handle.createWritable()
   169→      await writable.write(blob)
   170→      await writable.close()
   171→      return true
   172→    } catch (err) {
   173→      if (err instanceof Error && err.name === 'AbortError') return false
   174→    }
   175→  }
   176→
   177→  // 3) Fallback: descarga automática (Firefox, Safari, WebView sin Capacitor)
   178→  const url = URL.createObjectURL(blob)
   179→  const a = document.createElement('a')
   180→  a.href = url
   181→  a.download = defaultName
   182→  document.body.appendChild(a)
   183→  a.click()
   184→  document.body.removeChild(a)
   185→  URL.revokeObjectURL(url)
   186→  return true
   187→}

   189→// Exportar transacciones a CSV
   190→async function exportCSV(transactions: Transaction[], filename: string) {
   191→  const headers = ['Tipo', 'Fecha', 'Contraparte', 'Activo', 'Cantidad', 'Tasa', 'Total Bruto', 'Moneda', 'Comisión', 'Total Neto', 'Exchange', 'Banco Origen', 'Banco Destino', 'Referencia', 'Estado', 'Notas']
   192→  const rows = transactions.map((t) => [
   193→    t.type === 'compra' ? 'Compra' : 'Venta',
   194→    formatDate(t.date),
   195→    t.counterparty,
   196→    t.asset,
   197→    t.amount.toString(),
   198→    t.rate.toString(),
   199→    t.total.toString(),
   200→    t.currency,
   201→    t.fee.toString(),
   202→    (t.netTotal ?? t.total - t.fee).toString(),
   203→    t.exchange?.name ?? '',
   204→    t.fromBank?.name ?? '',
   205→    t.toBank?.name ?? '',
   206→    t.reference ?? '',
   207→    STATUS_LABELS[t.status],
   208→    (t.notes ?? '').replace(/"/g, '""'),
   209→  ])
   210→  const csvContent = [headers, ...rows]
   211→    .map((row) => row.map((cell) => `"${cell}"`).join(','))
   212→    .join('\n')
   213→  const blob = new Blob(['\uFEFF' + csvContent], { type: 'text/csv;charset=utf-8;' })
   214→  const saved = await saveFile(blob, `${filename}.csv`, 'Archivo CSV')
   215→  if (saved) {
   216→    toast({ title: 'CSV exportado', description: `${transactions.length} transacciones exportadas a CSV.` })
   217→  }
   218→}

   220→// Exportar transacciones a PDF profesional con jsPDF
   221→async function exportPDF(transactions: Transaction[], filename: string) {
   222→  try {
   223→    const { jsPDF } = await import('jspdf')
   224→    const autoTable = (await import('jspdf-autotable')).default
   225→
   226→    const doc = new jsPDF({ orientation: 'landscape', unit: 'mm', format: 'a4' })
   227→
   228→    // Título
   229→    doc.setFontSize(18)
   230→    doc.setTextColor(30, 41, 59)
   231→    doc.text('P2P Ledger - Reporte de Transacciones', 14, 18)
   232→
   233→    // Subtítulo
   234→    doc.setFontSize(10)
   235→    doc.setTextColor(100, 116, 139)
   236→    doc.text(`Generado: ${formatDate(new Date(), true)}`, 14, 25)
   237→    doc.text(`Total: ${transactions.length} operaciones`, 14, 30)
   238→
   239→    // Resumen financiero
   240→    const compras = transactions.filter((t) => t.type === 'compra')
   241→    const ventas = transactions.filter((t) => t.type === 'venta')
   242→    const totalCompras = compras.reduce((s, t) => s + t.total, 0)
   243→    const totalVentas = ventas.reduce((s, t) => s + t.total, 0)
   244→    const totalComisiones = transactions.reduce((s, t) => s + t.fee, 0)
   245→
   246→    doc.setFontSize(9)
   247→    doc.setTextColor(30, 41, 59)
   248→    const summaryY = 36
   249→    doc.text(`Compras: ${compras.length} ops | Total: ${formatCurrency(totalCompras, transactions[0]?.currency ?? 'VES')}`, 14, summaryY)
   250→    doc.text(`Ventas: ${ventas.length} ops | Total: ${formatCurrency(totalVentas, transactions[0]?.currency ?? 'VES')}`, 14, summaryY + 5)
   251→    doc.text(`Comisiones totales: ${formatCurrency(totalComisiones, transactions[0]?.currency ?? 'VES')}`, 14, summaryY + 10)
   252→
   253→    // Tabla de transacciones
   254→    const tableHeaders = [['Tipo', 'Fecha', 'Contraparte', 'Activo', 'Cantidad', 'Tasa', 'Total', 'Comisión', 'Neto', 'Exchange', 'Estado']]
   255→    const tableRows = transactions.map((t) => [
   256→      t.type === 'compra' ? 'Compra' : 'Venta',
   257→      formatDate(t.date),
   258→      t.counterparty,
   259→      t.asset,
   260→      formatNumber(t.amount, 2),
   261→      formatNumber(t.rate, 2),
   262→      formatCurrency(t.total, t.currency),
   263→      formatCurrency(t.fee, t.currency),
   264→      formatCurrency(t.netTotal ?? t.total - t.fee, t.currency),
   265→      t.exchange?.shortName ?? t.exchange?.name ?? '-',
   266→      STATUS_LABELS[t.status],
   267→    ])
   268→
   269→    autoTable(doc, {
   270→      head: tableHeaders,
   271→      body: tableRows,
   272→      startY: summaryY + 16,
   273→      theme: 'grid',
   274→      styles: {
   275→        fontSize: 7,
   276→        cellPadding: 2,
   277→        textColor: [30, 41, 59],
   278→        lineWidth: 0.1,
   279→        lineColor: [203, 213, 225],
   280→      },
   281→      headStyles: {
   282→        fillColor: [99, 102, 241],
   283→        textColor: [255, 255, 255],
   284→        fontStyle: 'bold',
   285→        fontSize: 7.5,
   286→      },
   287→      alternateRowStyles: {
   288→        fillColor: [248, 250, 252],
   289→      },
   290→      columnStyles: {
   291→        0: { cellWidth: 18 },
   292→        1: { cellWidth: 24 },
   293→        4: { halign: 'right' },
   294→        5: { halign: 'right' },
   295→        6: { halign: 'right' },
   296→        7: { halign: 'right' },
   297→        8: { halign: 'right' },
   298→      },
   299→      didParseCell: (data) => {
   300→        // Colorear tipo de transacción
   301→        if (data.column.index === 0 && data.section === 'body') {
   302→          const val = data.cell.raw as string
   303→          if (val === 'Compra') {
   304→            data.cell.styles.textColor = [16, 185, 129]
   305→            data.cell.styles.fontStyle = 'bold'
   306→          } else if (val === 'Venta') {
   307→            data.cell.styles.textColor = [244, 63, 94]
   308→            data.cell.styles.fontStyle = 'bold'
   309→          }
   310→        }
   311→      },
   312→    })
   313→
   314→    // Pie de página
   315→    const pageCount = doc.getNumberOfPages()
   316→    for (let i = 1; i <= pageCount; i++) {
   317→      doc.setPage(i)
   318→      doc.setFontSize(8)
   319→      doc.setTextColor(148, 163, 184)
   320→      doc.text(`P2P Ledger v1.0 - Página ${i} de ${pageCount}`, doc.internal.pageSize.getWidth() / 2, doc.internal.pageSize.getHeight() - 8, { align: 'center' })
   321→    }
   322→
   323→    // Generar blob del PDF y guardar preguntando dónde
   324→    const pdfBlob = doc.output('blob')
   325→    const saved = await saveFile(pdfBlob, `${filename}.pdf`, 'Archivo PDF')
   326→    if (saved) {
   327→      toast({ title: 'PDF exportado', description: `${transactions.length} transacciones exportadas a PDF.` })
   328→    }
   329→  } catch (error) {
   330→    console.error('Error generando PDF:', error)
   331→    toast({ title: 'Error al exportar PDF', description: 'No se pudo generar el archivo PDF.', variant: 'destructive' })
   332→  }
   333→}
