    26→function isBrowser(): boolean {
    27→  return typeof window !== 'undefined' && typeof window.localStorage !== 'undefined'
    28→}

    30→function readJSON<T>(key: string, fallback: T): T {
    31→  if (!isBrowser()) return fallback
    32→  try {
    33→    const raw = window.localStorage.getItem(key)
    34→    if (!raw) return fallback
    35→    return JSON.parse(raw) as T
    36→  } catch {
    37→    return fallback
    38→  }
    39→}
    40→
    41→function writeJSON(key: string, value: unknown): void {
    42→  if (!isBrowser()) return
    43→  window.localStorage.setItem(key, JSON.stringify(value))
    44→}

    46→function uid(): string {
    47→  if (typeof crypto !== 'undefined' && crypto.randomUUID) {
    48→    return crypto.randomUUID()
    49→  }
    50→  return Date.now().toString(36) + Math.random().toString(36).slice(2, 10)
    51→}

    53→// Pequeño EventBus para notificar cambios entre componentes
    54→type Listener = () => void
    55→const listeners = new Set<Listener>()
    56→
    57→export function subscribe(listener: Listener): () => void {
    58→  listeners.add(listener)
    59→  return () => listeners.delete(listener)
    60→}
    61→
    62→function notify() {
    63→  listeners.forEach((l) => l())
    64→}

    66→// Inicialización (solo una vez)
    67→let initialized = false
    68→function ensureInit() {
    69→  if (initialized || !isBrowser()) return
    70→  initialized = true
    71→  const version = window.localStorage.getItem(VERSION_KEY)
    72→  if (version !== CURRENT_VERSION) {
    73→    // Si vienen de una versión anterior, migrar exchanges existentes
    74→    if (version === '2') {
    75→      const existing = readJSON<Exchange[]>(EXCHANGES_KEY, [])
    76→      const migrated = existing.map(migrateExchange)
    77→      writeJSON(EXCHANGES_KEY, migrated)
    78→    }
    79→    // Si es primera instalación (sin versión), sembrar exchanges por defecto
    80→    if (!version) {
    81→      const defaults = defaultExchanges()
    82→      writeJSON(EXCHANGES_KEY, defaults)
    83→    }
    84→    window.localStorage.setItem(VERSION_KEY, CURRENT_VERSION)
    85→  }
    86→}

   243→function computeBankBalance(bank: Bank, transactions: Transaction[]): {
   244→  balance: number
   245→  totalEntradas: number
   246→  totalSalidas: number
   247→  countFrom: number
   248→  countTo: number
   249→} {
   250→  let entrada = 0
   251→  let salida = 0
   252→  let countFrom = 0
   253→  let countTo = 0
   254→  for (const t of transactions) {
   255→    if (t.status !== 'completada') continue
   256→    // Para el cálculo del balance del banco usamos el neto (después de comisiones)
   257→    // porque la comisión es un costo que se descuenta del monto que efectivamente entra/sale.
   258→    const neto = t.netTotal ?? t.total
   259→    if (t.toBankId === bank.id) {
   260→      entrada += neto
   261→      countTo++
   262→    }
   263→    if (t.fromBankId === bank.id) {
   264→      salida += neto
   265→      countFrom++
   266→    }
   267→  }
   268→  return {
   269→    balance: bank.initialBalance + entrada - salida,
   270→    totalEntradas: entrada,
   271→    totalSalidas: salida,
   272→    countFrom,
   273→    countTo,
   274→  }
   275→}

   445→/**
   446→ * Selecciona el tier aplicable para un monto dado.
   447→ * Si no hay tiers definidos, retorna null (usar feeType/feeValue base).
   448→ * Si hay tiers, retorna el de mayor minAmount <= totalFiat.
   449→ */
   450→function pickTier(tiers: FeeTier[], totalFiat: number): FeeTier | null {
   451→  if (!tiers || tiers.length === 0) return null
   452→  // Ordenar de mayor a menor minAmount y elegir el primero que aplique
   453→  const sorted = [...tiers].sort((a, b) => b.minAmount - a.minAmount)
   454→  for (const tier of sorted) {
   455→    if (totalFiat >= tier.minAmount) return tier
   456→  }
   457→  // Fallback al tier de menor minAmount (debería ser 0 normalmente)
   458→  return sorted[sorted.length - 1]
   459→}

   461→/**
   462→ * Calcula la comisión que aplicaría un exchange a una operación dada.
   463→ * Soporta:
   464→ *   - Tiers escalonados (si están definidos, reemplazan el fee base)
   465→ *   - Descuento VIP/BNB (porcentaje descontado de la comisión variable, NO de la fija)
   466→ *   - Comisión fija adicional
   467→ * Retorna desglose y total.
   468→ */
   469→export function calculateFee(
   470→  exchange: Exchange | null | undefined,
   471→  type: TransactionType,
   472→  totalFiat: number
   473→): { baseFee: number; discount: number; fixedFee: number; total: number } {
   474→  if (!exchange) return { baseFee: 0, discount: 0, fixedFee: 0, total: 0 }
   475→
   476→  const tiers = type === 'compra' ? exchange.buyTiers : exchange.sellTiers
   477→  const fallbackType = type === 'compra' ? exchange.buyFeeType : exchange.sellFeeType
   478→  const fallbackValue = type === 'compra' ? exchange.buyFeeValue : exchange.sellFeeValue
   479→
   480→  let baseFee: number
   481→  const tier = pickTier(tiers, totalFiat)
   482→  if (tier) {
   483→    baseFee = tier.feeType === 'percent'
   484→      ? (totalFiat * tier.feeValue) / 100
   485→      : tier.feeValue
   486→  } else {
   487→    baseFee = fallbackType === 'percent'
   488→      ? (totalFiat * fallbackValue) / 100
   489→      : fallbackValue
   490→  }
   491→
   492→  // Descuento VIP/BNB se aplica SOLO sobre la comisión variable (no sobre la fija)
   493→  const discountPercent = exchange.discountPercent || 0
   494→  const discount = (baseFee * discountPercent) / 100
   495→  const baseFeeAfterDiscount = baseFee - discount
   496→
   497→  const fixedFee = exchange.fixedFee || 0
   498→  const total = baseFeeAfterDiscount + fixedFee
   499→
   500→  return {
   501→    baseFee,
   502→    discount,
   503→    fixedFee,
   504→    total,
   505→  }
   506→}

   544→export async function listTransactions(filters: TransactionFilters = {}): Promise<Transaction[]> {
   545→  const banks = loadBanks()
   546→  const bankMap = new Map(banks.map((b) => [b.id, b]))
   547→  const exchanges = loadExchanges()
   548→  const exMap = new Map(exchanges.map((e) => [e.id, e]))
   549→  let transactions = loadTransactions().map(migrateTx)
   550→
   551→  if (filters.type) transactions = transactions.filter((t) => t.type === filters.type)
   552→  if (filters.status) transactions = transactions.filter((t) => t.status === filters.status)
   553→  if (filters.bankId)
   554→    transactions = transactions.filter(
   555→      (t) => t.fromBankId === filters.bankId || t.toBankId === filters.bankId
   556→    )
   557→  if (filters.exchangeId) {
   558→    if (filters.exchangeId === '__none__') {
   559→      transactions = transactions.filter((t) => !t.exchangeId)
   560→    } else {
   561→      transactions = transactions.filter((t) => t.exchangeId === filters.exchangeId)
   562→    }
   563→  }
   564→  if (filters.counterparty) {
   565→    const q = filters.counterparty.toLowerCase()
   566→    transactions = transactions.filter(
   567→      (t) =>
   568→        t.counterparty.toLowerCase().includes(q) ||
   569→        t.asset.toLowerCase().includes(q) ||
   570→        (t.reference ?? '').toLowerCase().includes(q) ||
   571→        (t.notes ?? '').toLowerCase().includes(q)
   572→    )
   573→  }
   574→  if (filters.from) {
   575→    const d = new Date(filters.from)
   576→    transactions = transactions.filter((t) => new Date(t.date) >= d)
   577→  }
   578→  if (filters.to) {
   579→    const d = new Date(filters.to)
   580→    transactions = transactions.filter((t) => new Date(t.date) <= d)
   581→  }
   582→
   583→  transactions = transactions.sort(
   584→    (a, b) => new Date(b.date).getTime() - new Date(a.date).getTime()
   585→  )
   586→
   587→  if (filters.limit) transactions = transactions.slice(0, filters.limit)
   588→
   589→  // Adjuntar info de bancos y exchange
   590→  return transactions.map((t) => ({
   591→    ...t,
   592→    fromBank: t.fromBankId && bankMap.has(t.fromBankId)
   593→      ? {
   594→          id: bankMap.get(t.fromBankId)!.id,
   595→          name: bankMap.get(t.fromBankId)!.name,
   596→          currency: bankMap.get(t.fromBankId)!.currency,
   597→          color: bankMap.get(t.fromBankId)!.color,
   598→        }
   599→      : null,
   600→    toBank: t.toBankId && bankMap.has(t.toBankId)
   601→      ? {
   602→          id: bankMap.get(t.toBankId)!.id,
   603→          name: bankMap.get(t.toBankId)!.name,
   604→          currency: bankMap.get(t.toBankId)!.currency,
   605→          color: bankMap.get(t.toBankId)!.color,
   606→        }
   607→      : null,
   608→    exchange: t.exchangeId && exMap.has(t.exchangeId)
   609→      ? {
   610→          id: exMap.get(t.exchangeId)!.id,
   611→          name: exMap.get(t.exchangeId)!.name,
   612→          shortName: exMap.get(t.exchangeId)!.shortName,
   613→          color: exMap.get(t.exchangeId)!.color,
   614→        }
   615→      : null,
   616→  }))
   617→}

   619→export async function createTransaction(input: TransactionInput): Promise<Transaction> {
   620→  const transactions = loadTransactions()
   621→  const now = new Date().toISOString()
   622→  const total = input.amount * input.rate
   623→  const exchanges = loadExchanges()
   624→  const exchange = input.exchangeId ? exchanges.find((e) => e.id === input.exchangeId) : null
   625→
   626→  let fee = input.fee ?? 0
   627→  let feeBreakdown: Transaction['feeBreakdown'] = null
   628→  // Si no se pasó fee explícito pero hay exchange, calcularlo
   629→  if (input.fee === undefined && exchange) {
   630→    const calc = calculateFee(exchange, input.type, total)
   631→    fee = calc.total
   632→    feeBreakdown = {
   633→      baseFee: calc.baseFee,
   634→      discount: calc.discount,
   635→      fixedFee: calc.fixedFee,
   636→      total: calc.total,
   637→    }
   638→  } else if (exchange) {
   639→    // Si el usuario pasó fee manual y hay exchange, reconstruir breakdown aproximado
   640→    const calc = calculateFee(exchange, input.type, total)
   641→    const baseAfterDiscount = fee - calc.fixedFee
   642→    feeBreakdown = {
   643→      baseFee: calc.baseFee,
   644→      discount: Math.max(0, calc.baseFee - baseAfterDiscount),
   645→      fixedFee: calc.fixedFee,
   646→      total: fee,
   647→    }
   648→  }
   649→
   650→  const tx: Transaction = {
   651→    id: uid(),
   652→    type: input.type,
   653→    counterparty: input.counterparty,
   654→    asset: input.asset ?? 'USDT',
   655→    amount: input.amount,
   656→    rate: input.rate,
   657→    total,
   658→    currency: input.currency ?? 'VES',
   659→    fromBankId: input.fromBankId ?? null,
   660→    toBankId: input.toBankId ?? null,
   661→    exchangeId: input.exchangeId ?? null,
   662→    status: input.status ?? 'completada',
   663→    reference: input.reference ?? null,
   664→    fee,
   665→    feeBreakdown,
   666→    netTotal: total - fee,
   667→    date: input.date ?? now,
   668→    notes: input.notes ?? null,
   669→    createdAt: now,
   670→    updatedAt: now,
   671→  }
   672→  transactions.push(tx)
   673→  saveTransactions(transactions)
   674→  return tx
   675→}

   677→export async function updateTransaction(
   678→  id: string,
   679→  input: Partial<TransactionInput>
   680→): Promise<Transaction> {
   681→  const transactions = loadTransactions()
   682→  const idx = transactions.findIndex((t) => t.id === id)
   683→  if (idx === -1) throw new Error('Transacción no encontrada')
   684→  const existing = migrateTx(transactions[idx])
   685→  const finalAmount = input.amount ?? existing.amount
   686→  const finalRate = input.rate ?? existing.rate
   687→  const total = finalAmount * finalRate
   688→  const exchanges = loadExchanges()
   689→  const exchangeId = input.exchangeId !== undefined ? input.exchangeId ?? null : existing.exchangeId
   690→  const exchange = exchangeId ? exchanges.find((e) => e.id === exchangeId) : null
   691→  const type = input.type ?? existing.type
   692→
   693→  let fee: number
   694→  let feeBreakdown: Transaction['feeBreakdown'] = existing.feeBreakdown ?? null
   695→
   696→  if (input.fee !== undefined) {
   697→    // Usuario envió fee explícito
   698→    fee = input.fee
   699→    if (exchange) {
   700→      const calc = calculateFee(exchange, type, total)
   701→      const baseAfterDiscount = fee - calc.fixedFee
   702→      feeBreakdown = {
   703→        baseFee: calc.baseFee,
   704→        discount: Math.max(0, calc.baseFee - baseAfterDiscount),
   705→        fixedFee: calc.fixedFee,
   706→        total: fee,
   707→      }
   708→    } else {
   709→      feeBreakdown = { baseFee: fee, discount: 0, fixedFee: 0, total: fee }
   710→    }
   711→  } else if (input.exchangeId !== undefined || input.amount !== undefined || input.rate !== undefined || input.type !== undefined) {
   712→    // Cambió algo que afecta el cálculo — recalcular si hay exchange
   713→    if (exchange) {
   714→      const calc = calculateFee(exchange, type, total)
   715→      fee = calc.total
   716→      feeBreakdown = {
   717→        baseFee: calc.baseFee,
   718→        discount: calc.discount,
   719→        fixedFee: calc.fixedFee,
   720→        total: calc.total,
   721→      }
   722→    } else {
   723→      fee = existing.fee
   724→      feeBreakdown = null
   725→    }
   726→  } else {
   727→    fee = existing.fee
   728→  }
   729→
   730→  const updated: Transaction = {
   731→    ...existing,
   732→    type,
   733→    counterparty: input.counterparty ?? existing.counterparty,
   734→    asset: input.asset ?? existing.asset,
   735→    amount: finalAmount,
   736→    rate: finalRate,
   737→    total,
   738→    currency: input.currency ?? existing.currency,
   739→    fromBankId: input.fromBankId !== undefined ? input.fromBankId ?? null : existing.fromBankId,
   740→    toBankId: input.toBankId !== undefined ? input.toBankId ?? null : existing.toBankId,
   741→    exchangeId,
   742→    status: input.status ?? existing.status,
   743→    reference: input.reference !== undefined ? input.reference ?? null : existing.reference,
   744→    fee,
   745→    feeBreakdown,
   746→    netTotal: total - fee,
   747→    date: input.date ?? existing.date,
   748→    notes: input.notes !== undefined ? input.notes ?? null : existing.notes,
   749→    updatedAt: new Date().toISOString(),
   750→  }
   751→  transactions[idx] = updated
   752→  saveTransactions(transactions)
   753→  return updated
   754→}

   765→export async function getStats(): Promise<Stats> {
   766→  const transactions = loadTransactions().map(migrateTx)
   767→  const completadas = transactions.filter((t) => t.status === 'completada')
   768→  const banks = loadBanks()
   769→  const exchanges = loadExchanges()
   770→
   771→  const compras = completadas.filter((t) => t.type === 'compra')
   772→  const ventas = completadas.filter((t) => t.type === 'venta')
   773→  const pendientes = transactions.filter((t) => t.status === 'pendiente').length
   774→
   775→  const totalCompras = compras.reduce((s, t) => s + t.total, 0)
   776→  const totalVentas = ventas.reduce((s, t) => s + t.total, 0)
   777→  const montoCompras = compras.reduce((s, t) => s + t.amount, 0)
   778→  const montoVentas = ventas.reduce((s, t) => s + t.amount, 0)
   779→  const feesCompras = compras.reduce((s, t) => s + (t.fee || 0), 0)
   780→  const feesVentas = ventas.reduce((s, t) => s + (t.fee || 0), 0)
   781→  const feesTotal = feesCompras + feesVentas
   782→  const netCompras = totalCompras - feesCompras
   783→  const netVentas = totalVentas - feesVentas
   784→
   785→  const avgRateCompra = montoCompras > 0 ? totalCompras / montoCompras : 0
   786→  const avgRateVenta = montoVentas > 0 ? totalVentas / montoVentas : 0
   787→  const activoNeto = montoCompras - montoVentas
   788→  const volumenCruzado = Math.min(montoCompras, montoVentas)
   789→  const gananciaBruta =
   790→    volumenCruzado > 0 ? (avgRateVenta - avgRateCompra) * volumenCruzado : 0
   791→  const gananciaEstimada = gananciaBruta // bruta (sin comisiones)
   792→  const gananciaNeta = gananciaBruta - feesTotal // neta (después de comisiones)
   793→
   794→  // Top contrapartes
   795→  const cpMap = new Map<string, { total: number; amount: number; count: number }>()
   796→  for (const t of completadas) {
   797→    const e = cpMap.get(t.counterparty) ?? { total: 0, amount: 0, count: 0 }
   798→    e.total += t.total
   799→    e.amount += t.amount
   800→    e.count++
   801→    cpMap.set(t.counterparty, e)
   802→  }
   803→  const topCounterpartes = Array.from(cpMap.entries())
   804→    .map(([counterparty, v]) => ({ counterparty, ...v }))
   805→    .sort((a, b) => b.total - a.total)
   806→    .slice(0, 5)
   807→
   808→  // Activos
   809→  const assetMap = new Map<string, { amount: number; total: number; count: number }>()
   810→  for (const t of completadas) {
   811→    const e = assetMap.get(t.asset) ?? { amount: 0, total: 0, count: 0 }
   812→    e.amount += t.amount
   813→    e.total += t.total
   814→    e.count++
   815→    assetMap.set(t.asset, e)
   816→  }
   817→  const activos = Array.from(assetMap.entries())
   818→    .map(([asset, v]) => ({ asset, ...v }))
   819→    .sort((a, b) => b.total - a.total)
   820→
   821→  // Comisiones por exchange
   822→  const exMap = new Map<string, {
   823→    exchangeId: string | null
   824→    exchangeName: string
   825→    exchangeColor: string
   826→    totalFees: number
   827→    count: number
   828→    compras: number
   829→    ventas: number
   830→  }>()
   831→  const exInfoMap = new Map(exchanges.map((e) => [e.id, e]))
   832→  for (const t of completadas) {
   833→    const key = t.exchangeId ?? '__none__'
   834→    const ex = t.exchangeId ? exInfoMap.get(t.exchangeId) : null
   835→    const entry = exMap.get(key) ?? {
   836→      exchangeId: t.exchangeId ?? null,
   837→      exchangeName: ex?.name ?? 'Sin exchange',
   838→      exchangeColor: ex?.color ?? '#64748b',
   839→      totalFees: 0,
   840→      count: 0,
   841→      compras: 0,
   842→      ventas: 0,
   843→    }
   844→    entry.totalFees += t.fee || 0
   845→    entry.count++
   846→    if (t.type === 'compra') entry.compras++
   847→    else entry.ventas++
   848→    exMap.set(key, entry)
   849→  }
   850→  const feesPorExchange = Array.from(exMap.values()).sort((a, b) => b.totalFees - a.totalFees)
   851→
   852→  // Evolución mensual (12 meses)
   853→  const now = new Date()
   854→  const monthlyMap = new Map<string, { compras: number; ventas: number }>()
   855→  for (let i = 11; i >= 0; i--) {
   856→    const d = new Date(now.getFullYear(), now.getMonth() - i, 1)
   857→    const key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`
   858→    monthlyMap.set(key, { compras: 0, ventas: 0 })
   859→  }
   860→  for (const t of completadas) {
   861→    const d = new Date(t.date)
   862→    const key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`
   863→    const entry = monthlyMap.get(key)
   864→    if (!entry) continue
   865→    if (t.type === 'compra') entry.compras += t.total
   866→    else if (t.type === 'venta') entry.ventas += t.total
   867→  }
   868→  const monthly = Array.from(monthlyMap.entries()).map(([month, val]) => ({
   869→    month,
   870→    ...val,
   871→  }))
   872→
   873→  return {
   874→    resumen: {
   875→      totalCompras,
   876→      totalVentas,
   877→      montoCompras,
   878→      montoVentas,
   879→      cantidadCompras: compras.length,
   880→      cantidadVentas: ventas.length,
   881→      pendientes,
   882→      totalBanks: banks.length,
   883→      totalExchanges: exchanges.filter((e) => e.isActive).length,
   884→      feesCompras,
   885→      feesVentas,
   886→      feesTotal,
   887→      netCompras,
   888→      netVentas,
   889→      avgRateCompra,
   890→      avgRateVenta,
   891→      activoNeto,
   892→      gananciaEstimada,
   893→      gananciaNeta,
   894→    },
   895→    topCounterpartes,
   896→    activos,
   897→    feesPorExchange,
   898→    monthly,
   899→  }
   900→}

   924→export function importData(data: BackupData, mode: 'replace' | 'merge' = 'replace'): void {
   925→  if (mode === 'replace') {
   926→    if (data.banks) saveBanks(data.banks)
   927→    if (data.transactions) saveTransactions(data.transactions.map(migrateTx))
   928→    if (data.exchanges) saveExchanges(data.exchanges)
   929→  } else {
   930→    const existingBanks = loadBanks()
   931→    const existingTx = loadTransactions().map(migrateTx)
   932→    const existingEx = loadExchanges()
   933→    const existingBankIds = new Set(existingBanks.map((b) => b.id))
   934→    const existingTxIds = new Set(existingTx.map((t) => t.id))
   935→    const existingExIds = new Set(existingEx.map((e) => e.id))
   936→    const newBanks = (data.banks ?? []).filter((b) => !existingBankIds.has(b.id))
   937→    const newTx = (data.transactions ?? []).map(migrateTx).filter((t) => !existingTxIds.has(t.id))
   938→    const newEx = (data.exchanges ?? []).filter((e) => !existingExIds.has(e.id))
   939→    saveBanks([...existingBanks, ...newBanks])
   940→    saveTransactions([...existingTx, ...newTx])
   941→    saveExchanges([...existingEx, ...newEx])
   942→  }
   943→}
