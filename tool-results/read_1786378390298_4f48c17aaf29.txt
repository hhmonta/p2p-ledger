   375→   375→export function TransactionsView({ mode }: TransactionsViewProps) {
   376→   376→  const queryClient = useQueryClient()
   377→   377→  const [search, setSearch] = useState('')
   378→   378→  const [statusFilter, setStatusFilter] = useState<string>('all')
   379→   379→  const [bankFilter, setBankFilter] = useState<string>('all')
   380→   380→  const [exchangeFilter, setExchangeFilter] = useState<string>('all')
   381→   381→  const [formOpen, setFormOpen] = useState(false)
   382→   382→  const [editing, setEditing] = useState<Transaction | null>(null)
   383→   383→  const [deleting, setDeleting] = useState<Transaction | null>(null)
   384→   384→
   385→   385→  // Cargar bancos y exchanges
   386→   386→  const { data: banks = [] } = useQuery<Bank[]>({
   387→   387→    queryKey: ['banks'],
   388→   388→    queryFn: () => storage.listBanks(),
   389→   389→  })
   390→   390→
   391→   391→  const { data: exchanges = [] } = useQuery<Exchange[]>({
   392→   392→    queryKey: ['exchanges'],
   393→   393→    queryFn: () => storage.listExchanges(),
   394→   394→  })
   395→   395→
   396→   396→  const filters = useMemo(
   397→   397→    () => ({
   398→   398→      type: mode !== 'all' ? (mode as TransactionType) : undefined,
   399→   399→      status: statusFilter !== 'all' ? (statusFilter as TransactionStatus) : undefined,
   400→   400→      bankId: bankFilter !== 'all' ? bankFilter : undefined,
   401→   401→      exchangeId: exchangeFilter !== 'all' ? exchangeFilter : undefined,
   402→   402→    }),
   403→   403→    [mode, statusFilter, bankFilter, exchangeFilter]
   404→   404→  )
   405→   405→
   406→   406→  const { data: transactions = [], isLoading } = useQuery<Transaction[]>({
   407→   407→    queryKey: ['transactions', filters],
   408→   408→    queryFn: () => storage.listTransactions(filters),
   409→   409→  })
   410→   410→
   411→   411→  const filtered = transactions.filter((t) => {
   412→   412→    const q = search.toLowerCase().trim()
   413→   413→    if (!q) return true
   414→   414→    return (
   415→   415→      t.counterparty.toLowerCase().includes(q) ||
   416→   416→      t.asset.toLowerCase().includes(q) ||
   417→   417→      (t.reference ?? '').toLowerCase().includes(q) ||
   418→   418→      (t.notes ?? '').toLowerCase().includes(q)
   419→   419→    )
   420→   420→  })
   421→   421→
   422→   422→  // KPIs globales
   423→   423→  const completadas = filtered.filter((t) => t.status === 'completada')
   424→   424→  const totalMonto = completadas.reduce((s, t) => s + t.amount, 0)
   425→   425→  const totalFiat = completadas.reduce((s, t) => s + t.total, 0)
   426→   426→  const totalFees = completadas.reduce((s, t) => s + t.fee, 0)
   427→   427→  const totalNeto = completadas.reduce((s, t) => s + (t.netTotal ?? t.total - t.fee), 0)
   428→   428→  const tasaPromedio = totalMonto > 0 ? totalFiat / totalMonto : 0
   429→   429→
   430→   430→  // Agrupar transacciones por mes (solo en modo "all" / Historial)
   431→   431→  const groupedByMonth = useMemo((): MonthGroup[] | null => {
   432→   432→    if (mode !== 'all') return null
   433→   433→    const groups: Record<string, Transaction[]> = {}
   434→   434→    for (const t of filtered) {
   435→   435→      const d = new Date(t.date)
   436→   436→      const key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`
   437→   437→      if (!groups[key]) groups[key] = []
   438→   438→      groups[key].push(t)
