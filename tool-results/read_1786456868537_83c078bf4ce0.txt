     1→'use client'
     2→
     3→import { useEffect, useState, useMemo } from 'react'
     4→import { useForm } from 'react-hook-form'
     5→import { zodResolver } from '@hookform/resolvers/zod'
     6→import { z } from 'zod'
     7→import {
     8→  Dialog,
     9→  DialogContent,
    10→  DialogDescription,
    11→  DialogFooter,
    12→  DialogHeader,
    13→  DialogTitle,
    14→} from '@/components/ui/dialog'
    15→import {
    16→  Form,
    17→  FormControl,
    18→  FormDescription,
    19→  FormField,
    20→  FormItem,
    21→  FormLabel,
    22→  FormMessage,
    23→} from '@/components/ui/form'
    24→import { Input } from '@/components/ui/input'
    25→import { Button } from '@/components/ui/button'
    26→import {
    27→  Select,
    28→  SelectContent,
    29→  SelectItem,
    30→  SelectTrigger,
    31→  SelectValue,
    32→} from '@/components/ui/select'
    33→import { Textarea } from '@/components/ui/textarea'
    34→import { Loader2, ArrowRight, Sparkles, Lock, Unlock, Camera, X, Image as ImageIcon } from 'lucide-react'
    35→import type {
    36→  Bank,
    37→  Exchange,
    38→  Transaction,
    39→  TransactionInput,
    40→  TransactionType,
    41→  TransactionStatus,
    42→} from '@/lib/types'
    43→import { toast } from '@/hooks/use-toast'
    44→import { formatCurrency, formatNumber, toDateTimeInputValue } from '@/lib/format'
    45→import * as storage from '@/lib/storage'
    46→
    47→const txSchema = z.object({
    48→  type: z.enum(['compra', 'venta']),
    49→  counterparty: z.string().min(1, 'La contraparte es obligatoria'),
    50→  asset: z.string().min(1, 'El activo es obligatorio'),
    51→  amount: z.coerce.number().positive('Debe ser > 0'),
    52→  rate: z.coerce.number().min(0, 'Debe ser ≥ 0'),
    53→  currency: z.string().min(1),
    54→  fromBankId: z.string().optional().nullable(),
    55→  toBankId: z.string().optional().nullable(),
    56→  exchangeId: z.string().optional().nullable(),
    57→  status: z.enum(['pendiente', 'completada', 'cancelada']),
    58→  reference: z.string().optional(),
    59→  fee: z.coerce.number().min(0).default(0),
    60→  captureUrl: z.string().optional().nullable(),
    61→  date: z.string().min(1),
    62→  notes: z.string().optional(),
    63→})
    64→
    65→type FormValues = z.infer<typeof txSchema>
    66→
    67→const ASSETS = ['USDT', 'USDC', 'BTC', 'ETH', 'BNB', 'USD', 'EUR', 'COP']
    68→const CURRENCIES = ['VES', 'USD', 'EUR', 'COP', 'ARS', 'PEN', 'MXN', 'BRL']
    69→
    70→interface TransactionFormProps {
    71→  open: boolean
    72→  onOpenChange: (open: boolean) => void
    73→  transaction?: Transaction | null
    74→  defaultType?: TransactionType
    75→  banks: Bank[]
    76→  onSaved: () => void
    77→}
    78→
    79→export function TransactionForm({
    80→  open,
    81→  onOpenChange,
    82→  transaction,
    83→  defaultType = 'compra',
    84→  banks,
    85→  onSaved,
    86→}: TransactionFormProps) {
    87→  const [submitting, setSubmitting] = useState(false)
    88→  const [exchanges, setExchanges] = useState<Exchange[]>([])
    89→  const [feeLocked, setFeeLocked] = useState(false) // si true, el fee NO se recalcula al cambiar inputs
    90→  const [rateLocked, setRateLocked] = useState(false) // si true, la tasa NO se recalcula desde total
    91→  const [totalInput, setTotalInput] = useState<string>('') // total bruto ingresado por el usuario
    92→  const [capturePreview, setCapturePreview] = useState<string | null>(null) // preview de la captura
    93→
    94→  const activeBanks = useMemo(() => banks.filter((b) => b.isActive), [banks])
    95→  const activeExchanges = useMemo(() => exchanges.filter((e) => e.isActive), [exchanges])
    96→
    97→  // Cargar exchanges
    98→  useEffect(() => {
    99→    storage.listExchanges().then(setExchanges).catch(() => {})
   100→  }, [open])
   101→
   102→  const form = useForm<FormValues>({
   103→    resolver: zodResolver(txSchema),
   104→    defaultValues: {
   105→      type: defaultType,
   106→      counterparty: '',
   107→      asset: 'USDT',
   108→      amount: 0,
   109→      rate: 0,
   110→      currency: 'VES',
   111→      fromBankId: null,
   112→      toBankId: null,
   113→      exchangeId: null,
   114→      status: 'completada',
   115→      reference: '',
   116→      fee: 0,
   117→      captureUrl: null,
   118→      date: toDateTimeInputValue(new Date()),
   119→      notes: '',
   120→    },
   121→  })
   122→
   123→  useEffect(() => {
   124→    if (open) {
   125→      if (transaction) {
   126→        form.reset({
   127→          type: transaction.type,
   128→          counterparty: transaction.counterparty,
   129→          asset: transaction.asset,
   130→          amount: transaction.amount,
   131→          rate: transaction.rate,
   132→          currency: transaction.currency,
   133→          fromBankId: transaction.fromBankId ?? null,
   134→          toBankId: transaction.toBankId ?? null,
   135→          exchangeId: transaction.exchangeId ?? null,
   136→          status: transaction.status,
   137→          reference: transaction.reference ?? '',
   138→          fee: transaction.fee,
   139→          captureUrl: transaction.captureUrl ?? null,
   140→          date: toDateTimeInputValue(transaction.date),
   141→          notes: transaction.notes ?? '',
   142→        })
   143→        setCapturePreview(transaction.captureUrl ?? null)
   144→        // Al editar, consideramos el fee como bloqueado (es el valor guardado)
   145→        setFeeLocked(true)
   146→      } else {
   147→        form.reset({
   148→          type: defaultType,
   149→          counterparty: '',
   150→          asset: 'USDT',
   151→          amount: 0,
   152→          rate: 0,
   153→          currency: 'VES',
   154→          fromBankId: null,
   155→          toBankId: null,
   156→          exchangeId: null,
   157→          status: 'completada',
   158→          reference: '',
   159→          fee: 0,
   160→          captureUrl: null,
   161→          date: toDateTimeInputValue(new Date()),
   162→          notes: '',
   163→        })
   164→        setCapturePreview(null)
   165→        setFeeLocked(false)
   166→      }
   167→    }
   168→  }, [open, transaction, defaultType, form])
   169→
   170→  const watchedAmount = form.watch('amount')
   171→  const watchedRate = form.watch('rate')
   172→  const watchedCurrency = form.watch('currency')
   173→  const watchedAsset = form.watch('asset')
   174→  const watchedType = form.watch('type')
   175→  const watchedExchangeId = form.watch('exchangeId')
   176→  const watchedFee = form.watch('fee')
   177→
   178→  // Total bruto: si el usuario lo ingresó manualmente, usar ese valor;
   179→  // si no, calcularlo desde amount * rate
   180→  const totalFromRate = (Number(watchedAmount) || 0) * (Number(watchedRate) || 0)
   181→  const total = totalInput !== '' ? Number(totalInput) : totalFromRate
   182→
   183→  // Calcular tasa automáticamente cuando cambia el total ingresado o el monto
   184→  useEffect(() => {
   185→    if (rateLocked) return
   186→    const amount = Number(watchedAmount) || 0
   187→    const totalVal = Number(totalInput) || 0
   188→    if (amount > 0 && totalVal > 0) {
   189→      const calculatedRate = totalVal / amount
   190→      form.setValue('rate', Number(calculatedRate.toFixed(6)))
   191→    }
   192→  }, [totalInput, watchedAmount, rateLocked, form])
   193→
   194→  // Al resetear el form, sincronizar totalInput
   195→  useEffect(() => {
   196→    if (open && transaction) {
   197→      const t = (Number(transaction.amount) || 0) * (Number(transaction.rate) || 0)
   198→      setTotalInput(t > 0 ? String(t) : '')
   199→      setRateLocked(true) // al editar, la tasa es manual
   200→    } else if (open && !transaction) {
   201→      setTotalInput('')
   202→      setRateLocked(false)
   203→    }
   204→  }, [open, transaction])
   205→
   206→  // Cálculo automático de comisión cuando cambia exchange, monto, tasa o tipo
   207→  // (solo si el fee NO está bloqueado manualmente)
   208→  useEffect(() => {
   209→    if (feeLocked) return
   210→    if (!watchedExchangeId || watchedExchangeId === '__none') {
   211→      form.setValue('fee', 0)
   212→      return
   213→    }
   214→    const ex = exchanges.find((e) => e.id === watchedExchangeId)
   215→    if (!ex) return
   216→    const calc = storage.calculateFee(ex, watchedType, total)
   217→    form.setValue('fee', Number(calc.total.toFixed(6)))
   218→  }, [watchedExchangeId, watchedAmount, watchedRate, watchedType, exchanges, total, feeLocked, form, totalInput])
   219→
   220→  function handleExchangeChange(v: string) {
   221→    const value = v === '__none' ? null : v
   222→    form.setValue('exchangeId', value)
   223→    setFeeLocked(false) // si cambias exchange, se recalcula
   224→  }
   225→
   226→  function unlockFee() {
   227→    setFeeLocked(false)
   228→  }
   229→
   230→  function lockFee() {
   231→    setFeeLocked(true)
   232→  }
   233→
   234→  function unlockRate() {
   235→    setRateLocked(false)
   236→  }
   237→
   238→  function lockRate() {
   239→    setRateLocked(true)
   240→  }
   241→
   242→  // Manejar cambio en total bruto ingresado
   243→  function handleTotalInputChange(value: string) {
   244→    setTotalInput(value)
   245→    // Si el usuario está escribiendo el total, desbloquear la tasa para que se calcule
   246→    if (!rateLocked && value !== '') {
   247→      const amount = Number(watchedAmount) || 0
   248→      const totalVal = Number(value) || 0
   249→      if (amount > 0 && totalVal > 0) {
   250→        form.setValue('rate', Number((totalVal / amount).toFixed(6)))
   251→      }
   252→    }
   253→  }
   254→
   255→  async function onSubmit(values: FormValues) {
   256→    setSubmitting(true)
   257→    try {
   258→      const payload: TransactionInput = {
   259→        type: values.type,
   260→        counterparty: values.counterparty,
   261→        asset: values.asset,
   262→        amount: Number(values.amount),
   263→        rate: Number(values.rate),
   264→        currency: values.currency,
   265→        fromBankId: values.fromBankId || null,
   266→        toBankId: values.toBankId || null,
   267→        exchangeId: values.exchangeId || null,
   268→        status: values.status,
   269→        reference: values.reference || null,
   270→        fee: Number(values.fee),
   271→        captureUrl: values.captureUrl || null,
   272→        date: new Date(values.date).toISOString(),
   273→        notes: values.notes || null,
   274→      }
   275→
   276→      if (transaction) {
   277→        await storage.updateTransaction(transaction.id, payload)
   278→      } else {
   279→        await storage.createTransaction(payload)
   280→      }
   281→
   282→      toast({
   283→        title: transaction
   284→          ? 'Transacción actualizada'
   285→          : values.type === 'compra'
   286→            ? 'Compra registrada'
   287→            : 'Venta registrada',
   288→        description: `${values.type === 'compra' ? 'Compra' : 'Venta'} de ${formatNumber(values.amount)} ${values.asset} con ${values.counterparty}.`,
   289→      })
   290→      onSaved()
   291→      onOpenChange(false)
   292→    } catch (e) {
   293→      toast({
   294→        title: 'Error',
   295→        description: e instanceof Error ? e.message : 'Error desconocido',
   296→        variant: 'destructive',
   297→      })
   298→    } finally {
   299→      setSubmitting(false)
   300→    }
   301→  }
   302→
   303→  // Neto (total menos comisión)
   304→  const fee = Number(watchedFee) || 0
   305→  const netTotal = total - fee
   306→
   307→  return (
   308→    <Dialog open={open} onOpenChange={onOpenChange}>
   309→      <DialogContent className="sm:max-w-[640px] max-h-[92vh] overflow-y-auto">
   310→        <DialogHeader>
   311→          <DialogTitle>
   312→            {transaction
   313→              ? 'Editar transacción'
   314→              : defaultType === 'compra'
   315→                ? 'Nueva compra P2P'
   316→                : 'Nueva venta P2P'}
   317→          </DialogTitle>
   318→          <DialogDescription>
   319→            {watchedType === 'compra'
   320→              ? 'Registra una compra: tú envías moneda fiat y recibes el activo.'
   321→              : 'Registra una venta: tú envías el activo y recibes moneda fiat.'}
   322→          </DialogDescription>
   323→        </DialogHeader>
   324→
   325→        <Form {...form}>
   326→          <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-4">
   327→            {/* Tipo de operación */}
   328→            <FormField
   329→              control={form.control}
   330→              name="type"
   331→              render={({ field }) => (
   332→                <FormItem>
   333→                  <FormLabel>Tipo de operación</FormLabel>
   334→                  <div className="grid grid-cols-2 gap-2">
   335→                    <button
   336→                      type="button"
   337→                      onClick={() => {
   338→                        field.onChange('compra')
   339→                        setFeeLocked(false) // recalcular fee con nuevo tipo
   340→                      }}
   341→                      className={`rounded-lg border-2 p-3 text-sm font-medium transition ${
   342→                        field.value === 'compra'
   343→                          ? 'border-emerald-500 bg-emerald-50 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300'
   344→                          : 'border-border hover:border-emerald-300'
   345→                      }`}
   346→                    >
   347→                      Compra
   348→                    </button>
   349→                    <button
   350→                      type="button"
   351→                      onClick={() => {
   352→                        field.onChange('venta')
   353→                        setFeeLocked(false)
   354→                      }}
   355→                      className={`rounded-lg border-2 p-3 text-sm font-medium transition ${
   356→                        field.value === 'venta'
   357→                          ? 'border-rose-500 bg-rose-50 text-rose-700 dark:bg-rose-950/40 dark:text-rose-300'
   358→                          : 'border-border hover:border-rose-300'
   359→                      }`}
   360→                    >
   361→                      Venta
   362→                    </button>
   363→                  </div>
   364→                  <FormMessage />
   365→                </FormItem>
   366→              )}
   367→            />
   368→
   369→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   370→              <FormField
   371→                control={form.control}
   372→                name="counterparty"
   373→                render={({ field }) => (
   374→                  <FormItem className="sm:col-span-2">
   375→                    <FormLabel>Contraparte *</FormLabel>
   376→                    <FormControl>
   377→                      <Input
   378→                        placeholder="Persona o plataforma (ej: @usuario, Binance P2P, etc.)"
   379→                        {...field}
   380→                      />
   381→                    </FormControl>
   382→                    <FormMessage />
   383→                  </FormItem>
   384→                )}
   385→              />
   386→
   387→              {/* EXCHANGE */}
   388→              <FormField
   389→                control={form.control}
   390→                name="exchangeId"
   391→                render={({ field }) => (
   392→                  <FormItem className="sm:col-span-2">
   393→                    <FormLabel className="flex items-center gap-1.5">
   394→                      <Sparkles className="h-3.5 w-3.5 text-blue-500" />
   395→                      Exchange / Plataforma
   396→                    </FormLabel>
   397→                    <Select
   398→                      onValueChange={handleExchangeChange}
   399→                      value={field.value ?? '__none'}
   400→                    >
   401→                      <FormControl>
   402→                        <SelectTrigger>
   403→                          <SelectValue placeholder="Sin exchange específico" />
   404→                        </SelectTrigger>
   405→                      </FormControl>
   406→                      <SelectContent>
   407→                        <SelectItem value="__none">Sin exchange (manual)</SelectItem>
   408→                        {activeExchanges.map((e) => (
   409→                          <SelectItem key={e.id} value={e.id}>
   410→                            <span className="flex items-center gap-2">
   411→                              <span
   412→                                className="inline-block w-2 h-2 rounded-full"
   413→                                style={{ backgroundColor: e.color }}
   414→                              />
   415→                              {e.name}
   416→                              <span className="text-xs text-muted-foreground ml-1">
   417→                                {watchedType === 'compra'
   418→                                  ? e.buyFeeType === 'percent'
   419→                                    ? `${e.buyFeeValue}%`
   420→                                    : `fijo ${e.buyFeeValue}`
   421→                                  : e.sellFeeType === 'percent'
   422→                                    ? `${e.sellFeeValue}%`
   423→                                    : `fijo ${e.sellFeeValue}`}
   424→                              </span>
   425→                            </span>
   426→                          </SelectItem>
   427→                        ))}
   428→                      </SelectContent>
   429→                    </Select>
   430→                    <FormDescription>
   431→                      Selecciona el exchange para calcular la comisión automáticamente. Puedes
   432→                      ajustarla después.
   433→                    </FormDescription>
   434→                    <FormMessage />
   435→                  </FormItem>
   436→                )}
   437→              />
   438→
   439→              <FormField
   440→                control={form.control}
   441→                name="asset"
   442→                render={({ field }) => (
   443→                  <FormItem>
   444→                    <FormLabel>Activo</FormLabel>
   445→                    <Select onValueChange={field.onChange} value={field.value}>
   446→                      <FormControl>
   447→                        <SelectTrigger>
   448→                          <SelectValue />
   449→                        </SelectTrigger>
   450→                      </FormControl>
   451→                      <SelectContent>
   452→                        {ASSETS.map((a) => (
   453→                          <SelectItem key={a} value={a}>
   454→                            {a}
   455→                          </SelectItem>
   456→                        ))}
   457→                      </SelectContent>
   458→                    </Select>
   459→                    <FormMessage />
   460→                  </FormItem>
   461→                )}
   462→              />
   463→
   464→              <FormField
   465→                control={form.control}
   466→                name="currency"
   467→                render={({ field }) => (
   468→                  <FormItem>
   469→                    <FormLabel>Moneda fiat</FormLabel>
   470→                    <Select onValueChange={field.onChange} value={field.value}>
   471→                      <FormControl>
   472→                        <SelectTrigger>
   473→                          <SelectValue />
   474→                        </SelectTrigger>
   475→                      </FormControl>
   476→                      <SelectContent>
   477→                        {CURRENCIES.map((c) => (
   478→                          <SelectItem key={c} value={c}>
   479→                            {c}
   480→                          </SelectItem>
   481→                        ))}
   482→                      </SelectContent>
   483→                    </Select>
   484→                    <FormMessage />
   485→                  </FormItem>
   486→                )}
   487→              />
   488→
   489→              <FormField
   490→                control={form.control}
   491→                name="amount"
   492→                render={({ field }) => (
   493→                  <FormItem>
   494→                    <FormLabel>Cantidad de {watchedAsset || 'activo'} *</FormLabel>
   495→                    <FormControl>
   496→                      <Input
   497→                        type="number"
   498→                        step="0.000001"
   499→                        placeholder="0.00"
   500→                        {...field}
   501→                      />
   502→                    </FormControl>
   503→                    <FormMessage />
   504→                  </FormItem>
   505→                )}
   506→              />
   507→
   508→              {/* Total bruto (input) — la tasa se calcula automáticamente */}
   509→              <div className="space-y-2">
   510→                <label className="text-sm font-medium flex items-center justify-between">
   511→                  <span>Total bruto ({watchedCurrency}) *</span>
   512→                  <button
   513→                    type="button"
   514→                    onClick={rateLocked ? unlockRate : lockRate}
   515→                    className="text-xs flex items-center gap-1 px-2 py-0.5 rounded border hover:bg-muted"
   516→                    title={rateLocked ? 'Desbloquear para calcular tasa automáticamente' : 'Tasa se calcula desde el total'}
   517→                  >
   518→                    {rateLocked ? (
   519→                      <>
   520→                        <Lock className="h-3 w-3" /> Tasa manual
   521→                      </>
   522→                    ) : (
   523→                      <>
   524→                        <Unlock className="h-3 w-3" /> Tasa auto
   525→                      </>
   526→                    )}
   527→                  </button>
   528→                </label>
   529→                <Input
   530→                  type="number"
   531→                  step="0.01"
   532→                  placeholder="0.00"
   533→                  value={totalInput}
   534→                  onChange={(e) => handleTotalInputChange(e.target.value)}
   535→                />
   536→                <p className="text-[0.8rem] text-muted-foreground">
   537→                  {rateLocked
   538→                    ? 'La tasa no se recalculará automáticamente.'
   539→                    : 'Ingresa el total bruto y la tasa se calculará automáticamente.'}
   540→                </p>
   541→              </div>
   542→
   543→              {/* Tasa calculada (solo lectura o editable si está bloqueada) */}
   544→              <FormField
   545→                control={form.control}
   546→                name="rate"
   547→                render={({ field }) => (
   548→                  <FormItem>
   549→                    <FormLabel>Tasa ({watchedCurrency}/{watchedAsset || '...'})</FormLabel>
   550→                    <FormControl>
   551→                      <Input
   552→                        type="number"
   553→                        step="0.000001"
   554→                        placeholder="0.00"
   555→                        className={rateLocked ? '' : 'bg-muted/50 cursor-default'}
   556→                        readOnly={!rateLocked}
   557→                        {...field}
   558→                      />
   559→                    </FormControl>
   560→                    <FormDescription>
   561→                      {rateLocked
   562→                        ? 'Edita la tasa manualmente.'
   563→                        : `Calculada: ${formatNumber(Number(watchedRate) || 0, 6)} ${watchedCurrency}/${watchedAsset}`}
   564→                    </FormDescription>
   565→                    <FormMessage />
   566→                  </FormItem>
   567→                )}
   568→              />
   569→
   570→              {/* Bancos */}
   571→              {watchedType === 'compra' ? (
   572→                <>
   573→                  <FormField
   574→                    control={form.control}
   575→                    name="fromBankId"
   576→                    render={({ field }) => (
   577→                      <FormItem>
   578→                        <FormLabel>Banco que paga (origen)</FormLabel>
   579→                        <Select
   580→                          onValueChange={(v) =>
   581→                            field.onChange(v === '__none' ? null : v)
   582→                          }
   583→                          value={field.value ?? '__none'}
   584→                        >
   585→                          <FormControl>
   586→                            <SelectTrigger>
   587→                              <SelectValue placeholder="Sin banco" />
   588→                            </SelectTrigger>
   589→                          </FormControl>
   590→                          <SelectContent>
   591→                            <SelectItem value="__none">
   592→                              Sin banco
   593→                            </SelectItem>
   594→                            {activeBanks.map((b) => (
   595→                              <SelectItem key={b.id} value={b.id}>
   596→                                {b.name} ({b.currency})
   597→                              </SelectItem>
   598→                            ))}
   599→                          </SelectContent>
   600→                        </Select>
   601→                        <FormDescription>
   602→                          Banco desde donde sale el dinero fiat.
   603→                        </FormDescription>
   604→                        <FormMessage />
   605→                      </FormItem>
   606→                    )}
   607→                  />
   608→                  <FormField
   609→                    control={form.control}
   610→                    name="toBankId"
   611→                    render={({ field }) => (
