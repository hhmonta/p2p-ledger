    91→// Exportar transacciones a CSV
    92→function exportCSV(transactions: Transaction[], filename: string) {
    93→  const headers = ['Tipo', 'Fecha', 'Contraparte', 'Activo', 'Cantidad', 'Tasa', 'Total Bruto', 'Moneda', 'Comisión', 'Total Neto', 'Exchange', 'Banco Origen', 'Banco Destino', 'Referencia', 'Estado', 'Notas']
    94→  const rows = transactions.map((t) => [
    95→    t.type === 'compra' ? 'Compra' : 'Venta',
    96→    formatDate(t.date, true),
    97→    t.counterparty,
    98→    t.asset,
    99→    t.amount.toString(),
   100→    t.rate.toString(),
   101→    t.total.toString(),
   102→    t.currency,
   103→    t.fee.toString(),
   104→    (t.netTotal ?? t.total - t.fee).toString(),
   105→    t.exchange?.name ?? '',
   106→    t.fromBank?.name ?? '',
   107→    t.toBank?.name ?? '',
   108→    t.reference ?? '',
   109→    STATUS_LABELS[t.status],
   110→    (t.notes ?? '').replace(/"/g, '""'),
   111→  ])
   112→  const csvContent = [headers, ...rows]
   113→    .map((row) => row.map((cell) => `"${cell}"`).join(','))
   114→    .join('\n')
   115→  const blob = new Blob(['\uFEFF' + csvContent], { type: 'text/csv;charset=utf-8;' })
   116→  const url = URL.createObjectURL(blob)
   117→  const a = document.createElement('a')
   118→  a.href = url
   119→  a.download = `${filename}.csv`
   120→  a.click()
   121→  URL.revokeObjectURL(url)
   122→  toast({ title: 'CSV exportado', description: `${transactions.length} transacciones exportadas.` })
   123→}

   125→// Exportar transacciones a PDF (formato simple con texto)
   126→function exportPDF(transactions: Transaction[], filename: string) {
   127→  // Generar un PDF simple usando jsPDF-like approach con canvas
   128→  const lines: string[] = []
   129→  lines.push('P2P Ledger - Reporte de Transacciones')
   130→  lines.push(`Generado: ${formatDate(new Date(), true)}`)
   131→  lines.push(`Total: ${transactions.length} operaciones`)
   132→  lines.push('─'.repeat(80))
   133→  lines.push('')
   134→
   135→  for (const t of transactions) {
   136→    const tipo = t.type === 'compra' ? 'COMPRA' : 'VENTA'
   137→    lines.push(`[${tipo}] ${formatDate(t.date, true)} | ${t.counterparty}`)
   138→    lines.push(`  ${formatNumber(t.amount, 2)} ${t.asset} @ ${formatNumber(t.rate, 2)} = ${formatCurrency(t.total, t.currency)}`)
   139→    if (t.fee > 0) {
   140→      lines.push(`  Comisión: ${formatCurrency(t.fee, t.currency)} | Neto: ${formatCurrency(t.netTotal ?? t.total - t.fee, t.currency)}`)
   141→    }
   142→    if (t.exchange) lines.push(`  Exchange: ${t.exchange.name}`)
   143→    if (t.reference) lines.push(`  Ref: ${t.reference}`)
   144→    lines.push(`  Estado: ${STATUS_LABELS[t.status]}`)
   145→    lines.push('')
   146→  }
   147→
   148→  const textContent = lines.join('\n')
   149→  const blob = new Blob([textContent], { type: 'text/plain;charset=utf-8;' })
   150→  const url = URL.createObjectURL(blob)
   151→  const a = document.createElement('a')
   152→  a.href = url
   153→  a.download = `${filename}.txt`
   154→  a.click()
   155→  URL.revokeObjectURL(url)
   156→  toast({ title: 'Reporte exportado', description: `${transactions.length} transacciones exportadas como texto.` })
   157→}

   460→export function TransactionsView({ mode }: TransactionsViewProps) {
   461→  const queryClient = useQueryClient()
   462→  const [search, setSearch] = useState('')
   463→  const [statusFilter, setStatusFilter] = useState<string>('all')
   464→  const [bankFilter, setBankFilter] = useState<string>('all')
   465→  const [exchangeFilter, setExchangeFilter] = useState<string>('all')
   466→  const [formOpen, setFormOpen] = useState(false)
   467→  const [editing, setEditing] = useState<Transaction | null>(null)
   468→  const [deleting, setDeleting] = useState<Transaction | null>(null)
   469→
   470→  // Cargar bancos y exchanges
   471→  const { data: banks = [] } = useQuery<Bank[]>({
   472→    queryKey: ['banks'],
   473→    queryFn: () => storage.listBanks(),
   474→  })
   475→
   476→  const { data: exchanges = [] } = useQuery<Exchange[]>({
   477→    queryKey: ['exchanges'],
   478→    queryFn: () => storage.listExchanges(),
   479→  })
   480→
   481→  const filters = useMemo(
   482→    () => ({
   483→      type: mode !== 'all' ? (mode as TransactionType) : undefined,
   484→      status: statusFilter !== 'all' ? (statusFilter as TransactionStatus) : undefined,
   485→      bankId: bankFilter !== 'all' ? bankFilter : undefined,
   486→      exchangeId: exchangeFilter !== 'all' ? exchangeFilter : undefined,
   487→    }),
   488→    [mode, statusFilter, bankFilter, exchangeFilter]
   489→  )
   490→
   491→  const { data: transactions = [], isLoading } = useQuery<Transaction[]>({
   492→    queryKey: ['transactions', filters],
   493→    queryFn: () => storage.listTransactions(filters),
   494→  })
   495→
   496→  const filtered = transactions.filter((t) => {
   497→    const q = search.toLowerCase().trim()
   498→    if (!q) return true
   499→    return (
   500→      t.counterparty.toLowerCase().includes(q) ||
   501→      t.asset.toLowerCase().includes(q) ||
   502→      (t.reference ?? '').toLowerCase().includes(q) ||
   503→      (t.notes ?? '').toLowerCase().includes(q)
   504→    )
   505→  })
   506→
   507→  // KPIs globales
   508→  const completadas = filtered.filter((t) => t.status === 'completada')
   509→  const totalMonto = completadas.reduce((s, t) => s + t.amount, 0)
   510→  const totalFiat = completadas.reduce((s, t) => s + t.total, 0)
   511→  const totalFees = completadas.reduce((s, t) => s + t.fee, 0)
   512→  const totalNeto = completadas.reduce((s, t) => s + (t.netTotal ?? t.total - t.fee), 0)
   513→  const tasaPromedio = totalMonto > 0 ? totalFiat / totalMonto : 0
   514→
   515→  // Agrupar transacciones por mes (solo en modo "all" / Historial)
   516→  const groupedByMonth = useMemo((): MonthGroup[] | null => {
   517→    if (mode !== 'all') return null
   518→    const groups: Record<string, Transaction[]> = {}
   519→    for (const t of filtered) {
   520→      const d = new Date(t.date)
   521→      const key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`
   522→      if (!groups[key]) groups[key] = []
   523→      groups[key].push(t)
   524→    }
   525→    const sorted = Object.entries(groups).sort(([a], [b]) => b.localeCompare(a))
   526→    return sorted.map(([month, txs]) => {
   527→      const completadasMes = txs.filter((t) => t.status === 'completada')
   528→      const compras = completadasMes.filter((t) => t.type === 'compra')
   529→      const ventas = completadasMes.filter((t) => t.type === 'venta')
   530→      const totalCompras = compras.reduce((s, t) => s + t.total, 0)
   531→      const totalVentas = ventas.reduce((s, t) => s + t.total, 0)
   532→      const totalFeesMes = completadasMes.reduce((s, t) => s + t.fee, 0)
   533→      const totalNetoMes = completadasMes.reduce((s, t) => s + (t.netTotal ?? t.total - t.fee), 0)
   534→      const montoCompras = compras.reduce((s, t) => s + t.amount, 0)
   535→      const montoVentas = ventas.reduce((s, t) => s + t.amount, 0)
   536→      return {
   537→        month,
   538→        label: formatMonthLabel(month),
   539→        transactions: txs,
   540→        totalCompras,
   541→        totalVentas,
   542→        totalFees: totalFeesMes,
   543→        totalNeto: totalNetoMes,
   544→        tasaPromCompras: montoCompras > 0 ? totalCompras / montoCompras : 0,
   545→        tasaPromVentas: montoVentas > 0 ? totalVentas / montoVentas : 0,
   546→        countCompras: compras.length,
   547→        countVentas: ventas.length,
   548→      }
   549→    })
   550→  }, [filtered, mode])
