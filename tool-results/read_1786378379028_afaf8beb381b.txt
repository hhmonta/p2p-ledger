    79→function formatMonthLabel(monthKey: string): string {
    80→  const [year, month] = monthKey.split('-')
    81→  const monthNames = [
    82→    'Enero', 'Febrero', 'Marzo', 'Abril', 'Mayo', 'Junio',
    83→    'Julio', 'Agosto', 'Septiembre', 'Octubre', 'Noviembre', 'Diciembre',
    84→  ]
    85→  return `${monthNames[parseInt(month) - 1]} ${year}`

   357→interface TransactionsViewProps {

   361→interface MonthGroup {
   362→  month: string
   363→  label: string
   364→  transactions: Transaction[]
   365→  totalCompras: number
   366→  totalVentas: number
   367→  totalFees: number
   368→  totalNeto: number
   369→  tasaPromCompras: number
   370→  tasaPromVentas: number
   371→  countCompras: number
   372→  countVentas: number
   373→}
   374→
   375→export function TransactionsView({ mode }: TransactionsViewProps) {
   376→  const queryClient = useQueryClient()
   377→  const [search, setSearch] = useState('')
   378→  const [statusFilter, setStatusFilter] = useState<string>('all')
   379→  const [bankFilter, setBankFilter] = useState<string>('all')
   380→  const [exchangeFilter, setExchangeFilter] = useState<string>('all')
   381→  const [formOpen, setFormOpen] = useState(false)
   382→  const [editing, setEditing] = useState<Transaction | null>(null)
   383→  const [deleting, setDeleting] = useState<Transaction | null>(null)
   384→
   385→  // Cargar bancos y exchanges
   386→  const { data: banks = [] } = useQuery<Bank[]>({
   387→    queryKey: ['banks'],
   388→    queryFn: () => storage.listBanks(),
   389→  })
   390→
   391→  const { data: exchanges = [] } = useQuery<Exchange[]>({
   392→    queryKey: ['exchanges'],
   393→    queryFn: () => storage.listExchanges(),
   394→  })
   395→
   396→  const filters = useMemo(
   397→    () => ({
   398→      type: mode !== 'all' ? (mode as TransactionType) : undefined,
   399→      status: statusFilter !== 'all' ? (statusFilter as TransactionStatus) : undefined,
   400→      bankId: bankFilter !== 'all' ? bankFilter : undefined,
   401→      exchangeId: exchangeFilter !== 'all' ? exchangeFilter : undefined,
   402→    }),
   403→    [mode, statusFilter, bankFilter, exchangeFilter]
   404→  )
   405→
   406→  const { data: transactions = [], isLoading } = useQuery<Transaction[]>({
   407→    queryKey: ['transactions', filters],
   408→    queryFn: () => storage.listTransactions(filters),
   409→  })
   410→
   411→  const filtered = transactions.filter((t) => {
   412→    const q = search.toLowerCase().trim()
   413→    if (!q) return true
   414→    return (
   415→      t.counterparty.toLowerCase().includes(q) ||
   416→      t.asset.toLowerCase().includes(q) ||
   417→      (t.reference ?? '').toLowerCase().includes(q) ||
   418→      (t.notes ?? '').toLowerCase().includes(q)
   419→    )
   420→  })
   421→
   422→  // KPIs globales
   423→  const completadas = filtered.filter((t) => t.status === 'completada')
   424→  const totalMonto = completadas.reduce((s, t) => s + t.amount, 0)
   425→  const totalFiat = completadas.reduce((s, t) => s + t.total, 0)
   426→  const totalFees = completadas.reduce((s, t) => s + t.fee, 0)
   427→  const totalNeto = completadas.reduce((s, t) => s + (t.netTotal ?? t.total - t.fee), 0)
   428→  const tasaPromedio = totalMonto > 0 ? totalFiat / totalMonto : 0
   429→
   430→  // Agrupar transacciones por mes (solo en modo "all" / Historial)
   431→  const groupedByMonth = useMemo((): MonthGroup[] | null => {
   432→    if (mode !== 'all') return null
   433→    const groups: Record<string, Transaction[]> = {}
   434→    for (const t of filtered) {
   435→      const d = new Date(t.date)
   436→      const key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`
   437→      if (!groups[key]) groups[key] = []
   438→      groups[key].push(t)
   439→    }
   440→    const sorted = Object.entries(groups).sort(([a], [b]) => b.localeCompare(a))
   441→    return sorted.map(([month, txs]) => {
   442→      const completadasMes = txs.filter((t) => t.status === 'completada')
   443→      const compras = completadasMes.filter((t) => t.type === 'compra')
   444→      const ventas = completadasMes.filter((t) => t.type === 'venta')
   445→      const totalCompras = compras.reduce((s, t) => s + t.total, 0)
   446→      const totalVentas = ventas.reduce((s, t) => s + t.total, 0)
   447→      const totalFeesMes = completadasMes.reduce((s, t) => s + t.fee, 0)
   448→      const totalNetoMes = completadasMes.reduce((s, t) => s + (t.netTotal ?? t.total - t.fee), 0)
   449→      const montoCompras = compras.reduce((s, t) => s + t.amount, 0)
   450→      const montoVentas = ventas.reduce((s, t) => s + t.amount, 0)
   451→      return {
   452→        month,
   453→        label: formatMonthLabel(month),
   454→        transactions: txs,
   455→        totalCompras,
   456→        totalVentas,
   457→        totalFees: totalFeesMes,
   458→        totalNeto: totalNetoMes,
   459→        tasaPromCompras: montoCompras > 0 ? totalCompras / montoCompras : 0,
   460→        tasaPromVentas: montoVentas > 0 ? totalVentas / montoVentas : 0,
   461→        countCompras: compras.length,
   462→        countVentas: ventas.length,
   463→      }
   464→    })
   465→  }, [filtered, mode])
   466→
   467→  async function refetch() {
   468→    await queryClient.invalidateQueries({ queryKey: ['transactions'] })
   469→    await queryClient.invalidateQueries({ queryKey: ['banks'] })
   470→    await queryClient.invalidateQueries({ queryKey: ['exchanges'] })
   471→    await queryClient.invalidateQueries({ queryKey: ['stats'] })
   472→  }
   473→
   474→  function openNew() {
   475→    setEditing(null)
   476→    setFormOpen(true)
   477→  }
   478→
   479→  function openEdit(t: Transaction) {
   480→    setEditing(t)
   481→    setFormOpen(true)
   482→  }
   483→
   484→  async function confirmDelete() {
   485→    if (!deleting) return
   486→    try {
   487→      await storage.deleteTransaction(deleting.id)
   488→      toast({
   489→        title: 'Transacción eliminada',
   490→        description: `Operación con ${deleting.counterparty} fue eliminada.`,
   491→      })
   492→      await refetch()
   493→    } catch (e) {
   494→      toast({
   495→        title: 'Error',
   496→        description: e instanceof Error ? e.message : 'Error desconocido',
   497→        variant: 'destructive',
   498→      })
   499→    } finally {
   500→      setDeleting(null)
   501→    }
   502→  }
   503→
   504→  const title =
   505→    mode === 'compra'
   506→      ? 'Compras P2P'
   507→      : mode === 'venta'
   508→        ? 'Ventas P2P'
   509→        : 'Historial por mes'
   510→
   511→  const description =
   512→    mode === 'compra'
   513→      ? 'Operaciones donde recibiste un activo a cambio de moneda fiat.'
   514→      : mode === 'venta'
   515→        ? 'Operaciones donde entregaste un activo a cambio de moneda fiat.'
   516→        : 'Historial completo agrupado por mes con resumen de movimientos.'
   517→
   518→  const defaultType: TransactionType = mode === 'venta' ? 'venta' : 'compra'
   519→
   520→  const isEmptyState = !isLoading && filtered.length === 0
   521→  const emptyMessage =
   522→    search || statusFilter !== 'all' || bankFilter !== 'all' || exchangeFilter !== 'all'
