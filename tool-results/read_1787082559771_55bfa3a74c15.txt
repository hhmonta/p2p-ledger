    49→function isBrowser(): boolean {
    50→  return typeof window !== 'undefined' && typeof window.localStorage !== 'undefined'
    51→}

    53→function readJSONSync<T>(key: string, fallback: T): T {
    54→  if (!isBrowser()) return fallback
    55→  try {
    56→    const raw = window.localStorage.getItem(key)
    57→    if (!raw) return fallback
    58→    return JSON.parse(raw) as T
    59→  } catch {
    60→    return fallback
    61→  }
    62→}
    63→
    64→async function readJSON<T>(key: string, fallback: T): Promise<T> {
    65→  if (!isBrowser()) return fallback
    66→  try {
    67→    const raw = window.localStorage.getItem(key)
    68→    if (!raw) return fallback
    69→    // Si la entrada está cifrada y tenemos DEK, descifrar
    70→    if (raw.startsWith(ENC_PREFIX)) {
    71→      if (!activeDek) {
    72→        // App bloqueada o sin DEK: no podemos descifrar
    73→        return fallback
    74→      }
    75→      const payload = raw.slice(ENC_PREFIX.length)
    76→      return decryptJson<T>(payload, activeDek)
    77→    }
    78→    return JSON.parse(raw) as T
    79→  } catch {
    80→    return fallback
    81→  }
    82→}
    83→
    84→function writeJSON(key: string, value: unknown): void {
    85→  if (!isBrowser()) return
    86→  if (ENCRYPTED_KEYS.has(key) && activeDek) {
    87→    // Cifrar antes de guardar
    88→    encryptJson(value, activeDek)
    89→      .then((payload) => {
    90→        window.localStorage.setItem(key, ENC_PREFIX + payload)
    91→      })
    92→      .catch(() => {
    93→        // Si falla el cifrado, no escribir (mejor perder el dato que guardarlo en claro)
    94→      })
    95→    return
    96→  }
    97→  window.localStorage.setItem(key, JSON.stringify(value))
    98→}

   100→/**
   101→ * Migra entradas en claro a cifradas cuando se activa el PIN.
   102→ * Lee todas las claves sensibles y las re-escribe cifradas.
   103→ */
   104→export async function encryptAllStorage(dek: CryptoKey) {
   105→  if (!isBrowser()) return
   106→  for (const key of ENCRYPTED_KEYS) {
   107→    const raw = window.localStorage.getItem(key)
   108→    if (!raw) continue
   109→    if (raw.startsWith(ENC_PREFIX)) continue // ya cifrada
   110→    try {
   111→      const parsed = JSON.parse(raw)
   112→      const payload = await encryptJson(parsed, dek)
   113→      window.localStorage.setItem(key, ENC_PREFIX + payload)
   114→    } catch {
   115→      // ignorar entradas inválidas
   116→    }
   117→  }
   118→}

   120→/**
   121→ * Migra entradas cifradas a claro cuando se desactiva el PIN.
   122→ * Requiere DEK para descifrar antes de quitarla.
   123→ */
   124→export async function decryptAllStorage(dek: CryptoKey) {
   125→  if (!isBrowser()) return
   126→  for (const key of ENCRYPTED_KEYS) {
   127→    const raw = window.localStorage.getItem(key)
   128→    if (!raw) continue
   129→    if (!raw.startsWith(ENC_PREFIX)) continue // ya en claro
   130→    try {
   131→      const payload = raw.slice(ENC_PREFIX.length)
   132→      const parsed = await decryptJson<unknown>(payload, dek)
   133→      window.localStorage.setItem(key, JSON.stringify(parsed))
   134→    } catch {
   135→      // ignorar
   136→    }
   137→  }
   138→}

   140→function uid(): string {
   141→  if (typeof crypto !== 'undefined' && crypto.randomUUID) {
   142→    return crypto.randomUUID()
   143→  }
   144→  return Date.now().toString(36) + Math.random().toString(36).slice(2, 10)
   145→}

   151→export function subscribe(listener: Listener): () => void {

   160→// Inicialización (solo una vez)
   161→let initialized = false
   162→function ensureInit() {
   163→  if (initialized || !isBrowser()) return
   164→  initialized = true
   165→  const version = window.localStorage.getItem(VERSION_KEY)
   166→  if (version !== CURRENT_VERSION) {
   167→    // Si vienen de una versión anterior, migrar exchanges existentes
   168→    // v3 → v4: añadimos Bybit, KuCoin y Paxful si no existen
   169→    if (version === '3') {
   170→      const existing = readJSONSync<Exchange[]>(EXCHANGES_KEY, [])
   171→      const migrated = existing.map(migrateExchange)
   172→      const existingNames = new Set(migrated.map((e) => e.name.toLowerCase()))
   173→      const additions = defaultExchanges().filter(
   174→        (e) => !existingNames.has(e.name.toLowerCase())
   175→      )
   176→      // Solo añadimos los nuevos (Bybit, KuCoin, Paxful) - no duplicamos los antiguos
   177→      const newOnes = additions.filter((e) =>
   178→        ['Bybit P2P', 'KuCoin P2P', 'Paxful', 'Bybit Spot', 'KuCoin Spot'].includes(e.name)
   179→      )
   180→      writeJSON(EXCHANGES_KEY, [...migrated, ...newOnes])
   181→    }
   182→    // v4 → v5: añadimos P2p.me si no existe
   183→    if (version === '4') {
   184→      const existing = readJSONSync<Exchange[]>(EXCHANGES_KEY, []).map(migrateExchange)
   185→      const existingNames = new Set(existing.map((e) => e.name.toLowerCase()))
   186→      const p2pme = defaultExchanges().find((e) => e.name === 'P2p.me')!
   187→      if (!existingNames.has('p2p.me')) {
   188→        writeJSON(EXCHANGES_KEY, [...existing, p2pme])
   189→      }
   190→    }
   191→    // Si es primera instalación (sin versión), sembrar exchanges por defecto
   192→    if (!version) {
   193→      const defaults = defaultExchanges()
   194→      writeJSON(EXCHANGES_KEY, defaults)
   195→    }
   196→    window.localStorage.setItem(VERSION_KEY, CURRENT_VERSION)
   197→  }
   198→}

   500→function computeBankBalance(bank: Bank, transactions: Transaction[]): {
   501→  balance: number
   502→  totalEntradas: number
   503→  totalSalidas: number
   504→  countFrom: number
   505→  countTo: number
   506→} {
   507→  let entrada = 0
   508→  let salida = 0
   509→  let countFrom = 0
   510→  let countTo = 0
   511→  for (const t of transactions) {
   512→    if (t.status !== 'completada') continue
   513→    // Para el cálculo del balance del banco usamos el neto (después de comisiones)
   514→    // porque la comisión es un costo que se descuenta del monto que efectivamente entra/sale.
   515→    const neto = t.netTotal ?? t.total
   516→    if (t.toBankId === bank.id) {
   517→      entrada += neto
   518→      countTo++
   519→    }
   520→    if (t.fromBankId === bank.id) {
   521→      salida += neto
   522→      countFrom++
   523→    }
   524→  }
   525→  return {
   526→    balance: bank.initialBalance + entrada - salida,
   527→    totalEntradas: entrada,
   528→    totalSalidas: salida,
   529→    countFrom,
   530→    countTo,
   531→  }
   532→}

   534→export async function listBanks(): Promise<Bank[]> {
   535→  const banks = await loadBanks()
   536→  const transactions = await loadTransactions()
   537→  return banks
   538→    .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime())
   539→    .map((b) => {
   540→      const computed = computeBankBalance(b, transactions)
   541→      return {
   542→        ...b,
   543→        balance: computed.balance,
   544→        totalEntradas: computed.totalEntradas,
   545→        totalSalidas: computed.totalSalidas,
   546→        _count: {
   547→          transactionsFrom: computed.countFrom,
   548→          transactionsTo: computed.countTo,

   703→/**
   704→ * Selecciona el tier aplicable para un monto dado.
   705→ * Si no hay tiers definidos, retorna null (usar feeType/feeValue base).
   706→ * Si hay tiers, retorna el de mayor minAmount <= totalFiat.
   707→ */
   708→function pickTier(tiers: FeeTier[], totalFiat: number): FeeTier | null {
   709→  if (!tiers || tiers.length === 0) return null
   710→  // Ordenar de mayor a menor minAmount y elegir el primero que aplique
   711→  const sorted = [...tiers].sort((a, b) => b.minAmount - a.minAmount)
   712→  for (const tier of sorted) {
   713→    if (totalFiat >= tier.minAmount) return tier
   714→  }
   715→  // Fallback al tier de menor minAmount (debería ser 0 normalmente)
   716→  return sorted[sorted.length - 1]
   717→}
   718→
   719→/**
   720→ * Calcula la comisión que aplicaría un exchange a una operación dada.
   721→ * Soporta:
   722→ *   - Tiers escalonados (si están definidos, reemplazan el fee base)
   723→ *   - Descuento VIP/BNB (porcentaje descontado de la comisión variable, NO de la fija)
   724→ *   - Comisión fija adicional
   725→ * Retorna desglose y total.
   726→ */
   727→export function calculateFee(
   728→  exchange: Exchange | null | undefined,
   729→  type: TransactionType,
   730→  totalFiat: number
   731→): { baseFee: number; discount: number; fixedFee: number; total: number } {
   732→  if (!exchange) return { baseFee: 0, discount: 0, fixedFee: 0, total: 0 }
   733→
   734→  const tiers = type === 'compra' ? exchange.buyTiers : exchange.sellTiers
   735→  const fallbackType = type === 'compra' ? exchange.buyFeeType : exchange.sellFeeType
   736→  const fallbackValue = type === 'compra' ? exchange.buyFeeValue : exchange.sellFeeValue
   737→
   738→  let baseFee: number
   739→  const tier = pickTier(tiers, totalFiat)
   740→  if (tier) {
   741→    baseFee = tier.feeType === 'percent'
   742→      ? (totalFiat * tier.feeValue) / 100
   743→      : tier.feeValue
   744→  } else {
   745→    baseFee = fallbackType === 'percent'
   746→      ? (totalFiat * fallbackValue) / 100
   747→      : fallbackValue
   748→  }
   749→
   750→  // Descuento VIP/BNB se aplica SOLO sobre la comisión variable (no sobre la fija)
   751→  const discountPercent = exchange.discountPercent || 0
   752→  const discount = (baseFee * discountPercent) / 100
   753→  const baseFeeAfterDiscount = baseFee - discount
   754→
   755→  const fixedFee = exchange.fixedFee || 0
   756→  const total = baseFeeAfterDiscount + fixedFee
   757→
   758→  return {
   759→    baseFee,
   760→    discount,
   761→    fixedFee,
   762→    total,
   763→  }
   764→}

   805→export async function listTransactions(filters: TransactionFilters = {}): Promise<Transaction[]> {
   806→  const banks = await loadBanks()
   807→  const bankMap = new Map(banks.map((b) => [b.id, b]))
   808→  const exchanges = await loadExchanges()
   809→  const exMap = new Map(exchanges.map((e) => [e.id, e]))
   810→  let transactions = (await loadTransactions()).map(migrateTx)
   811→
   812→  if (filters.type) transactions = transactions.filter((t) => t.type === filters.type)
   813→  if (filters.status) transactions = transactions.filter((t) => t.status === filters.status)
   814→  if (filters.bankId)
   815→    transactions = transactions.filter(
   816→      (t) => t.fromBankId === filters.bankId || t.toBankId === filters.bankId
   817→    )
   818→  if (filters.exchangeId) {
   819→    if (filters.exchangeId === '__none__') {
   820→      transactions = transactions.filter((t) => !t.exchangeId)
   821→    } else {
   822→      transactions = transactions.filter((t) => t.exchangeId === filters.exchangeId)
   823→    }
   824→  }
   825→  if (filters.counterparty) {
   826→    const q = filters.counterparty.toLowerCase()
   827→    transactions = transactions.filter(
   828→      (t) =>
   829→        t.counterparty.toLowerCase().includes(q) ||
   830→        t.asset.toLowerCase().includes(q) ||
   831→        (t.reference ?? '').toLowerCase().includes(q) ||
   832→        (t.notes ?? '').toLowerCase().includes(q)
   833→    )
   834→  }
   835→  if (filters.from) {
   836→    const d = new Date(filters.from)
   837→    transactions = transactions.filter((t) => new Date(t.date) >= d)
   838→  }
   839→  if (filters.to) {
   840→    const d = new Date(filters.to)
   841→    transactions = transactions.filter((t) => new Date(t.date) <= d)
   842→  }
   843→
   844→  transactions = transactions.sort(
   845→    (a, b) => new Date(b.date).getTime() - new Date(a.date).getTime()
   846→  )
   847→
   848→  if (filters.limit) transactions = transactions.slice(0, filters.limit)
   849→
   850→  // Adjuntar info de bancos y exchange
   851→  return transactions.map((t) => ({
   852→    ...t,
   853→    fromBank: t.fromBankId && bankMap.has(t.fromBankId)
   854→      ? {
   855→          id: bankMap.get(t.fromBankId)!.id,
   856→          name: bankMap.get(t.fromBankId)!.name,
   857→          currency: bankMap.get(t.fromBankId)!.currency,
   858→          color: bankMap.get(t.fromBankId)!.color,
   859→        }
   860→      : null,
   861→    toBank: t.toBankId && bankMap.has(t.toBankId)
   862→      ? {
   863→          id: bankMap.get(t.toBankId)!.id,
   864→          name: bankMap.get(t.toBankId)!.name,
   865→          currency: bankMap.get(t.toBankId)!.currency,
   866→          color: bankMap.get(t.toBankId)!.color,
   867→        }
   868→      : null,
   869→    exchange: t.exchangeId && exMap.has(t.exchangeId)
   870→      ? {
   871→          id: exMap.get(t.exchangeId)!.id,
   872→          name: exMap.get(t.exchangeId)!.name,
   873→          shortName: exMap.get(t.exchangeId)!.shortName,
   874→          color: exMap.get(t.exchangeId)!.color,
   875→        }
   876→      : null,
   877→  }))
   878→}

   880→export async function createTransaction(input: TransactionInput): Promise<Transaction> {
   881→  const transactions = await loadTransactions()
   882→  const now = new Date().toISOString()
   883→  const total = input.amount * input.rate
   884→  const exchanges = await loadExchanges()
   885→  const exchange = input.exchangeId ? exchanges.find((e) => e.id === input.exchangeId) : null
   886→
   887→  let fee = input.fee ?? 0
   888→  let feeBreakdown: Transaction['feeBreakdown'] = null
   889→  // Si no se pasó fee explícito pero hay exchange, calcularlo
   890→  if (input.fee === undefined && exchange) {
   891→    const calc = calculateFee(exchange, input.type, total)
   892→    fee = calc.total
   893→    feeBreakdown = {
   894→      baseFee: calc.baseFee,
   895→      discount: calc.discount,
   896→      fixedFee: calc.fixedFee,
   897→      total: calc.total,
   898→    }
   899→  } else if (exchange) {
   900→    // Si el usuario pasó fee manual y hay exchange, reconstruir breakdown aproximado
   901→    const calc = calculateFee(exchange, input.type, total)
   902→    const baseAfterDiscount = fee - calc.fixedFee
   903→    feeBreakdown = {
   904→      baseFee: calc.baseFee,
   905→      discount: Math.max(0, calc.baseFee - baseAfterDiscount),
   906→      fixedFee: calc.fixedFee,
   907→      total: fee,
   908→    }
   909→  }
   910→
   911→  const tx: Transaction = {
   912→    id: uid(),
   913→    type: input.type,
   914→    counterparty: input.counterparty,
   915→    asset: input.asset ?? 'USDT',
   916→    amount: input.amount,
   917→    rate: input.rate,
   918→    total,
   919→    currency: input.currency ?? 'VES',
   920→    fromBankId: input.fromBankId ?? null,
   921→    toBankId: input.toBankId ?? null,
   922→    exchangeId: input.exchangeId ?? null,
   923→    status: input.status ?? 'completada',
   924→    reference: input.reference ?? null,
   925→    fee,
   926→    feeBreakdown,
   927→    netTotal: total - fee,
   928→    captureUrl: input.captureUrl ?? null,
   929→    date: input.date ?? now,
   930→    notes: input.notes ?? null,
   931→    createdAt: now,
   932→    updatedAt: now,
   933→  }
   934→  transactions.push(tx)
   935→  saveTransactions(transactions)
   936→  return tx
   937→}

  1077→  }
  1078→  const topCounterpartes = Array.from(cpMap.entries())
  1079→    .map(([counterparty, v]) => ({ counterparty, ...v }))
  1080→    .sort((a, b) => b.total - a.total)
  1081→    .slice(0, 5)
  1082→
  1083→  // Activos
  1084→  const assetMap = new Map<string, { amount: number; total: number; count: number }>()
  1085→  for (const t of completadas) {
  1086→    const e = assetMap.get(t.asset) ?? { amount: 0, total: 0, count: 0 }
  1087→    e.amount += t.amount
  1088→    e.total += t.total
  1089→    e.count++
  1090→    assetMap.set(t.asset, e)
  1091→  }
  1092→  const activos = Array.from(assetMap.entries())
  1093→    .map(([asset, v]) => ({ asset, ...v }))
  1094→    .sort((a, b) => b.total - a.total)
  1095→
  1096→  // Comisiones por exchange
  1097→  const exMap = new Map<string, {
  1098→    exchangeId: string | null
  1099→    exchangeName: string
  1100→    exchangeColor: string
  1101→    totalFees: number
  1102→    count: number
  1103→    compras: number
  1104→    ventas: number
  1105→  }>()
  1106→  const exInfoMap = new Map(exchanges.map((e) => [e.id, e]))
  1107→  for (const t of completadas) {
  1108→    const key = t.exchangeId ?? '__none__'
  1109→    const ex = t.exchangeId ? exInfoMap.get(t.exchangeId) : null
  1110→    const entry = exMap.get(key) ?? {
  1111→      exchangeId: t.exchangeId ?? null,
  1112→      exchangeName: ex?.name ?? 'Sin exchange',
  1113→      exchangeColor: ex?.color ?? '#64748b',
  1114→      totalFees: 0,
  1115→      count: 0,
  1116→      compras: 0,
  1117→      ventas: 0,
  1118→    }
  1119→    entry.totalFees += t.fee || 0
  1120→    entry.count++
  1121→    if (t.type === 'compra') entry.compras++
  1122→    else entry.ventas++
  1123→    exMap.set(key, entry)
  1124→  }
  1125→  const feesPorExchange = Array.from(exMap.values()).sort((a, b) => b.totalFees - a.totalFees)
  1126→
  1127→  // Evolución mensual (12 meses)
  1128→  const now = new Date()
  1129→  const monthlyMap = new Map<string, { compras: number; ventas: number }>()
  1130→  for (let i = 11; i >= 0; i--) {
  1131→    const d = new Date(now.getFullYear(), now.getMonth() - i, 1)
  1132→    const key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`
  1133→    monthlyMap.set(key, { compras: 0, ventas: 0 })
  1134→  }
  1135→  for (const t of completadas) {
  1136→    const d = new Date(t.date)
  1137→    const key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`
  1138→    const entry = monthlyMap.get(key)
  1139→    if (!entry) continue
  1140→    if (t.type === 'compra') entry.compras += t.total
  1141→    else if (t.type === 'venta') entry.ventas += t.total
  1142→  }
  1143→  const monthly = Array.from(monthlyMap.entries()).map(([month, val]) => ({
  1144→    month,
  1145→    ...val,
  1146→  }))
  1147→
  1148→  return {
  1149→    resumen: {
  1150→      totalCompras,
  1151→      totalVentas,
  1152→      montoCompras,
  1153→      montoVentas,
  1154→      cantidadCompras: compras.length,
  1155→      cantidadVentas: ventas.length,
  1156→      pendientes,
  1157→      totalBanks: banks.length,
  1158→      totalExchanges: exchanges.filter((e) => e.isActive).length,
  1159→      feesCompras,
  1160→      feesVentas,
  1161→      feesTotal,
  1162→      netCompras,
  1163→      netVentas,
  1164→      avgRateCompra,
  1165→      avgRateVenta,
  1166→      activoNeto,
  1167→      gananciaEstimada,
  1168→      gananciaNeta,
  1169→      totalComprasUSD,
  1170→      totalVentasUSD,
  1171→      feesComprasUSD,
  1172→      feesVentasUSD,
  1173→      feesTotalUSD,
  1174→      netComprasUSD,
  1175→      netVentasUSD,
  1176→      gananciaNetaUSD,
  1177→      activoNetoUSD,
  1178→    },
  1179→    topCounterpartes,
  1180→    activos,
  1181→    feesPorExchange,
  1182→    monthly,
  1183→  }
  1184→}

  1208→export async function importData(data: BackupData, mode: 'replace' | 'merge' = 'replace'): Promise<void> {
  1209→  if (mode === 'replace') {
  1210→    if (data.banks) saveBanks(data.banks)
  1211→    if (data.transactions) saveTransactions(data.transactions.map(migrateTx))
  1212→    if (data.exchanges) saveExchanges(data.exchanges)
  1213→  } else {
  1214→    const existingBanks = await loadBanks()
  1215→    const existingTx = (await loadTransactions()).map(migrateTx)
  1216→    const existingEx = await loadExchanges()
  1217→    const existingBankIds = new Set(existingBanks.map((b) => b.id))
  1218→    const existingTxIds = new Set(existingTx.map((t) => t.id))
  1219→    const existingExIds = new Set(existingEx.map((e) => e.id))
  1220→    const newBanks = (data.banks ?? []).filter((b) => !existingBankIds.has(b.id))
  1221→    const newTx = (data.transactions ?? []).map(migrateTx).filter((t) => !existingTxIds.has(t.id))
  1222→    const newEx = (data.exchanges ?? []).filter((e) => !existingExIds.has(e.id))
  1223→    saveBanks([...existingBanks, ...newBanks])
  1224→    saveTransactions([...existingTx, ...newTx])
  1225→    saveExchanges([...existingEx, ...newEx])
  1226→  }
  1227→}
