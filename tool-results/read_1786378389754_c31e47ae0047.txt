    49→    49→function isBrowser(): boolean {

    53→    53→function readJSONSync<T>(key: string, fallback: T): T {
    54→    54→  if (!isBrowser()) return fallback
    55→    55→  try {
    56→    56→    const raw = window.localStorage.getItem(key)
    57→    57→    if (!raw) return fallback
    58→    58→    return JSON.parse(raw) as T
    59→    59→  } catch {
    60→    60→    return fallback
    61→    61→  }
    62→    62→}
    63→    63→
    64→    64→async function readJSON<T>(key: string, fallback: T): Promise<T> {
    65→    65→  if (!isBrowser()) return fallback
    66→    66→  try {
    67→    67→    const raw = window.localStorage.getItem(key)
    68→    68→    if (!raw) return fallback
    69→    69→    // Si la entrada está cifrada y tenemos DEK, descifrar
    70→    70→    if (raw.startsWith(ENC_PREFIX)) {
    71→    71→      if (!activeDek) {
    72→    72→        // App bloqueada o sin DEK: no podemos descifrar
    73→    73→        return fallback
    74→    74→      }
    75→    75→      const payload = raw.slice(ENC_PREFIX.length)
    76→    76→      return decryptJson<T>(payload, activeDek)
    77→    77→    }
    78→    78→    return JSON.parse(raw) as T
    79→    79→  } catch {
    80→    80→    return fallback
    81→    81→  }
    82→    82→}
    83→    83→
    84→    84→function writeJSON(key: string, value: unknown): void {
    85→    85→  if (!isBrowser()) return
    86→    86→  if (ENCRYPTED_KEYS.has(key) && activeDek) {
    87→    87→    // Cifrar antes de guardar
    88→    88→    encryptJson(value, activeDek)
    89→    89→      .then((payload) => {
    90→    90→        window.localStorage.setItem(key, ENC_PREFIX + payload)
    91→    91→      })
    92→    92→      .catch(() => {
    93→    93→        // Si falla el cifrado, no escribir (mejor perder el dato que guardarlo en claro)
    94→    94→      })
    95→    95→    return
    96→    96→  }
    97→    97→  window.localStorage.setItem(key, JSON.stringify(value))
    98→    98→}

   100→   100→/**
   101→   101→ * Migra entradas en claro a cifradas cuando se activa el PIN.
   102→   102→ * Lee todas las claves sensibles y las re-escribe cifradas.
   103→   103→ */
   104→   104→export async function encryptAllStorage(dek: CryptoKey) {
   105→   105→  if (!isBrowser()) return
   106→   106→  for (const key of ENCRYPTED_KEYS) {
   107→   107→    const raw = window.localStorage.getItem(key)
   108→   108→    if (!raw) continue
   109→   109→    if (raw.startsWith(ENC_PREFIX)) continue // ya cifrada
   110→   110→    try {
   111→   111→      const parsed = JSON.parse(raw)
   112→   112→      const payload = await encryptJson(parsed, dek)
   113→   113→      window.localStorage.setItem(key, ENC_PREFIX + payload)
   114→   114→    } catch {
   115→   115→      // ignorar entradas inválidas
   116→   116→    }
   117→   117→  }
   118→   118→}
   119→   119→
   120→   120→/**
   121→   121→ * Migra entradas cifradas a claro cuando se desactiva el PIN.
   122→   122→ * Requiere DEK para descifrar antes de quitarla.
   123→   123→ */
   124→   124→export async function decryptAllStorage(dek: CryptoKey) {
   125→   125→  if (!isBrowser()) return
   126→   126→  for (const key of ENCRYPTED_KEYS) {
   127→   127→    const raw = window.localStorage.getItem(key)
   128→   128→    if (!raw) continue
   129→   129→    if (!raw.startsWith(ENC_PREFIX)) continue // ya en claro
   130→   130→    try {
   131→   131→      const payload = raw.slice(ENC_PREFIX.length)
   132→   132→      const parsed = await decryptJson<unknown>(payload, dek)
   133→   133→      window.localStorage.setItem(key, JSON.stringify(parsed))
   134→   134→    } catch {
   135→   135→      // ignorar
   136→   136→    }
   137→   137→  }
   138→   138→}

   140→   140→function uid(): string {
   141→   141→  if (typeof crypto !== 'undefined' && crypto.randomUUID) {
   142→   142→    return crypto.randomUUID()
   143→   143→  }
   144→   144→  return Date.now().toString(36) + Math.random().toString(36).slice(2, 10)
   145→   145→}

   147→   147→// Pequeño EventBus para notificar cambios entre componentes
   148→   148→type Listener = () => void
   149→   149→const listeners = new Set<Listener>()
   150→   150→
   151→   151→export function subscribe(listener: Listener): () => void {
   152→   152→  listeners.add(listener)

   160→   160→// Inicialización (solo una vez)
   161→   161→let initialized = false
   162→   162→function ensureInit() {
   163→   163→  if (initialized || !isBrowser()) return
   164→   164→  initialized = true
   165→   165→  const version = window.localStorage.getItem(VERSION_KEY)
   166→   166→  if (version !== CURRENT_VERSION) {
   167→   167→    // Si vienen de una versión anterior, migrar exchanges existentes
   168→   168→    // v3 → v4: añadimos Bybit, KuCoin y Paxful si no existen
   169→   169→    if (version === '3') {
   170→   170→      const existing = readJSONSync<Exchange[]>(EXCHANGES_KEY, [])
   171→   171→      const migrated = existing.map(migrateExchange)
   172→   172→      const existingNames = new Set(migrated.map((e) => e.name.toLowerCase()))
   173→   173→      const additions = defaultExchanges().filter(
   174→   174→        (e) => !existingNames.has(e.name.toLowerCase())
   175→   175→      )
   176→   176→      // Solo añadimos los nuevos (Bybit, KuCoin, Paxful) - no duplicamos los antiguos
   177→   177→      const newOnes = additions.filter((e) =>
   178→   178→        ['Bybit P2P', 'KuCoin P2P', 'Paxful', 'Bybit Spot', 'KuCoin Spot'].includes(e.name)
   179→   179→      )
   180→   180→      writeJSON(EXCHANGES_KEY, [...migrated, ...newOnes])
   181→   181→    }
   182→   182→    // v4 → v5: añadimos P2p.me si no existe
   183→   183→    if (version === '4') {
   184→   184→      const existing = readJSONSync<Exchange[]>(EXCHANGES_KEY, []).map(migrateExchange)
   185→   185→      const existingNames = new Set(existing.map((e) => e.name.toLowerCase()))
   186→   186→      const p2pme = defaultExchanges().find((e) => e.name === 'P2p.me')!
   187→   187→      if (!existingNames.has('p2p.me')) {
   188→   188→        writeJSON(EXCHANGES_KEY, [...existing, p2pme])
   189→   189→      }
   190→   190→    }
   191→   191→    // Si es primera instalación (sin versión), sembrar exchanges por defecto
   192→   192→    if (!version) {
   193→   193→      const defaults = defaultExchanges()
   194→   194→      writeJSON(EXCHANGES_KEY, defaults)
   195→   195→    }
   196→   196→    window.localStorage.setItem(VERSION_KEY, CURRENT_VERSION)
   197→   197→  }
   198→   198→}

   500→   500→function computeBankBalance(bank: Bank, transactions: Transaction[]): {
   501→   501→  balance: number
   502→   502→  totalEntradas: number
   503→   503→  totalSalidas: number
   504→   504→  countFrom: number
   505→   505→  countTo: number
   506→   506→} {
   507→   507→  let entrada = 0
   508→   508→  let salida = 0
   509→   509→  let countFrom = 0
   510→   510→  let countTo = 0
   511→   511→  for (const t of transactions) {
   512→   512→    if (t.status !== 'completada') continue
   513→   513→    // Para el cálculo del balance del banco usamos el neto (después de comisiones)
   514→   514→    // porque la comisión es un costo que se descuenta del monto que efectivamente entra/sale.
   515→   515→    const neto = t.netTotal ?? t.total
   516→   516→    if (t.toBankId === bank.id) {
   517→   517→      entrada += neto
   518→   518→      countTo++
   519→   519→    }
   520→   520→    if (t.fromBankId === bank.id) {
   521→   521→      salida += neto
   522→   522→      countFrom++
   523→   523→    }
   524→   524→  }
   525→   525→  return {
   526→   526→    balance: bank.initialBalance + entrada - salida,
   527→   527→    totalEntradas: entrada,
   528→   528→    totalSalidas: salida,
   529→   529→    countFrom,
   530→   530→    countTo,
   531→   531→  }
   532→   532→}

   703→   703→/**
   704→   704→ * Selecciona el tier aplicable para un monto dado.
   705→   705→ * Si no hay tiers definidos, retorna null (usar feeType/feeValue base).
   706→   706→ * Si hay tiers, retorna el de mayor minAmount <= totalFiat.
   707→   707→ */
   708→   708→function pickTier(tiers: FeeTier[], totalFiat: number): FeeTier | null {
   709→   709→  if (!tiers || tiers.length === 0) return null
   710→   710→  // Ordenar de mayor a menor minAmount y elegir el primero que aplique
   711→   711→  const sorted = [...tiers].sort((a, b) => b.minAmount - a.minAmount)
   712→   712→  for (const tier of sorted) {
   713→   713→    if (totalFiat >= tier.minAmount) return tier
   714→   714→  }
   715→   715→  // Fallback al tier de menor minAmount (debería ser 0 normalmente)
   716→   716→  return sorted[sorted.length - 1]
   717→   717→}

   719→   719→/**
   720→   720→ * Calcula la comisión que aplicaría un exchange a una operación dada.
   721→   721→ * Soporta:
   722→   722→ *   - Tiers escalonados (si están definidos, reemplazan el fee base)
   723→   723→ *   - Descuento VIP/BNB (porcentaje descontado de la comisión variable, NO de la fija)
   724→   724→ *   - Comisión fija adicional
   725→   725→ * Retorna desglose y total.
   726→   726→ */
   727→   727→export function calculateFee(
   728→   728→  exchange: Exchange | null | undefined,
   729→   729→  type: TransactionType,
   730→   730→  totalFiat: number
   731→   731→): { baseFee: number; discount: number; fixedFee: number; total: number } {
   732→   732→  if (!exchange) return { baseFee: 0, discount: 0, fixedFee: 0, total: 0 }
   733→   733→
   734→   734→  const tiers = type === 'compra' ? exchange.buyTiers : exchange.sellTiers
   735→   735→  const fallbackType = type === 'compra' ? exchange.buyFeeType : exchange.sellFeeType
   736→   736→  const fallbackValue = type === 'compra' ? exchange.buyFeeValue : exchange.sellFeeValue
   737→   737→
   738→   738→  let baseFee: number
   739→   739→  const tier = pickTier(tiers, totalFiat)
   740→   740→  if (tier) {
   741→   741→    baseFee = tier.feeType === 'percent'
   742→   742→      ? (totalFiat * tier.feeValue) / 100
   743→   743→      : tier.feeValue
   744→   744→  } else {
   745→   745→    baseFee = fallbackType === 'percent'
   746→   746→      ? (totalFiat * fallbackValue) / 100
   747→   747→      : fallbackValue
   748→   748→  }
   749→   749→
   750→   750→  // Descuento VIP/BNB se aplica SOLO sobre la comisión variable (no sobre la fija)
   751→   751→  const discountPercent = exchange.discountPercent || 0
   752→   752→  const discount = (baseFee * discountPercent) / 100
   753→   753→  const baseFeeAfterDiscount = baseFee - discount
   754→   754→
   755→   755→  const fixedFee = exchange.fixedFee || 0
   756→   756→  const total = baseFeeAfterDiscount + fixedFee
   757→   757→
   758→   758→  return {
   759→   759→    baseFee,
   760→   760→    discount,
   761→   761→    fixedFee,
   762→   762→    total,
   763→   763→  }
   764→   764→}

   832→   832→  if (filters.from) {

   856→   856→        }
   857→   857→      : null,
   858→   858→    toBank: t.toBankId && bankMap.has(t.toBankId)
   859→   859→      ? {
   860→   860→          id: bankMap.get(t.toBankId)!.id,
   861→   861→          name: bankMap.get(t.toBankId)!.name,
   862→   862→          currency: bankMap.get(t.toBankId)!.currency,
   863→   863→          color: bankMap.get(t.toBankId)!.color,
   864→   864→        }
   865→   865→      : null,
   866→   866→    exchange: t.exchangeId && exMap.has(t.exchangeId)
   867→   867→      ? {
   868→   868→          id: exMap.get(t.exchangeId)!.id,
   869→   869→          name: exMap.get(t.exchangeId)!.name,
   870→   870→          shortName: exMap.get(t.exchangeId)!.shortName,
   871→   871→          color: exMap.get(t.exchangeId)!.color,
   872→   872→        }
   873→   873→      : null,
   874→   874→  }))

   877→   877→export async function createTransaction(input: TransactionInput): Promise<Transaction> {
   878→   878→  const transactions = await loadTransactions()
   879→   879→  const now = new Date().toISOString()
   880→   880→  const total = input.amount * input.rate
   881→   881→  const exchanges = await loadExchanges()
   882→   882→  const exchange = input.exchangeId ? exchanges.find((e) => e.id === input.exchangeId) : null
   883→   883→
   884→   884→  let fee = input.fee ?? 0
   885→   885→  let feeBreakdown: Transaction['feeBreakdown'] = null
   886→   886→  // Si no se pasó fee explícito pero hay exchange, calcularlo
   887→   887→  if (input.fee === undefined && exchange) {
   888→   888→    const calc = calculateFee(exchange, input.type, total)
   889→   889→    fee = calc.total
   890→   890→    feeBreakdown = {
   891→   891→      baseFee: calc.baseFee,
   892→   892→      discount: calc.discount,
   893→   893→      fixedFee: calc.fixedFee,
   894→   894→      total: calc.total,
   895→   895→    }
   896→   896→  } else if (exchange) {
   897→   897→    // Si el usuario pasó fee manual y hay exchange, reconstruir breakdown aproximado
   898→   898→    const calc = calculateFee(exchange, input.type, total)
   899→   899→    const baseAfterDiscount = fee - calc.fixedFee
   900→   900→    feeBreakdown = {
   901→   901→      baseFee: calc.baseFee,
   902→   902→      discount: Math.max(0, calc.baseFee - baseAfterDiscount),
   903→   903→      fixedFee: calc.fixedFee,
   904→   904→      total: fee,
   905→   905→    }
   906→   906→  }
   907→   907→
   908→   908→  const tx: Transaction = {
   909→   909→    id: uid(),
   910→   910→    type: input.type,
   911→   911→    counterparty: input.counterparty,
   912→   912→    asset: input.asset ?? 'USDT',
   913→   913→    amount: input.amount,
   914→   914→    rate: input.rate,
   915→   915→    total,

   935→   935→export async function updateTransaction(
   936→   936→  id: string,
   937→   937→  input: Partial<TransactionInput>
   938→   938→): Promise<Transaction> {
   939→   939→  const transactions = await loadTransactions()
   940→   940→  const idx = transactions.findIndex((t) => t.id === id)
   941→   941→  if (idx === -1) throw new Error('Transacción no encontrada')
   942→   942→  const existing = migrateTx(transactions[idx])
   943→   943→  const finalAmount = input.amount ?? existing.amount
   944→   944→  const finalRate = input.rate ?? existing.rate
   945→   945→  const total = finalAmount * finalRate
   946→   946→  const exchanges = await loadExchanges()
   947→   947→  const exchangeId = input.exchangeId !== undefined ? input.exchangeId ?? null : existing.exchangeId
   948→   948→  const exchange = exchangeId ? exchanges.find((e) => e.id === exchangeId) : null
   949→   949→  const type = input.type ?? existing.type
   950→   950→
   951→   951→  let fee: number
   952→   952→  let feeBreakdown: Transaction['feeBreakdown'] = existing.feeBreakdown ?? null
   953→   953→
   954→   954→  if (input.fee !== undefined) {
   955→   955→    // Usuario envió fee explícito
   956→   956→    fee = input.fee
   957→   957→    if (exchange) {
   958→   958→      const calc = calculateFee(exchange, type, total)
   959→   959→      const baseAfterDiscount = fee - calc.fixedFee
   960→   960→      feeBreakdown = {
   961→   961→        baseFee: calc.baseFee,
   962→   962→        discount: Math.max(0, calc.baseFee - baseAfterDiscount),
   963→   963→        fixedFee: calc.fixedFee,
   964→   964→        total: fee,
   965→   965→      }
   966→   966→    } else {
   967→   967→      feeBreakdown = { baseFee: fee, discount: 0, fixedFee: 0, total: fee }
   968→   968→    }
   969→   969→  } else if (input.exchangeId !== undefined || input.amount !== undefined || input.rate !== undefined || input.type !== undefined) {
   970→   970→    // Cambió algo que afecta el cálculo — recalcular si hay exchange
   971→   971→    if (exchange) {
   972→   972→      const calc = calculateFee(exchange, type, total)
   973→   973→      fee = calc.total
   974→   974→      feeBreakdown = {
   975→   975→        baseFee: calc.baseFee,
   976→   976→        discount: calc.discount,
   977→   977→        fixedFee: calc.fixedFee,
   978→   978→        total: calc.total,
   979→   979→      }
   980→   980→    } else {
   981→   981→      fee = existing.fee
   982→   982→      feeBreakdown = null
   983→   983→    }
   984→   984→  } else {
   985→   985→    fee = existing.fee
   986→   986→  }
   987→   987→
   988→   988→  const updated: Transaction = {
   989→   989→    ...existing,
   990→   990→    type,
   991→   991→    counterparty: input.counterparty ?? existing.counterparty,
   992→   992→    asset: input.asset ?? existing.asset,
   993→   993→    amount: finalAmount,
   994→   994→    rate: finalRate,
   995→   995→    total,
   996→   996→    currency: input.currency ?? existing.currency,
   997→   997→    fromBankId: input.fromBankId !== undefined ? input.fromBankId ?? null : existing.fromBankId,
   998→   998→    toBankId: input.toBankId !== undefined ? input.toBankId ?? null : existing.toBankId,
   999→   999→    exchangeId,
  1000→  1000→    status: input.status ?? existing.status,
  1001→  1001→    reference: input.reference !== undefined ? input.reference ?? null : existing.reference,
  1002→  1002→    fee,
  1003→  1003→    feeBreakdown,
  1004→  1004→    netTotal: total - fee,
  1005→  1005→    date: input.date ?? existing.date,
  1006→  1006→    notes: input.notes !== undefined ? input.notes ?? null : existing.notes,
  1007→  1007→    updatedAt: new Date().toISOString(),
  1008→  1008→  }
  1009→  1009→  transactions[idx] = updated
  1010→  1010→  saveTransactions(transactions)
  1011→  1011→  return updated
  1012→  1012→}

  1023→  1023→export async function getStats(): Promise<Stats> {
  1024→  1024→  const transactions = (await loadTransactions()).map(migrateTx)
  1025→  1025→  const completadas = transactions.filter((t) => t.status === 'completada')
  1026→  1026→  const banks = await loadBanks()
  1027→  1027→  const exchanges = await loadExchanges()
  1028→  1028→
  1029→  1029→  const compras = completadas.filter((t) => t.type === 'compra')
  1030→  1030→  const ventas = completadas.filter((t) => t.type === 'venta')
  1031→  1031→  const pendientes = transactions.filter((t) => t.status === 'pendiente').length
  1032→  1032→
  1033→  1033→  const totalCompras = compras.reduce((s, t) => s + t.total, 0)
  1034→  1034→  const totalVentas = ventas.reduce((s, t) => s + t.total, 0)
  1035→  1035→  const montoCompras = compras.reduce((s, t) => s + t.amount, 0)
  1036→  1036→  const montoVentas = ventas.reduce((s, t) => s + t.amount, 0)
  1037→  1037→  const feesCompras = compras.reduce((s, t) => s + (t.fee || 0), 0)
  1038→  1038→  const feesVentas = ventas.reduce((s, t) => s + (t.fee || 0), 0)
  1039→  1039→  const feesTotal = feesCompras + feesVentas
  1040→  1040→  const netCompras = totalCompras - feesCompras
  1041→  1041→  const netVentas = totalVentas - feesVentas
  1042→  1042→
  1043→  1043→  const avgRateCompra = montoCompras > 0 ? totalCompras / montoCompras : 0
  1044→  1044→  const avgRateVenta = montoVentas > 0 ? totalVentas / montoVentas : 0
  1045→  1045→  const activoNeto = montoCompras - montoVentas
  1046→  1046→  const volumenCruzado = Math.min(montoCompras, montoVentas)
  1047→  1047→  const gananciaBruta =
  1048→  1048→    volumenCruzado > 0 ? (avgRateVenta - avgRateCompra) * volumenCruzado : 0
  1049→  1049→  const gananciaEstimada = gananciaBruta // bruta (sin comisiones)
  1050→  1050→  const gananciaNeta = gananciaBruta - feesTotal // neta (después de comisiones)
  1051→  1051→
  1052→  1052→  // Top contrapartes
  1053→  1053→  const cpMap = new Map<string, { total: number; amount: number; count: number }>()
  1054→  1054→  for (const t of completadas) {
  1055→  1055→    const e = cpMap.get(t.counterparty) ?? { total: 0, amount: 0, count: 0 }
  1056→  1056→    e.total += t.total
  1057→  1057→    e.amount += t.amount
  1058→  1058→    e.count++
  1059→  1059→    cpMap.set(t.counterparty, e)
  1060→  1060→  }
  1061→  1061→  const topCounterpartes = Array.from(cpMap.entries())
  1062→  1062→    .map(([counterparty, v]) => ({ counterparty, ...v }))
  1063→  1063→    .sort((a, b) => b.total - a.total)
  1064→  1064→    .slice(0, 5)
  1065→  1065→
  1066→  1066→  // Activos
  1067→  1067→  const assetMap = new Map<string, { amount: number; total: number; count: number }>()
  1068→  1068→  for (const t of completadas) {
  1069→  1069→    const e = assetMap.get(t.asset) ?? { amount: 0, total: 0, count: 0 }
  1070→  1070→    e.amount += t.amount
  1071→  1071→    e.total += t.total
  1072→  1072→    e.count++
  1073→  1073→    assetMap.set(t.asset, e)
  1074→  1074→  }
  1075→  1075→  const activos = Array.from(assetMap.entries())
  1076→  1076→    .map(([asset, v]) => ({ asset, ...v }))
  1077→  1077→    .sort((a, b) => b.total - a.total)
  1078→  1078→
  1079→  1079→  // Comisiones por exchange
  1080→  1080→  const exMap = new Map<string, {
  1081→  1081→    exchangeId: string | null
  1082→  1082→    exchangeName: string
  1083→  1083→    exchangeColor: string
  1084→  1084→    totalFees: number
  1085→  1085→    count: number
  1086→  1086→    compras: number
  1087→  1087→    ventas: number
  1088→  1088→  }>()
  1089→  1089→  const exInfoMap = new Map(exchanges.map((e) => [e.id, e]))
  1090→  1090→  for (const t of completadas) {
  1091→  1091→    const key = t.exchangeId ?? '__none__'
  1092→  1092→    const ex = t.exchangeId ? exInfoMap.get(t.exchangeId) : null
  1093→  1093→    const entry = exMap.get(key) ?? {
  1094→  1094→      exchangeId: t.exchangeId ?? null,
  1095→  1095→      exchangeName: ex?.name ?? 'Sin exchange',
  1096→  1096→      exchangeColor: ex?.color ?? '#64748b',
  1097→  1097→      totalFees: 0,
  1098→  1098→      count: 0,
  1099→  1099→      compras: 0,
  1100→  1100→      ventas: 0,
  1101→  1101→    }
  1102→  1102→    entry.totalFees += t.fee || 0
  1103→  1103→    entry.count++
  1104→  1104→    if (t.type === 'compra') entry.compras++
  1105→  1105→    else entry.ventas++
  1106→  1106→    exMap.set(key, entry)
  1107→  1107→  }
  1108→  1108→  const feesPorExchange = Array.from(exMap.values()).sort((a, b) => b.totalFees - a.totalFees)
  1109→  1109→
  1110→  1110→  // Evolución mensual (12 meses)
  1111→  1111→  const now = new Date()
  1112→  1112→  const monthlyMap = new Map<string, { compras: number; ventas: number }>()
  1113→  1113→  for (let i = 11; i >= 0; i--) {
  1114→  1114→    const d = new Date(now.getFullYear(), now.getMonth() - i, 1)
  1115→  1115→    const key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`
  1116→  1116→    monthlyMap.set(key, { compras: 0, ventas: 0 })
  1117→  1117→  }
  1118→  1118→  for (const t of completadas) {
  1119→  1119→    const d = new Date(t.date)
  1120→  1120→    const key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`
  1121→  1121→    const entry = monthlyMap.get(key)
  1122→  1122→    if (!entry) continue
  1123→  1123→    if (t.type === 'compra') entry.compras += t.total
  1124→  1124→    else if (t.type === 'venta') entry.ventas += t.total
  1125→  1125→  }
  1126→  1126→  const monthly = Array.from(monthlyMap.entries()).map(([month, val]) => ({
  1127→  1127→    month,
  1128→  1128→    ...val,
  1129→  1129→  }))
  1130→  1130→
  1131→  1131→  return {
  1132→  1132→    resumen: {
  1133→  1133→      totalCompras,
  1134→  1134→      totalVentas,
  1135→  1135→      montoCompras,
  1136→  1136→      montoVentas,
  1137→  1137→      cantidadCompras: compras.length,
  1138→  1138→      cantidadVentas: ventas.length,
  1139→  1139→      pendientes,
  1140→  1140→      totalBanks: banks.length,
  1141→  1141→      totalExchanges: exchanges.filter((e) => e.isActive).length,
  1142→  1142→      feesCompras,
  1143→  1143→      feesVentas,
  1144→  1144→      feesTotal,
  1145→  1145→      netCompras,
  1146→  1146→      netVentas,
  1147→  1147→      avgRateCompra,
  1148→  1148→      avgRateVenta,
  1149→  1149→      activoNeto,
  1150→  1150→      gananciaEstimada,
  1151→  1151→      gananciaNeta,
  1152→  1152→    },
  1153→  1153→    topCounterpartes,
  1154→  1154→    activos,
  1155→  1155→    feesPorExchange,
  1156→  1156→    monthly,
  1157→  1157→  }
  1158→  1158→}

  1194→  1194→    const newBanks = (data.banks ?? []).filter((b) => !existingBankIds.has(b.id))
  1195→  1195→    const newTx = (data.transactions ?? []).map(migrateTx).filter((t) => !existingTxIds.has(t.id))
  1196→  1196→    const newEx = (data.exchanges ?? []).filter((e) => !existingExIds.has(e.id))
  1197→  1197→    saveBanks([...existingBanks, ...newBanks])

  1200→  1200→  }
