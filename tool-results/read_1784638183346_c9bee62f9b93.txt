    49→function isBrowser(): boolean {

    53→function readJSONSync<T>(key: string, fallback: T): T {
    54→  if (!isBrowser()) return fallback
    55→  try {
    56→    const raw = window.localStorage.getItem(key)
    57→    if (!raw) return fallback
    58→    return JSON.parse(raw) as T
    59→  } catch {
    60→    return fallback
    61→  }
    62→}
    63→
    64→async function readJSON<T>(key: string, fallback: T): Promise<T> {
    65→  if (!isBrowser()) return fallback
    66→  try {
    67→    const raw = window.localStorage.getItem(key)
    68→    if (!raw) return fallback
    69→    // Si la entrada está cifrada y tenemos DEK, descifrar
    70→    if (raw.startsWith(ENC_PREFIX)) {
    71→      if (!activeDek) {
    72→        // App bloqueada o sin DEK: no podemos descifrar
    73→        return fallback
    74→      }
    75→      const payload = raw.slice(ENC_PREFIX.length)
    76→      return decryptJson<T>(payload, activeDek)
    77→    }
    78→    return JSON.parse(raw) as T
    79→  } catch {
    80→    return fallback
    81→  }
    82→}
    83→
    84→function writeJSON(key: string, value: unknown): void {
    85→  if (!isBrowser()) return
    86→  if (ENCRYPTED_KEYS.has(key) && activeDek) {
    87→    // Cifrar antes de guardar
    88→    encryptJson(value, activeDek)
    89→      .then((payload) => {
    90→        window.localStorage.setItem(key, ENC_PREFIX + payload)
    91→      })
    92→      .catch(() => {
    93→        // Si falla el cifrado, no escribir (mejor perder el dato que guardarlo en claro)
    94→      })
    95→    return
    96→  }
    97→  window.localStorage.setItem(key, JSON.stringify(value))
    98→}

   100→/**
   101→ * Migra entradas en claro a cifradas cuando se activa el PIN.
   102→ * Lee todas las claves sensibles y las re-escribe cifradas.
   103→ */
   104→export async function encryptAllStorage(dek: CryptoKey) {
   105→  if (!isBrowser()) return
   106→  for (const key of ENCRYPTED_KEYS) {
   107→    const raw = window.localStorage.getItem(key)
   108→    if (!raw) continue
   109→    if (raw.startsWith(ENC_PREFIX)) continue // ya cifrada
   110→    try {
   111→      const parsed = JSON.parse(raw)
   112→      const payload = await encryptJson(parsed, dek)
   113→      window.localStorage.setItem(key, ENC_PREFIX + payload)
   114→    } catch {
   115→      // ignorar entradas inválidas
   116→    }
   117→  }
   118→}
   119→
   120→/**
   121→ * Migra entradas cifradas a claro cuando se desactiva el PIN.
   122→ * Requiere DEK para descifrar antes de quitarla.
   123→ */
   124→export async function decryptAllStorage(dek: CryptoKey) {
   125→  if (!isBrowser()) return
   126→  for (const key of ENCRYPTED_KEYS) {
   127→    const raw = window.localStorage.getItem(key)
   128→    if (!raw) continue
   129→    if (!raw.startsWith(ENC_PREFIX)) continue // ya en claro
   130→    try {
   131→      const payload = raw.slice(ENC_PREFIX.length)
   132→      const parsed = await decryptJson<unknown>(payload, dek)
   133→      window.localStorage.setItem(key, JSON.stringify(parsed))
   134→    } catch {
   135→      // ignorar
   136→    }
   137→  }
   138→}

   140→function uid(): string {
   141→  if (typeof crypto !== 'undefined' && crypto.randomUUID) {
   142→    return crypto.randomUUID()
   143→  }
   144→  return Date.now().toString(36) + Math.random().toString(36).slice(2, 10)
   145→}

   149→const listeners = new Set<Listener>()

   151→export function subscribe(listener: Listener): () => void {

   160→// Inicialización (solo una vez)
   161→let initialized = false
   162→function ensureInit() {
   163→  if (initialized || !isBrowser()) return
   164→  initialized = true
   165→  const version = window.localStorage.getItem(VERSION_KEY)
   166→  if (version !== CURRENT_VERSION) {
   167→    // Si vienen de una versión anterior, migrar exchanges existentes
   168→    if (version === '2' || version === '3') {
   169→      const existing = readJSONSync<Exchange[]>(EXCHANGES_KEY, [])
   170→      const migrated = existing.map(migrateExchange)
   171→      // En la migración v3 → v4 añadimos Bybit, KuCoin y Paxful si no existen
   172→      if (version === '3') {
   173→        const existingNames = new Set(migrated.map((e) => e.name.toLowerCase()))
   174→        const additions = defaultExchanges().filter(
   175→          (e) => !existingNames.has(e.name.toLowerCase())
   176→        )
   177→        // Solo añadimos los nuevos (Bybit, KuCoin, Paxful) - no duplicamos los antiguos
   178→        const newOnes = additions.filter((e) =>
   179→          ['Bybit P2P', 'KuCoin P2P', 'Paxful', 'Bybit Spot', 'KuCoin Spot'].includes(e.name)
   180→        )
   181→        writeJSON(EXCHANGES_KEY, [...migrated, ...newOnes])
   182→      } else {
   183→        writeJSON(EXCHANGES_KEY, migrated)
   184→      }
   185→    }
   186→    // Si es primera instalación (sin versión), sembrar exchanges por defecto
   187→    if (!version) {
   188→      const defaults = defaultExchanges()
   189→      writeJSON(EXCHANGES_KEY, defaults)
   190→    }
   191→    window.localStorage.setItem(VERSION_KEY, CURRENT_VERSION)
   192→  }
   193→}

   465→async function loadBanks(): Promise<Bank[]> {
   466→  ensureInit()

   470→function saveBanks(banks: Bank[]): void {
   471→  writeJSON(BANKS_KEY, banks)
   472→  notify()
   473→}

   475→function computeBankBalance(bank: Bank, transactions: Transaction[]): {
   476→  balance: number
   477→  totalEntradas: number
   478→  totalSalidas: number
   479→  countFrom: number
   480→  countTo: number
   481→} {
   482→  let entrada = 0
   483→  let salida = 0
   484→  let countFrom = 0
   485→  let countTo = 0
   486→  for (const t of transactions) {
   487→    if (t.status !== 'completada') continue
   488→    // Para el cálculo del balance del banco usamos el neto (después de comisiones)
   489→    // porque la comisión es un costo que se descuenta del monto que efectivamente entra/sale.
   490→    const neto = t.netTotal ?? t.total
   491→    if (t.toBankId === bank.id) {
   492→      entrada += neto
   493→      countTo++
   494→    }
   495→    if (t.fromBankId === bank.id) {
   496→      salida += neto
   497→      countFrom++
   498→    }
   499→  }
   500→  return {
   501→    balance: bank.initialBalance + entrada - salida,
   502→    totalEntradas: entrada,
   503→    totalSalidas: salida,
   504→    countFrom,
   505→    countTo,
   506→  }
   507→}

   509→export async function listBanks(): Promise<Bank[]> {
   510→  const banks = await loadBanks()
   511→  const transactions = await loadTransactions()
   512→  return banks
   513→    .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime())
   514→    .map((b) => {
   515→      const computed = computeBankBalance(b, transactions)
   516→      return {
   517→        ...b,
   518→        balance: computed.balance,
   519→        totalEntradas: computed.totalEntradas,
   520→        totalSalidas: computed.totalSalidas,
   521→        _count: {
   522→          transactionsFrom: computed.countFrom,
   523→          transactionsTo: computed.countTo,
   524→        },
   525→      }
   526→    })
   527→}

   529→export async function createBank(input: BankInput): Promise<Bank> {
   530→  const banks = await loadBanks()
   531→  const now = new Date().toISOString()
   532→  const bank: Bank = {
   533→    id: uid(),
   534→    name: input.name,
   535→    accountType: input.accountType,
   536→    accountNumber: input.accountNumber ?? null,
   537→    holderName: input.holderName ?? null,
   538→    currency: input.currency,
   539→    initialBalance: input.initialBalance,
   540→    isActive: input.isActive ?? true,
   541→    color: input.color,
   542→    notes: input.notes ?? null,
   543→    createdAt: now,
   544→    updatedAt: now,
   545→  }
   546→  banks.push(bank)
   547→  saveBanks(banks)
   548→  return bank
   549→}

   551→export async function updateBank(id: string, input: Partial<BankInput>): Promise<Bank> {
   552→  const banks = await loadBanks()
   553→  const idx = banks.findIndex((b) => b.id === id)
   554→  if (idx === -1) throw new Error('Banco no encontrado')
   555→  const updated: Bank = {

   678→/**
   679→ * Selecciona el tier aplicable para un monto dado.
   680→ * Si no hay tiers definidos, retorna null (usar feeType/feeValue base).
   681→ * Si hay tiers, retorna el de mayor minAmount <= totalFiat.
   682→ */
   683→function pickTier(tiers: FeeTier[], totalFiat: number): FeeTier | null {
   684→  if (!tiers || tiers.length === 0) return null
   685→  // Ordenar de mayor a menor minAmount y elegir el primero que aplique
   686→  const sorted = [...tiers].sort((a, b) => b.minAmount - a.minAmount)
   687→  for (const tier of sorted) {
   688→    if (totalFiat >= tier.minAmount) return tier
   689→  }
   690→  // Fallback al tier de menor minAmount (debería ser 0 normalmente)
   691→  return sorted[sorted.length - 1]
   692→}

   694→/**
   695→ * Calcula la comisión que aplicaría un exchange a una operación dada.
   696→ * Soporta:
   697→ *   - Tiers escalonados (si están definidos, reemplazan el fee base)
   698→ *   - Descuento VIP/BNB (porcentaje descontado de la comisión variable, NO de la fija)
   699→ *   - Comisión fija adicional
   700→ * Retorna desglose y total.
   701→ */
   702→export function calculateFee(
   703→  exchange: Exchange | null | undefined,
   704→  type: TransactionType,
   705→  totalFiat: number
   706→): { baseFee: number; discount: number; fixedFee: number; total: number } {
   707→  if (!exchange) return { baseFee: 0, discount: 0, fixedFee: 0, total: 0 }
   708→
   709→  const tiers = type === 'compra' ? exchange.buyTiers : exchange.sellTiers
   710→  const fallbackType = type === 'compra' ? exchange.buyFeeType : exchange.sellFeeType
   711→  const fallbackValue = type === 'compra' ? exchange.buyFeeValue : exchange.sellFeeValue
   712→
   713→  let baseFee: number
   714→  const tier = pickTier(tiers, totalFiat)
   715→  if (tier) {
   716→    baseFee = tier.feeType === 'percent'
   717→      ? (totalFiat * tier.feeValue) / 100
   718→      : tier.feeValue
   719→  } else {
   720→    baseFee = fallbackType === 'percent'
   721→      ? (totalFiat * fallbackValue) / 100
   722→      : fallbackValue
   723→  }
   724→
   725→  // Descuento VIP/BNB se aplica SOLO sobre la comisión variable (no sobre la fija)
   726→  const discountPercent = exchange.discountPercent || 0
   727→  const discount = (baseFee * discountPercent) / 100
   728→  const baseFeeAfterDiscount = baseFee - discount
   729→
   730→  const fixedFee = exchange.fixedFee || 0
   731→  const total = baseFeeAfterDiscount + fixedFee
   732→
   733→  return {
   734→    baseFee,
   735→    discount,
   736→    fixedFee,
   737→    total,
   738→  }
   739→}

   782→  let transactions = (await loadTransactions()).map(migrateTx)
   783→
   784→  if (filters.type) transactions = transactions.filter((t) => t.type === filters.type)
   785→  if (filters.status) transactions = transactions.filter((t) => t.status === filters.status)
   786→  if (filters.bankId)
   787→    transactions = transactions.filter(
   788→      (t) => t.fromBankId === filters.bankId || t.toBankId === filters.bankId
   789→    )
   790→  if (filters.exchangeId) {
   791→    if (filters.exchangeId === '__none__') {
   792→      transactions = transactions.filter((t) => !t.exchangeId)
   793→    } else {
   794→      transactions = transactions.filter((t) => t.exchangeId === filters.exchangeId)
   795→    }
   796→  }
   797→  if (filters.counterparty) {
   798→    const q = filters.counterparty.toLowerCase()
   799→    transactions = transactions.filter(
   800→      (t) =>
   801→        t.counterparty.toLowerCase().includes(q) ||
   802→        t.asset.toLowerCase().includes(q) ||
   803→        (t.reference ?? '').toLowerCase().includes(q) ||
   804→        (t.notes ?? '').toLowerCase().includes(q)
   805→    )
   806→  }
   807→  if (filters.from) {
   808→    const d = new Date(filters.from)
   809→    transactions = transactions.filter((t) => new Date(t.date) >= d)
   810→  }
   811→  if (filters.to) {
   812→    const d = new Date(filters.to)
   813→    transactions = transactions.filter((t) => new Date(t.date) <= d)
   814→  }
   815→
   816→  transactions = transactions.sort(
   817→    (a, b) => new Date(b.date).getTime() - new Date(a.date).getTime()
   818→  )
   819→
   820→  if (filters.limit) transactions = transactions.slice(0, filters.limit)
   821→
   822→  // Adjuntar info de bancos y exchange
   823→  return transactions.map((t) => ({
   824→    ...t,
   825→    fromBank: t.fromBankId && bankMap.has(t.fromBankId)
   826→      ? {
   827→          id: bankMap.get(t.fromBankId)!.id,
   828→          name: bankMap.get(t.fromBankId)!.name,
   829→          currency: bankMap.get(t.fromBankId)!.currency,
   830→          color: bankMap.get(t.fromBankId)!.color,
   831→        }
   832→      : null,

   835→          id: bankMap.get(t.toBankId)!.id,
   836→          name: bankMap.get(t.toBankId)!.name,
   837→          currency: bankMap.get(t.toBankId)!.currency,
   838→          color: bankMap.get(t.toBankId)!.color,
   839→        }
   840→      : null,

   842→      ? {
   843→          id: exMap.get(t.exchangeId)!.id,
   844→          name: exMap.get(t.exchangeId)!.name,

   852→export async function createTransaction(input: TransactionInput): Promise<Transaction> {
   853→  const transactions = await loadTransactions()
   854→  const now = new Date().toISOString()
   855→  const total = input.amount * input.rate
   856→  const exchanges = await loadExchanges()
   857→  const exchange = input.exchangeId ? exchanges.find((e) => e.id === input.exchangeId) : null
   858→
   859→  let fee = input.fee ?? 0
   860→  let feeBreakdown: Transaction['feeBreakdown'] = null
   861→  // Si no se pasó fee explícito pero hay exchange, calcularlo
   862→  if (input.fee === undefined && exchange) {
   863→    const calc = calculateFee(exchange, input.type, total)
   864→    fee = calc.total
   865→    feeBreakdown = {
   866→      baseFee: calc.baseFee,
   867→      discount: calc.discount,
   868→      fixedFee: calc.fixedFee,
   869→      total: calc.total,
   870→    }
   871→  } else if (exchange) {
   872→    // Si el usuario pasó fee manual y hay exchange, reconstruir breakdown aproximado
   873→    const calc = calculateFee(exchange, input.type, total)
   874→    const baseAfterDiscount = fee - calc.fixedFee
   875→    feeBreakdown = {
   876→      baseFee: calc.baseFee,
   877→      discount: Math.max(0, calc.baseFee - baseAfterDiscount),
   878→      fixedFee: calc.fixedFee,
   879→      total: fee,
   880→    }
   881→  }
   882→
   883→  const tx: Transaction = {
   884→    id: uid(),
   885→    type: input.type,
   886→    counterparty: input.counterparty,
   887→    asset: input.asset ?? 'USDT',
   888→    amount: input.amount,
   889→    rate: input.rate,
   890→    total,
   891→    currency: input.currency ?? 'VES',
   892→    fromBankId: input.fromBankId ?? null,
   893→    toBankId: input.toBankId ?? null,
   894→    exchangeId: input.exchangeId ?? null,
   895→    status: input.status ?? 'completada',
   896→    reference: input.reference ?? null,
   897→    fee,
   898→    feeBreakdown,
   899→    netTotal: total - fee,
   900→    date: input.date ?? now,
   901→    notes: input.notes ?? null,
   902→    createdAt: now,
   903→    updatedAt: now,
   904→  }
   905→  transactions.push(tx)
   906→  saveTransactions(transactions)
   907→  return tx
   908→}

  1157→export async function importData(data: BackupData, mode: 'replace' | 'merge' = 'replace'): Promise<void> {
  1158→  if (mode === 'replace') {
  1159→    if (data.banks) saveBanks(data.banks)
  1160→    if (data.transactions) saveTransactions(data.transactions.map(migrateTx))
  1161→    if (data.exchanges) saveExchanges(data.exchanges)
  1162→  } else {
  1163→    const existingBanks = await loadBanks()
  1164→    const existingTx = (await loadTransactions()).map(migrateTx)
  1165→    const existingEx = await loadExchanges()
  1166→    const existingBankIds = new Set(existingBanks.map((b) => b.id))
  1167→    const existingTxIds = new Set(existingTx.map((t) => t.id))
  1168→    const existingExIds = new Set(existingEx.map((e) => e.id))
  1169→    const newBanks = (data.banks ?? []).filter((b) => !existingBankIds.has(b.id))
  1170→    const newTx = (data.transactions ?? []).map(migrateTx).filter((t) => !existingTxIds.has(t.id))
  1171→    const newEx = (data.exchanges ?? []).filter((e) => !existingExIds.has(e.id))
  1172→    saveBanks([...existingBanks, ...newBanks])
  1173→    saveTransactions([...existingTx, ...newTx])
  1174→    saveExchanges([...existingEx, ...newEx])
  1175→  }
  1176→}
