     1→'use client'
     2→
     3→import { useEffect, useState, useMemo } from 'react'
     4→import { useForm } from 'react-hook-form'
     5→import { zodResolver } from '@hookform/resolvers/zod'
     6→import { z } from 'zod'
     7→import {
     8→  Dialog,
     9→  DialogContent,
    10→  DialogDescription,
    11→  DialogFooter,
    12→  DialogHeader,
    13→  DialogTitle,
    14→} from '@/components/ui/dialog'
    15→import {
    16→  Form,
    17→  FormControl,
    18→  FormDescription,
    19→  FormField,
    20→  FormItem,
    21→  FormLabel,
    22→  FormMessage,
    23→} from '@/components/ui/form'
    24→import { Input } from '@/components/ui/input'
    25→import { Button } from '@/components/ui/button'
    26→import {
    27→  Select,
    28→  SelectContent,
    29→  SelectItem,
    30→  SelectTrigger,
    31→  SelectValue,
    32→} from '@/components/ui/select'
    33→import { Textarea } from '@/components/ui/textarea'
    34→import { Loader2, ArrowRight, Sparkles, Lock, Unlock } from 'lucide-react'
    35→import type {
    36→  Bank,
    37→  Exchange,
    38→  Transaction,
    39→  TransactionInput,
    40→  TransactionType,
    41→  TransactionStatus,
    42→} from '@/lib/types'
    43→import { toast } from '@/hooks/use-toast'
    44→import { formatCurrency, formatNumber, toDateTimeInputValue } from '@/lib/format'
    45→import * as storage from '@/lib/storage'
    46→
    47→const txSchema = z.object({
    48→  type: z.enum(['compra', 'venta']),
    49→  counterparty: z.string().min(1, 'La contraparte es obligatoria'),
    50→  asset: z.string().min(1, 'El activo es obligatorio'),
    51→  amount: z.coerce.number().positive('Debe ser > 0'),
    52→  rate: z.coerce.number().min(0, 'Debe ser ≥ 0'),
    53→  currency: z.string().min(1),
    54→  fromBankId: z.string().optional().nullable(),
    55→  toBankId: z.string().optional().nullable(),
    56→  exchangeId: z.string().optional().nullable(),
    57→  status: z.enum(['pendiente', 'completada', 'cancelada']),
    58→  reference: z.string().optional(),
    59→  fee: z.coerce.number().min(0).default(0),
    60→  date: z.string().min(1),
    61→  notes: z.string().optional(),
    62→})
    63→
    64→type FormValues = z.infer<typeof txSchema>
    65→
    66→const ASSETS = ['USDT', 'USDC', 'BTC', 'ETH', 'BNB', 'USD', 'EUR', 'COP']
    67→const CURRENCIES = ['VES', 'USD', 'EUR', 'COP', 'ARS', 'PEN', 'MXN', 'BRL']
    68→
    69→interface TransactionFormProps {
    70→  open: boolean
    71→  onOpenChange: (open: boolean) => void
    72→  transaction?: Transaction | null
    73→  defaultType?: TransactionType
    74→  banks: Bank[]
    75→  onSaved: () => void
    76→}
    77→
    78→export function TransactionForm({
    79→  open,
    80→  onOpenChange,
    81→  transaction,
    82→  defaultType = 'compra',
    83→  banks,
    84→  onSaved,
    85→}: TransactionFormProps) {
    86→  const [submitting, setSubmitting] = useState(false)
    87→  const [exchanges, setExchanges] = useState<Exchange[]>([])
    88→  const [feeLocked, setFeeLocked] = useState(false) // si true, el fee NO se recalcula al cambiar inputs
    89→  const [rateLocked, setRateLocked] = useState(false) // si true, la tasa NO se recalcula desde total
    90→  const [totalInput, setTotalInput] = useState<string>('') // total bruto ingresado por el usuario
    91→
    92→  const activeBanks = useMemo(() => banks.filter((b) => b.isActive), [banks])
    93→  const activeExchanges = useMemo(() => exchanges.filter((e) => e.isActive), [exchanges])
    94→
    95→  // Cargar exchanges
    96→  useEffect(() => {
    97→    storage.listExchanges().then(setExchanges).catch(() => {})
    98→  }, [open])
    99→
   100→  const form = useForm<FormValues>({
   101→    resolver: zodResolver(txSchema),
   102→    defaultValues: {
   103→      type: defaultType,
   104→      counterparty: '',
   105→      asset: 'USDT',
   106→      amount: 0,
   107→      rate: 0,
   108→      currency: 'VES',
   109→      fromBankId: null,
   110→      toBankId: null,
   111→      exchangeId: null,
   112→      status: 'completada',
   113→      reference: '',
   114→      fee: 0,
   115→      date: toDateTimeInputValue(new Date()),
   116→      notes: '',
   117→    },
   118→  })
   119→
   120→  useEffect(() => {
   121→    if (open) {
   122→      if (transaction) {
   123→        form.reset({
   124→          type: transaction.type,
   125→          counterparty: transaction.counterparty,
   126→          asset: transaction.asset,
   127→          amount: transaction.amount,
   128→          rate: transaction.rate,
   129→          currency: transaction.currency,
   130→          fromBankId: transaction.fromBankId ?? null,
   131→          toBankId: transaction.toBankId ?? null,
   132→          exchangeId: transaction.exchangeId ?? null,
   133→          status: transaction.status,
   134→          reference: transaction.reference ?? '',
   135→          fee: transaction.fee,
   136→          date: toDateTimeInputValue(transaction.date),
   137→          notes: transaction.notes ?? '',
   138→        })
   139→        // Al editar, consideramos el fee como bloqueado (es el valor guardado)
   140→        setFeeLocked(true)
   141→      } else {
   142→        form.reset({
   143→          type: defaultType,
   144→          counterparty: '',
   145→          asset: 'USDT',
   146→          amount: 0,
   147→          rate: 0,
   148→          currency: 'VES',
   149→          fromBankId: null,
   150→          toBankId: null,
   151→          exchangeId: null,
   152→          status: 'completada',
   153→          reference: '',
   154→          fee: 0,
   155→          date: toDateTimeInputValue(new Date()),
   156→          notes: '',
   157→        })
   158→        setFeeLocked(false)
   159→      }
   160→    }
   161→  }, [open, transaction, defaultType, form])
   162→
   163→  const watchedAmount = form.watch('amount')
   164→  const watchedRate = form.watch('rate')
   165→  const watchedCurrency = form.watch('currency')
   166→  const watchedAsset = form.watch('asset')
   167→  const watchedType = form.watch('type')
   168→  const watchedExchangeId = form.watch('exchangeId')
   169→  const watchedFee = form.watch('fee')
   170→
   171→  // Total bruto: si el usuario lo ingresó manualmente, usar ese valor;
   172→  // si no, calcularlo desde amount * rate
   173→  const totalFromRate = (Number(watchedAmount) || 0) * (Number(watchedRate) || 0)
   174→  const total = totalInput !== '' ? Number(totalInput) : totalFromRate
   175→
   176→  // Calcular tasa automáticamente cuando cambia el total ingresado o el monto
   177→  useEffect(() => {
   178→    if (rateLocked) return
   179→    const amount = Number(watchedAmount) || 0
   180→    const totalVal = Number(totalInput) || 0
   181→    if (amount > 0 && totalVal > 0) {
   182→      const calculatedRate = totalVal / amount
   183→      form.setValue('rate', Number(calculatedRate.toFixed(6)))
   184→    }
   185→  }, [totalInput, watchedAmount, rateLocked, form])
   186→
   187→  // Al resetear el form, sincronizar totalInput
   188→  useEffect(() => {
   189→    if (open && transaction) {
   190→      const t = (Number(transaction.amount) || 0) * (Number(transaction.rate) || 0)
   191→      setTotalInput(t > 0 ? String(t) : '')
   192→      setRateLocked(true) // al editar, la tasa es manual
   193→    } else if (open && !transaction) {
   194→      setTotalInput('')
   195→      setRateLocked(false)
   196→    }
   197→  }, [open, transaction])
   198→
   199→  // Cálculo automático de comisión cuando cambia exchange, monto, tasa o tipo
   200→  // (solo si el fee NO está bloqueado manualmente)
   201→  useEffect(() => {
   202→    if (feeLocked) return
   203→    if (!watchedExchangeId || watchedExchangeId === '__none') {
   204→      form.setValue('fee', 0)
   205→      return
   206→    }
   207→    const ex = exchanges.find((e) => e.id === watchedExchangeId)
   208→    if (!ex) return
   209→    const calc = storage.calculateFee(ex, watchedType, total)
   210→    form.setValue('fee', Number(calc.total.toFixed(6)))
   211→  }, [watchedExchangeId, watchedAmount, watchedRate, watchedType, exchanges, total, feeLocked, form, totalInput])
   212→
   213→  function handleExchangeChange(v: string) {
   214→    const value = v === '__none' ? null : v
   215→    form.setValue('exchangeId', value)
   216→    setFeeLocked(false) // si cambias exchange, se recalcula
   217→  }
   218→
   219→  function unlockFee() {
   220→    setFeeLocked(false)
   221→  }
   222→
   223→  function lockFee() {
   224→    setFeeLocked(true)
   225→  }
   226→
   227→  function unlockRate() {
   228→    setRateLocked(false)
   229→  }
   230→
   231→  function lockRate() {
   232→    setRateLocked(true)
   233→  }
   234→
   235→  // Manejar cambio en total bruto ingresado
   236→  function handleTotalInputChange(value: string) {
   237→    setTotalInput(value)
   238→    // Si el usuario está escribiendo el total, desbloquear la tasa para que se calcule
   239→    if (!rateLocked && value !== '') {
   240→      const amount = Number(watchedAmount) || 0
   241→      const totalVal = Number(value) || 0
   242→      if (amount > 0 && totalVal > 0) {
   243→        form.setValue('rate', Number((totalVal / amount).toFixed(6)))
   244→      }
   245→    }
   246→  }
   247→
   248→  async function onSubmit(values: FormValues) {
   249→    setSubmitting(true)
   250→    try {
   251→      const payload: TransactionInput = {
   252→        type: values.type,
   253→        counterparty: values.counterparty,
   254→        asset: values.asset,
   255→        amount: Number(values.amount),
   256→        rate: Number(values.rate),
   257→        currency: values.currency,
   258→        fromBankId: values.fromBankId || null,
   259→        toBankId: values.toBankId || null,
   260→        exchangeId: values.exchangeId || null,
   261→        status: values.status,
   262→        reference: values.reference || null,
   263→        fee: Number(values.fee),
   264→        date: new Date(values.date).toISOString(),
   265→        notes: values.notes || null,
   266→      }
   267→
   268→      if (transaction) {
   269→        await storage.updateTransaction(transaction.id, payload)
   270→      } else {
   271→        await storage.createTransaction(payload)
   272→      }
   273→
   274→      toast({
   275→        title: transaction
   276→          ? 'Transacción actualizada'
   277→          : values.type === 'compra'
   278→            ? 'Compra registrada'
   279→            : 'Venta registrada',
   280→        description: `${values.type === 'compra' ? 'Compra' : 'Venta'} de ${formatNumber(values.amount)} ${values.asset} con ${values.counterparty}.`,
   281→      })
   282→      onSaved()
   283→      onOpenChange(false)
   284→    } catch (e) {
   285→      toast({
   286→        title: 'Error',
   287→        description: e instanceof Error ? e.message : 'Error desconocido',
   288→        variant: 'destructive',
   289→      })
   290→    } finally {
   291→      setSubmitting(false)
   292→    }
   293→  }
   294→
   295→  // Neto (total menos comisión)
   296→  const fee = Number(watchedFee) || 0
   297→  const netTotal = total - fee
   298→
   299→  return (
   300→    <Dialog open={open} onOpenChange={onOpenChange}>
   301→      <DialogContent className="sm:max-w-[640px] max-h-[92vh] overflow-y-auto">
   302→        <DialogHeader>
   303→          <DialogTitle>
   304→            {transaction
   305→              ? 'Editar transacción'
   306→              : defaultType === 'compra'
   307→                ? 'Nueva compra P2P'
   308→                : 'Nueva venta P2P'}
   309→          </DialogTitle>
   310→          <DialogDescription>
   311→            {watchedType === 'compra'
   312→              ? 'Registra una compra: tú envías moneda fiat y recibes el activo.'
   313→              : 'Registra una venta: tú envías el activo y recibes moneda fiat.'}
   314→          </DialogDescription>
   315→        </DialogHeader>
   316→
   317→        <Form {...form}>
   318→          <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-4">
   319→            {/* Tipo de operación */}
   320→            <FormField
   321→              control={form.control}
   322→              name="type"
   323→              render={({ field }) => (
   324→                <FormItem>
   325→                  <FormLabel>Tipo de operación</FormLabel>
   326→                  <div className="grid grid-cols-2 gap-2">
   327→                    <button
   328→                      type="button"
   329→                      onClick={() => {
   330→                        field.onChange('compra')
   331→                        setFeeLocked(false) // recalcular fee con nuevo tipo
   332→                      }}
   333→                      className={`rounded-lg border-2 p-3 text-sm font-medium transition ${
   334→                        field.value === 'compra'
   335→                          ? 'border-emerald-500 bg-emerald-50 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300'
   336→                          : 'border-border hover:border-emerald-300'
   337→                      }`}
   338→                    >
   339→                      Compra
   340→                    </button>
   341→                    <button
   342→                      type="button"
   343→                      onClick={() => {
   344→                        field.onChange('venta')
   345→                        setFeeLocked(false)
   346→                      }}
   347→                      className={`rounded-lg border-2 p-3 text-sm font-medium transition ${
   348→                        field.value === 'venta'
   349→                          ? 'border-rose-500 bg-rose-50 text-rose-700 dark:bg-rose-950/40 dark:text-rose-300'
   350→                          : 'border-border hover:border-rose-300'
   351→                      }`}
   352→                    >
   353→                      Venta
   354→                    </button>
   355→                  </div>
   356→                  <FormMessage />
   357→                </FormItem>
   358→              )}
   359→            />
   360→
   361→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   362→              <FormField
   363→                control={form.control}
   364→                name="counterparty"
   365→                render={({ field }) => (
   366→                  <FormItem className="sm:col-span-2">
   367→                    <FormLabel>Contraparte *</FormLabel>
   368→                    <FormControl>
   369→                      <Input
   370→                        placeholder="Persona o plataforma (ej: @usuario, Binance P2P, etc.)"
   371→                        {...field}
   372→                      />
   373→                    </FormControl>
   374→                    <FormMessage />
   375→                  </FormItem>
   376→                )}
   377→              />
   378→
   379→              {/* EXCHANGE */}
   380→              <FormField
   381→                control={form.control}
   382→                name="exchangeId"
   383→                render={({ field }) => (
   384→                  <FormItem className="sm:col-span-2">
   385→                    <FormLabel className="flex items-center gap-1.5">
   386→                      <Sparkles className="h-3.5 w-3.5 text-blue-500" />
   387→                      Exchange / Plataforma
   388→                    </FormLabel>
   389→                    <Select
   390→                      onValueChange={handleExchangeChange}
   391→                      value={field.value ?? '__none'}
   392→                    >
   393→                      <FormControl>
   394→                        <SelectTrigger>
   395→                          <SelectValue placeholder="Sin exchange específico" />
   396→                        </SelectTrigger>
   397→                      </FormControl>
   398→                      <SelectContent>
   399→                        <SelectItem value="__none">Sin exchange (manual)</SelectItem>
   400→                        {activeExchanges.map((e) => (
   401→                          <SelectItem key={e.id} value={e.id}>
   402→                            <span className="flex items-center gap-2">
   403→                              <span
   404→                                className="inline-block w-2 h-2 rounded-full"
   405→                                style={{ backgroundColor: e.color }}
   406→                              />
   407→                              {e.name}
   408→                              <span className="text-xs text-muted-foreground ml-1">
   409→                                {watchedType === 'compra'
   410→                                  ? e.buyFeeType === 'percent'
   411→                                    ? `${e.buyFeeValue}%`
   412→                                    : `fijo ${e.buyFeeValue}`
   413→                                  : e.sellFeeType === 'percent'
   414→                                    ? `${e.sellFeeValue}%`
   415→                                    : `fijo ${e.sellFeeValue}`}
   416→                              </span>
   417→                            </span>
   418→                          </SelectItem>
   419→                        ))}
   420→                      </SelectContent>
   421→                    </Select>
   422→                    <FormDescription>
   423→                      Selecciona el exchange para calcular la comisión automáticamente. Puedes
   424→                      ajustarla después.
   425→                    </FormDescription>
   426→                    <FormMessage />
   427→                  </FormItem>
   428→                )}
   429→              />
   430→
   431→              <FormField
   432→                control={form.control}
   433→                name="asset"
   434→                render={({ field }) => (
   435→                  <FormItem>
   436→                    <FormLabel>Activo</FormLabel>
   437→                    <Select onValueChange={field.onChange} value={field.value}>
   438→                      <FormControl>
   439→                        <SelectTrigger>
   440→                          <SelectValue />
   441→                        </SelectTrigger>
   442→                      </FormControl>
   443→                      <SelectContent>
   444→                        {ASSETS.map((a) => (
   445→                          <SelectItem key={a} value={a}>
   446→                            {a}
   447→                          </SelectItem>
   448→                        ))}
   449→                      </SelectContent>
   450→                    </Select>
   451→                    <FormMessage />
   452→                  </FormItem>
   453→                )}
   454→              />
   455→
   456→              <FormField
   457→                control={form.control}
   458→                name="currency"
   459→                render={({ field }) => (
   460→                  <FormItem>
   461→                    <FormLabel>Moneda fiat</FormLabel>
   462→                    <Select onValueChange={field.onChange} value={field.value}>
   463→                      <FormControl>
   464→                        <SelectTrigger>
   465→                          <SelectValue />
   466→                        </SelectTrigger>
   467→                      </FormControl>
   468→                      <SelectContent>
   469→                        {CURRENCIES.map((c) => (
   470→                          <SelectItem key={c} value={c}>
   471→                            {c}
   472→                          </SelectItem>
   473→                        ))}
   474→                      </SelectContent>
   475→                    </Select>
   476→                    <FormMessage />
   477→                  </FormItem>
   478→                )}
   479→              />
   480→
   481→              <FormField
   482→                control={form.control}
   483→                name="amount"
   484→                render={({ field }) => (
   485→                  <FormItem>
   486→                    <FormLabel>Cantidad de {watchedAsset || 'activo'} *</FormLabel>
   487→                    <FormControl>
   488→                      <Input
   489→                        type="number"
   490→                        step="0.000001"
   491→                        placeholder="0.00"
   492→                        {...field}
   493→                      />
   494→                    </FormControl>
   495→                    <FormMessage />
   496→                  </FormItem>
   497→                )}
   498→              />
   499→
   500→              {/* Total bruto (input) — la tasa se calcula automáticamente */}
   501→              <div className="space-y-2">
   502→                <label className="text-sm font-medium flex items-center justify-between">
   503→                  <span>Total bruto ({watchedCurrency}) *</span>
   504→                  <button
   505→                    type="button"
   506→                    onClick={rateLocked ? unlockRate : lockRate}
   507→                    className="text-xs flex items-center gap-1 px-2 py-0.5 rounded border hover:bg-muted"
   508→                    title={rateLocked ? 'Desbloquear para calcular tasa automáticamente' : 'Tasa se calcula desde el total'}
   509→                  >
   510→                    {rateLocked ? (
   511→                      <>
   512→                        <Lock className="h-3 w-3" /> Tasa manual
   513→                      </>
   514→                    ) : (
   515→                      <>
   516→                        <Unlock className="h-3 w-3" /> Tasa auto
   517→                      </>
   518→                    )}
   519→                  </button>
   520→                </label>
   521→                <Input
   522→                  type="number"
   523→                  step="0.01"
   524→                  placeholder="0.00"
   525→                  value={totalInput}
   526→                  onChange={(e) => handleTotalInputChange(e.target.value)}
   527→                />
   528→                <p className="text-[0.8rem] text-muted-foreground">
   529→                  {rateLocked
   530→                    ? 'La tasa no se recalculará automáticamente.'
   531→                    : 'Ingresa el total bruto y la tasa se calculará automáticamente.'}
   532→                </p>
   533→              </div>
   534→
   535→              {/* Tasa calculada (solo lectura o editable si está bloqueada) */}
   536→              <FormField
   537→                control={form.control}
   538→                name="rate"
   539→                render={({ field }) => (
   540→                  <FormItem>
   541→                    <FormLabel>Tasa ({watchedCurrency}/{watchedAsset || '...'})</FormLabel>
   542→                    <FormControl>
   543→                      <Input
   544→                        type="number"
   545→                        step="0.000001"
   546→                        placeholder="0.00"
   547→                        className={rateLocked ? '' : 'bg-muted/50 cursor-default'}
   548→                        readOnly={!rateLocked}
   549→                        {...field}
   550→                      />
   551→                    </FormControl>
   552→                    <FormDescription>
   553→                      {rateLocked
   554→                        ? 'Edita la tasa manualmente.'
   555→                        : `Calculada: ${formatNumber(Number(watchedRate) || 0, 6)} ${watchedCurrency}/${watchedAsset}`}
   556→                    </FormDescription>
   557→                    <FormMessage />
   558→                  </FormItem>
   559→                )}
   560→              />
   561→
   562→              {/* Bancos */}
   563→              {watchedType === 'compra' ? (
   564→                <>
   565→                  <FormField
   566→                    control={form.control}
   567→                    name="fromBankId"
   568→                    render={({ field }) => (
   569→                      <FormItem>
   570→                        <FormLabel>Banco que paga (origen)</FormLabel>
   571→                        <Select
   572→                          onValueChange={(v) =>
   573→                            field.onChange(v === '__none' ? null : v)
   574→                          }
   575→                          value={field.value ?? '__none'}
   576→                        >
   577→                          <FormControl>
   578→                            <SelectTrigger>
   579→                              <SelectValue placeholder="Sin banco" />
   580→                            </SelectTrigger>
   581→                          </FormControl>
   582→                          <SelectContent>
   583→                            <SelectItem value="__none">
   584→                              Sin banco
   585→                            </SelectItem>
   586→                            {activeBanks.map((b) => (
   587→                              <SelectItem key={b.id} value={b.id}>
   588→                                {b.name} ({b.currency})
   589→                              </SelectItem>
   590→                            ))}
   591→                          </SelectContent>
   592→                        </Select>
   593→                        <FormDescription>
   594→                          Banco desde donde sale el dinero fiat.
   595→                        </FormDescription>
   596→                        <FormMessage />
   597→                      </FormItem>
   598→                    )}
   599→                  />
   600→                  <FormField
   601→                    control={form.control}
   602→                    name="toBankId"
   603→                    render={({ field }) => (
   604→                      <FormItem>
   605→                        <FormLabel>Banco receptor (opcional)</FormLabel>
   606→                        <Select
   607→                          onValueChange={(v) =>
   608→                            field.onChange(v === '__none' ? null : v)
   609→                          }
   610→                          value={field.value ?? '__none'}
   611→                        >
   612→                          <FormControl>
   613→                            <SelectTrigger>
   614→                              <SelectValue placeholder="Sin banco" />
   615→                            </SelectTrigger>
