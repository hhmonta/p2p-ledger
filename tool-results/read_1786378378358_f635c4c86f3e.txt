    49→function isBrowser(): boolean {
    50→  return typeof window !== 'undefined' && typeof window.localStorage !== 'undefined'
    51→}

    53→function readJSONSync<T>(key: string, fallback: T): T {
    54→  if (!isBrowser()) return fallback
    55→  try {
    56→    const raw = window.localStorage.getItem(key)
    57→    if (!raw) return fallback
    58→    return JSON.parse(raw) as T
    59→  } catch {
    60→    return fallback
    61→  }
    62→}
    63→
    64→async function readJSON<T>(key: string, fallback: T): Promise<T> {
    65→  if (!isBrowser()) return fallback
    66→  try {
    67→    const raw = window.localStorage.getItem(key)
    68→    if (!raw) return fallback
    69→    // Si la entrada está cifrada y tenemos DEK, descifrar
    70→    if (raw.startsWith(ENC_PREFIX)) {
    71→      if (!activeDek) {
    72→        // App bloqueada o sin DEK: no podemos descifrar
    73→        return fallback
    74→      }
    75→      const payload = raw.slice(ENC_PREFIX.length)
    76→      return decryptJson<T>(payload, activeDek)
    77→    }
    78→    return JSON.parse(raw) as T
    79→  } catch {
    80→    return fallback
    81→  }
    82→}
    83→
    84→function writeJSON(key: string, value: unknown): void {
    85→  if (!isBrowser()) return
    86→  if (ENCRYPTED_KEYS.has(key) && activeDek) {
    87→    // Cifrar antes de guardar
    88→    encryptJson(value, activeDek)
    89→      .then((payload) => {
    90→        window.localStorage.setItem(key, ENC_PREFIX + payload)
    91→      })
    92→      .catch(() => {
    93→        // Si falla el cifrado, no escribir (mejor perder el dato que guardarlo en claro)
    94→      })
    95→    return
    96→  }
    97→  window.localStorage.setItem(key, JSON.stringify(value))
    98→}

   100→/**
   101→ * Migra entradas en claro a cifradas cuando se activa el PIN.
   102→ * Lee todas las claves sensibles y las re-escribe cifradas.
   103→ */
   104→export async function encryptAllStorage(dek: CryptoKey) {
   105→  if (!isBrowser()) return
   106→  for (const key of ENCRYPTED_KEYS) {
   107→    const raw = window.localStorage.getItem(key)
   108→    if (!raw) continue
   109→    if (raw.startsWith(ENC_PREFIX)) continue // ya cifrada
   110→    try {
   111→      const parsed = JSON.parse(raw)
   112→      const payload = await encryptJson(parsed, dek)
   113→      window.localStorage.setItem(key, ENC_PREFIX + payload)
   114→    } catch {
   115→      // ignorar entradas inválidas
   116→    }
   117→  }
   118→}

   120→/**
   121→ * Migra entradas cifradas a claro cuando se desactiva el PIN.
   122→ * Requiere DEK para descifrar antes de quitarla.
   123→ */
   124→export async function decryptAllStorage(dek: CryptoKey) {
   125→  if (!isBrowser()) return
   126→  for (const key of ENCRYPTED_KEYS) {
   127→    const raw = window.localStorage.getItem(key)
   128→    if (!raw) continue
   129→    if (!raw.startsWith(ENC_PREFIX)) continue // ya en claro
   130→    try {
   131→      const payload = raw.slice(ENC_PREFIX.length)
   132→      const parsed = await decryptJson<unknown>(payload, dek)
   133→      window.localStorage.setItem(key, JSON.stringify(parsed))
   134→    } catch {
   135→      // ignorar
   136→    }
   137→  }
   138→}

   140→function uid(): string {
   141→  if (typeof crypto !== 'undefined' && crypto.randomUUID) {
   142→    return crypto.randomUUID()
   143→  }
   144→  return Date.now().toString(36) + Math.random().toString(36).slice(2, 10)
   145→}

   151→export function subscribe(listener: Listener): () => void {

   160→// Inicialización (solo una vez)
   161→let initialized = false
   162→function ensureInit() {
   163→  if (initialized || !isBrowser()) return
   164→  initialized = true
   165→  const version = window.localStorage.getItem(VERSION_KEY)
   166→  if (version !== CURRENT_VERSION) {
   167→    // Si vienen de una versión anterior, migrar exchanges existentes
   168→    // v3 → v4: añadimos Bybit, KuCoin y Paxful si no existen
   169→    if (version === '3') {
   170→      const existing = readJSONSync<Exchange[]>(EXCHANGES_KEY, [])
   171→      const migrated = existing.map(migrateExchange)
   172→      const existingNames = new Set(migrated.map((e) => e.name.toLowerCase()))
   173→      const additions = defaultExchanges().filter(
   174→        (e) => !existingNames.has(e.name.toLowerCase())
   175→      )
   176→      // Solo añadimos los nuevos (Bybit, KuCoin, Paxful) - no duplicamos los antiguos
   177→      const newOnes = additions.filter((e) =>
   178→        ['Bybit P2P', 'KuCoin P2P', 'Paxful', 'Bybit Spot', 'KuCoin Spot'].includes(e.name)
   179→      )
   180→      writeJSON(EXCHANGES_KEY, [...migrated, ...newOnes])
   181→    }
   182→    // v4 → v5: añadimos P2p.me si no existe
   183→    if (version === '4') {
   184→      const existing = readJSONSync<Exchange[]>(EXCHANGES_KEY, []).map(migrateExchange)
   185→      const existingNames = new Set(existing.map((e) => e.name.toLowerCase()))
   186→      const p2pme = defaultExchanges().find((e) => e.name === 'P2p.me')!
   187→      if (!existingNames.has('p2p.me')) {
   188→        writeJSON(EXCHANGES_KEY, [...existing, p2pme])
   189→      }
   190→    }
   191→    // Si es primera instalación (sin versión), sembrar exchanges por defecto
   192→    if (!version) {
   193→      const defaults = defaultExchanges()
   194→      writeJSON(EXCHANGES_KEY, defaults)
   195→    }
   196→    window.localStorage.setItem(VERSION_KEY, CURRENT_VERSION)
   197→  }
   198→}

   500→function computeBankBalance(bank: Bank, transactions: Transaction[]): {
   501→  balance: number
   502→  totalEntradas: number
   503→  totalSalidas: number
   504→  countFrom: number
   505→  countTo: number
   506→} {
   507→  let entrada = 0
   508→  let salida = 0
   509→  let countFrom = 0
   510→  let countTo = 0
   511→  for (const t of transactions) {
   512→    if (t.status !== 'completada') continue
   513→    // Para el cálculo del balance del banco usamos el neto (después de comisiones)
   514→    // porque la comisión es un costo que se descuenta del monto que efectivamente entra/sale.
   515→    const neto = t.netTotal ?? t.total
   516→    if (t.toBankId === bank.id) {
   517→      entrada += neto
   518→      countTo++
   519→    }
   520→    if (t.fromBankId === bank.id) {
   521→      salida += neto
   522→      countFrom++
   523→    }
   524→  }
   525→  return {
   526→    balance: bank.initialBalance + entrada - salida,
   527→    totalEntradas: entrada,
   528→    totalSalidas: salida,
   529→    countFrom,
   530→    countTo,
   531→  }
   532→}

   534→export async function listBanks(): Promise<Bank[]> {
   535→  const banks = await loadBanks()
   536→  const transactions = await loadTransactions()
   537→  return banks
   538→    .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime())
   539→    .map((b) => {
   540→      const computed = computeBankBalance(b, transactions)
   541→      return {
   542→        ...b,
   543→        balance: computed.balance,
   544→        totalEntradas: computed.totalEntradas,
   545→        totalSalidas: computed.totalSalidas,
   546→        _count: {
   547→          transactionsFrom: computed.countFrom,
   548→          transactionsTo: computed.countTo,

   703→/**
   704→ * Selecciona el tier aplicable para un monto dado.
   705→ * Si no hay tiers definidos, retorna null (usar feeType/feeValue base).
   706→ * Si hay tiers, retorna el de mayor minAmount <= totalFiat.
   707→ */
   708→function pickTier(tiers: FeeTier[], totalFiat: number): FeeTier | null {
   709→  if (!tiers || tiers.length === 0) return null
   710→  // Ordenar de mayor a menor minAmount y elegir el primero que aplique
   711→  const sorted = [...tiers].sort((a, b) => b.minAmount - a.minAmount)
   712→  for (const tier of sorted) {
   713→    if (totalFiat >= tier.minAmount) return tier
   714→  }
   715→  // Fallback al tier de menor minAmount (debería ser 0 normalmente)
   716→  return sorted[sorted.length - 1]
   717→}
   718→
   719→/**
   720→ * Calcula la comisión que aplicaría un exchange a una operación dada.
   721→ * Soporta:
   722→ *   - Tiers escalonados (si están definidos, reemplazan el fee base)
   723→ *   - Descuento VIP/BNB (porcentaje descontado de la comisión variable, NO de la fija)
   724→ *   - Comisión fija adicional
   725→ * Retorna desglose y total.
   726→ */
   727→export function calculateFee(
   728→  exchange: Exchange | null | undefined,
   729→  type: TransactionType,
   730→  totalFiat: number
   731→): { baseFee: number; discount: number; fixedFee: number; total: number } {
   732→  if (!exchange) return { baseFee: 0, discount: 0, fixedFee: 0, total: 0 }
   733→
   734→  const tiers = type === 'compra' ? exchange.buyTiers : exchange.sellTiers
   735→  const fallbackType = type === 'compra' ? exchange.buyFeeType : exchange.sellFeeType
   736→  const fallbackValue = type === 'compra' ? exchange.buyFeeValue : exchange.sellFeeValue
   737→
   738→  let baseFee: number
   739→  const tier = pickTier(tiers, totalFiat)
   740→  if (tier) {
   741→    baseFee = tier.feeType === 'percent'
   742→      ? (totalFiat * tier.feeValue) / 100
   743→      : tier.feeValue
   744→  } else {
   745→    baseFee = fallbackType === 'percent'
   746→      ? (totalFiat * fallbackValue) / 100
   747→      : fallbackValue
   748→  }
   749→
   750→  // Descuento VIP/BNB se aplica SOLO sobre la comisión variable (no sobre la fija)
   751→  const discountPercent = exchange.discountPercent || 0
   752→  const discount = (baseFee * discountPercent) / 100
   753→  const baseFeeAfterDiscount = baseFee - discount
   754→
   755→  const fixedFee = exchange.fixedFee || 0
   756→  const total = baseFeeAfterDiscount + fixedFee
   757→
   758→  return {
   759→    baseFee,
   760→    discount,
   761→    fixedFee,
   762→    total,
   763→  }
   764→}

   802→export async function listTransactions(filters: TransactionFilters = {}): Promise<Transaction[]> {
   803→  const banks = await loadBanks()
   804→  const bankMap = new Map(banks.map((b) => [b.id, b]))
   805→  const exchanges = await loadExchanges()
   806→  const exMap = new Map(exchanges.map((e) => [e.id, e]))
   807→  let transactions = (await loadTransactions()).map(migrateTx)
   808→
   809→  if (filters.type) transactions = transactions.filter((t) => t.type === filters.type)
   810→  if (filters.status) transactions = transactions.filter((t) => t.status === filters.status)
   811→  if (filters.bankId)
   812→    transactions = transactions.filter(
   813→      (t) => t.fromBankId === filters.bankId || t.toBankId === filters.bankId
   814→    )
   815→  if (filters.exchangeId) {
   816→    if (filters.exchangeId === '__none__') {
   817→      transactions = transactions.filter((t) => !t.exchangeId)
   818→    } else {
   819→      transactions = transactions.filter((t) => t.exchangeId === filters.exchangeId)
   820→    }
   821→  }
   822→  if (filters.counterparty) {
   823→    const q = filters.counterparty.toLowerCase()
   824→    transactions = transactions.filter(
   825→      (t) =>
   826→        t.counterparty.toLowerCase().includes(q) ||
   827→        t.asset.toLowerCase().includes(q) ||
   828→        (t.reference ?? '').toLowerCase().includes(q) ||
   829→        (t.notes ?? '').toLowerCase().includes(q)
   830→    )
   831→  }
   832→  if (filters.from) {
   833→    const d = new Date(filters.from)
   834→    transactions = transactions.filter((t) => new Date(t.date) >= d)
   835→  }
   836→  if (filters.to) {
   837→    const d = new Date(filters.to)
   838→    transactions = transactions.filter((t) => new Date(t.date) <= d)
   839→  }
   840→
   841→  transactions = transactions.sort(
   842→    (a, b) => new Date(b.date).getTime() - new Date(a.date).getTime()
   843→  )
   844→
   845→  if (filters.limit) transactions = transactions.slice(0, filters.limit)
   846→
   847→  // Adjuntar info de bancos y exchange
   848→  return transactions.map((t) => ({
   849→    ...t,
   850→    fromBank: t.fromBankId && bankMap.has(t.fromBankId)
   851→      ? {
   852→          id: bankMap.get(t.fromBankId)!.id,
   853→          name: bankMap.get(t.fromBankId)!.name,
   854→          currency: bankMap.get(t.fromBankId)!.currency,
   855→          color: bankMap.get(t.fromBankId)!.color,
   856→        }
   857→      : null,
   858→    toBank: t.toBankId && bankMap.has(t.toBankId)
   859→      ? {
   860→          id: bankMap.get(t.toBankId)!.id,
   861→          name: bankMap.get(t.toBankId)!.name,
   862→          currency: bankMap.get(t.toBankId)!.currency,
   863→          color: bankMap.get(t.toBankId)!.color,
   864→        }
   865→      : null,
   866→    exchange: t.exchangeId && exMap.has(t.exchangeId)
   867→      ? {
   868→          id: exMap.get(t.exchangeId)!.id,
   869→          name: exMap.get(t.exchangeId)!.name,
   870→          shortName: exMap.get(t.exchangeId)!.shortName,
   871→          color: exMap.get(t.exchangeId)!.color,
   872→        }
   873→      : null,
   874→  }))
   875→}

   877→export async function createTransaction(input: TransactionInput): Promise<Transaction> {
   878→  const transactions = await loadTransactions()
   879→  const now = new Date().toISOString()
   880→  const total = input.amount * input.rate
   881→  const exchanges = await loadExchanges()
   882→  const exchange = input.exchangeId ? exchanges.find((e) => e.id === input.exchangeId) : null
   883→
   884→  let fee = input.fee ?? 0
   885→  let feeBreakdown: Transaction['feeBreakdown'] = null
   886→  // Si no se pasó fee explícito pero hay exchange, calcularlo
   887→  if (input.fee === undefined && exchange) {
   888→    const calc = calculateFee(exchange, input.type, total)
   889→    fee = calc.total
   890→    feeBreakdown = {
   891→      baseFee: calc.baseFee,
   892→      discount: calc.discount,
   893→      fixedFee: calc.fixedFee,
   894→      total: calc.total,
   895→    }
   896→  } else if (exchange) {
   897→    // Si el usuario pasó fee manual y hay exchange, reconstruir breakdown aproximado
   898→    const calc = calculateFee(exchange, input.type, total)
   899→    const baseAfterDiscount = fee - calc.fixedFee
   900→    feeBreakdown = {
   901→      baseFee: calc.baseFee,
   902→      discount: Math.max(0, calc.baseFee - baseAfterDiscount),
   903→      fixedFee: calc.fixedFee,
   904→      total: fee,
   905→    }
   906→  }
   907→
   908→  const tx: Transaction = {
   909→    id: uid(),
   910→    type: input.type,
   911→    counterparty: input.counterparty,
   912→    asset: input.asset ?? 'USDT',
   913→    amount: input.amount,
   914→    rate: input.rate,
   915→    total,
   916→    currency: input.currency ?? 'VES',
   917→    fromBankId: input.fromBankId ?? null,
   918→    toBankId: input.toBankId ?? null,
   919→    exchangeId: input.exchangeId ?? null,
   920→    status: input.status ?? 'completada',
   921→    reference: input.reference ?? null,
   922→    fee,
   923→    feeBreakdown,
   924→    netTotal: total - fee,
   925→    date: input.date ?? now,
   926→    notes: input.notes ?? null,
   927→    createdAt: now,
   928→    updatedAt: now,
   929→  }
   930→  transactions.push(tx)
   931→  saveTransactions(transactions)
   932→  return tx
   933→}

  1091→    const key = t.exchangeId ?? '__none__'
  1092→    const ex = t.exchangeId ? exInfoMap.get(t.exchangeId) : null
  1093→    const entry = exMap.get(key) ?? {
  1094→      exchangeId: t.exchangeId ?? null,
  1095→      exchangeName: ex?.name ?? 'Sin exchange',
  1096→      exchangeColor: ex?.color ?? '#64748b',
  1097→      totalFees: 0,
  1098→      count: 0,
  1099→      compras: 0,
  1100→      ventas: 0,
  1101→    }
  1102→    entry.totalFees += t.fee || 0
  1103→    entry.count++
  1104→    if (t.type === 'compra') entry.compras++
  1105→    else entry.ventas++
  1106→    exMap.set(key, entry)
  1107→  }
  1108→  const feesPorExchange = Array.from(exMap.values()).sort((a, b) => b.totalFees - a.totalFees)
  1109→
  1110→  // Evolución mensual (12 meses)
  1111→  const now = new Date()
  1112→  const monthlyMap = new Map<string, { compras: number; ventas: number }>()
  1113→  for (let i = 11; i >= 0; i--) {
  1114→    const d = new Date(now.getFullYear(), now.getMonth() - i, 1)
  1115→    const key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`
  1116→    monthlyMap.set(key, { compras: 0, ventas: 0 })
  1117→  }
  1118→  for (const t of completadas) {
  1119→    const d = new Date(t.date)
  1120→    const key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`
  1121→    const entry = monthlyMap.get(key)
  1122→    if (!entry) continue
  1123→    if (t.type === 'compra') entry.compras += t.total
  1124→    else if (t.type === 'venta') entry.ventas += t.total
  1125→  }
  1126→  const monthly = Array.from(monthlyMap.entries()).map(([month, val]) => ({
  1127→    month,
  1128→    ...val,
  1129→  }))
  1130→
  1131→  return {
  1132→    resumen: {

  1135→      montoCompras,
  1136→      montoVentas,
  1137→      cantidadCompras: compras.length,
  1138→      cantidadVentas: ventas.length,

  1140→      totalBanks: banks.length,
  1141→      totalExchanges: exchanges.filter((e) => e.isActive).length,
  1142→      feesCompras,

  1146→      netVentas,

  1151→      gananciaNeta,

  1153→    topCounterpartes,

  1156→    monthly,

  1182→export async function importData(data: BackupData, mode: 'replace' | 'merge' = 'replace'): Promise<void> {
  1183→  if (mode === 'replace') {
  1184→    if (data.banks) saveBanks(data.banks)
  1185→    if (data.transactions) saveTransactions(data.transactions.map(migrateTx))
  1186→    if (data.exchanges) saveExchanges(data.exchanges)
  1187→  } else {
  1188→    const existingBanks = await loadBanks()
  1189→    const existingTx = (await loadTransactions()).map(migrateTx)
  1190→    const existingEx = await loadExchanges()
  1191→    const existingBankIds = new Set(existingBanks.map((b) => b.id))
  1192→    const existingTxIds = new Set(existingTx.map((t) => t.id))
  1193→    const existingExIds = new Set(existingEx.map((e) => e.id))
  1194→    const newBanks = (data.banks ?? []).filter((b) => !existingBankIds.has(b.id))
  1195→    const newTx = (data.transactions ?? []).map(migrateTx).filter((t) => !existingTxIds.has(t.id))
  1196→    const newEx = (data.exchanges ?? []).filter((e) => !existingExIds.has(e.id))
  1197→    saveBanks([...existingBanks, ...newBanks])
  1198→    saveTransactions([...existingTx, ...newTx])
  1199→    saveExchanges([...existingEx, ...newEx])
  1200→  }
  1201→}
