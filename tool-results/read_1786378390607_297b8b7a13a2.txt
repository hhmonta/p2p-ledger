    47→    47→const txSchema = z.object({

    58→    58→  reference: z.string().optional(),
    59→    59→  fee: z.coerce.number().min(0).default(0),
    60→    60→  date: z.string().min(1),

    69→    69→interface TransactionFormProps {

    77→    77→
    78→    78→export function TransactionForm({
    79→    79→  open,
    80→    80→  onOpenChange,
    81→    81→  transaction,
    82→    82→  defaultType = 'compra',
    83→    83→  banks,
    84→    84→  onSaved,
    85→    85→}: TransactionFormProps) {
    86→    86→  const [submitting, setSubmitting] = useState(false)
    87→    87→  const [exchanges, setExchanges] = useState<Exchange[]>([])
    88→    88→  const [feeLocked, setFeeLocked] = useState(false) // si true, el fee NO se recalcula al cambiar inputs
    89→    89→  const [rateLocked, setRateLocked] = useState(false) // si true, la tasa NO se recalcula desde total
    90→    90→  const [totalInput, setTotalInput] = useState<string>('') // total bruto ingresado por el usuario
    91→    91→
    92→    92→  const activeBanks = useMemo(() => banks.filter((b) => b.isActive), [banks])
    93→    93→  const activeExchanges = useMemo(() => exchanges.filter((e) => e.isActive), [exchanges])
    94→    94→
    95→    95→  // Cargar exchanges
    96→    96→  useEffect(() => {
    97→    97→    storage.listExchanges().then(setExchanges).catch(() => {})
    98→    98→  }, [open])
    99→    99→
   100→   100→  const form = useForm<FormValues>({
   101→   101→    resolver: zodResolver(txSchema),
   102→   102→    defaultValues: {
   103→   103→      type: defaultType,
   104→   104→      counterparty: '',
   105→   105→      asset: 'USDT',
   106→   106→      amount: 0,
   107→   107→      rate: 0,
   108→   108→      currency: 'VES',
   109→   109→      fromBankId: null,
   110→   110→      toBankId: null,
   111→   111→      exchangeId: null,
   112→   112→      status: 'completada',
   113→   113→      reference: '',
   114→   114→      fee: 0,
   115→   115→      date: toDateTimeInputValue(new Date()),
   116→   116→      notes: '',
   117→   117→    },
   118→   118→  })
   119→   119→
   120→   120→  useEffect(() => {
   121→   121→    if (open) {
   122→   122→      if (transaction) {
   123→   123→        form.reset({
   124→   124→          type: transaction.type,
   125→   125→          counterparty: transaction.counterparty,
   126→   126→          asset: transaction.asset,
   127→   127→          amount: transaction.amount,
   128→   128→          rate: transaction.rate,
   129→   129→          currency: transaction.currency,
   130→   130→          fromBankId: transaction.fromBankId ?? null,
   131→   131→          toBankId: transaction.toBankId ?? null,
   132→   132→          exchangeId: transaction.exchangeId ?? null,
   133→   133→          status: transaction.status,
   134→   134→          reference: transaction.reference ?? '',
   135→   135→          fee: transaction.fee,
   136→   136→          date: toDateTimeInputValue(transaction.date),
   137→   137→          notes: transaction.notes ?? '',
   138→   138→        })
   139→   139→        // Al editar, consideramos el fee como bloqueado (es el valor guardado)
   140→   140→        setFeeLocked(true)
   141→   141→      } else {
   142→   142→        form.reset({
   143→   143→          type: defaultType,
   144→   144→          counterparty: '',
   145→   145→          asset: 'USDT',
   146→   146→          amount: 0,
   147→   147→          rate: 0,
   148→   148→          currency: 'VES',
   149→   149→          fromBankId: null,
   150→   150→          toBankId: null,
   151→   151→          exchangeId: null,
   152→   152→          status: 'completada',
   153→   153→          reference: '',
   154→   154→          fee: 0,
   155→   155→          date: toDateTimeInputValue(new Date()),
   156→   156→          notes: '',
   157→   157→        })
   158→   158→        setFeeLocked(false)
   159→   159→      }
   160→   160→    }
   161→   161→  }, [open, transaction, defaultType, form])
   162→   162→
   163→   163→  const watchedAmount = form.watch('amount')
   164→   164→  const watchedRate = form.watch('rate')
   165→   165→  const watchedCurrency = form.watch('currency')
   166→   166→  const watchedAsset = form.watch('asset')
   167→   167→  const watchedType = form.watch('type')
   168→   168→  const watchedExchangeId = form.watch('exchangeId')
   169→   169→  const watchedFee = form.watch('fee')
   170→   170→
   171→   171→  // Total bruto: si el usuario lo ingresó manualmente, usar ese valor;
   172→   172→  // si no, calcularlo desde amount * rate
   173→   173→  const totalFromRate = (Number(watchedAmount) || 0) * (Number(watchedRate) || 0)
   174→   174→  const total = totalInput !== '' ? Number(totalInput) : totalFromRate
   175→   175→
   176→   176→  // Calcular tasa automáticamente cuando cambia el total ingresado o el monto
   177→   177→  useEffect(() => {
   178→   178→    if (rateLocked) return
   179→   179→    const amount = Number(watchedAmount) || 0
   180→   180→    const totalVal = Number(totalInput) || 0
   181→   181→    if (amount > 0 && totalVal > 0) {
   182→   182→      const calculatedRate = totalVal / amount
   183→   183→      form.setValue('rate', Number(calculatedRate.toFixed(6)))
   184→   184→    }
   185→   185→  }, [totalInput, watchedAmount, rateLocked, form])
   186→   186→
   187→   187→  // Al resetear el form, sincronizar totalInput
   188→   188→  useEffect(() => {
   189→   189→    if (open && transaction) {
   190→   190→      const t = (Number(transaction.amount) || 0) * (Number(transaction.rate) || 0)
   191→   191→      setTotalInput(t > 0 ? String(t) : '')
   192→   192→      setRateLocked(true) // al editar, la tasa es manual
   193→   193→    } else if (open && !transaction) {
   194→   194→      setTotalInput('')
   195→   195→      setRateLocked(false)
   196→   196→    }
   197→   197→  }, [open, transaction])
   198→   198→
   199→   199→  // Cálculo automático de comisión cuando cambia exchange, monto, tasa o tipo
   200→   200→  // (solo si el fee NO está bloqueado manualmente)
   201→   201→  useEffect(() => {
   202→   202→    if (feeLocked) return
   203→   203→    if (!watchedExchangeId || watchedExchangeId === '__none') {
   204→   204→      form.setValue('fee', 0)
   205→   205→      return
   206→   206→    }
   207→   207→    const ex = exchanges.find((e) => e.id === watchedExchangeId)
   208→   208→    if (!ex) return
   209→   209→    const calc = storage.calculateFee(ex, watchedType, total)
   210→   210→    form.setValue('fee', Number(calc.total.toFixed(6)))
   211→   211→  }, [watchedExchangeId, watchedAmount, watchedRate, watchedType, exchanges, total, feeLocked, form, totalInput])
   212→   212→
   213→   213→  function handleExchangeChange(v: string) {
   214→   214→    const value = v === '__none' ? null : v
   215→   215→    form.setValue('exchangeId', value)
   216→   216→    setFeeLocked(false) // si cambias exchange, se recalcula
   217→   217→  }
   218→   218→
   219→   219→  function unlockFee() {
   220→   220→    setFeeLocked(false)
   221→   221→  }
   222→   222→
   223→   223→  function lockFee() {
   224→   224→    setFeeLocked(true)
   225→   225→  }
   226→   226→
   227→   227→  function unlockRate() {
   228→   228→    setRateLocked(false)
   229→   229→  }
   230→   230→
   231→   231→  function lockRate() {
   232→   232→    setRateLocked(true)
   233→   233→  }
   234→   234→
   235→   235→  // Manejar cambio en total bruto ingresado
   236→   236→  function handleTotalInputChange(value: string) {
   237→   237→    setTotalInput(value)
   238→   238→    // Si el usuario está escribiendo el total, desbloquear la tasa para que se calcule
   239→   239→    if (!rateLocked && value !== '') {
   240→   240→      const amount = Number(watchedAmount) || 0
   241→   241→      const totalVal = Number(value) || 0
   242→   242→      if (amount > 0 && totalVal > 0) {
   243→   243→        form.setValue('rate', Number((totalVal / amount).toFixed(6)))
   244→   244→      }
   245→   245→    }
   246→   246→  }
   247→   247→
   248→   248→  async function onSubmit(values: FormValues) {
   249→   249→    setSubmitting(true)
   250→   250→    try {
   251→   251→      const payload: TransactionInput = {
   252→   252→        type: values.type,
   253→   253→        counterparty: values.counterparty,
   254→   254→        asset: values.asset,
   255→   255→        amount: Number(values.amount),
   256→   256→        rate: Number(values.rate),
   257→   257→        currency: values.currency,
   258→   258→        fromBankId: values.fromBankId || null,
   259→   259→        toBankId: values.toBankId || null,
   260→   260→        exchangeId: values.exchangeId || null,
   261→   261→        status: values.status,
   262→   262→        reference: values.reference || null,
   263→   263→        fee: Number(values.fee),
   264→   264→        date: new Date(values.date).toISOString(),
   265→   265→        notes: values.notes || null,
   266→   266→      }
   267→   267→
   268→   268→      if (transaction) {
   269→   269→        await storage.updateTransaction(transaction.id, payload)
   270→   270→      } else {
   271→   271→        await storage.createTransaction(payload)
   272→   272→      }
   273→   273→
   274→   274→      toast({
   275→   275→        title: transaction
   276→   276→          ? 'Transacción actualizada'
   277→   277→          : values.type === 'compra'
   278→   278→            ? 'Compra registrada'
   279→   279→            : 'Venta registrada',
   280→   280→        description: `${values.type === 'compra' ? 'Compra' : 'Venta'} de ${formatNumber(values.amount)} ${values.asset} con ${values.counterparty}.`,
   281→   281→      })
   282→   282→      onSaved()
   283→   283→      onOpenChange(false)
   284→   284→    } catch (e) {
   285→   285→      toast({
   286→   286→        title: 'Error',
   287→   287→        description: e instanceof Error ? e.message : 'Error desconocido',
   288→   288→        variant: 'destructive',
   289→   289→      })
   290→   290→    } finally {
   291→   291→      setSubmitting(false)
   292→   292→    }
   293→   293→  }
   294→   294→
   295→   295→  // Neto (total menos comisión)
   296→   296→  const fee = Number(watchedFee) || 0
   297→   297→  const netTotal = total - fee
   298→   298→
   299→   299→  return (
   300→   300→    <Dialog open={open} onOpenChange={onOpenChange}>
   301→   301→      <DialogContent className="sm:max-w-[640px] max-h-[92vh] overflow-y-auto">
   302→   302→        <DialogHeader>
   303→   303→          <DialogTitle>
   304→   304→            {transaction
   305→   305→              ? 'Editar transacción'
   306→   306→              : defaultType === 'compra'
   307→   307→                ? 'Nueva compra P2P'
   308→   308→                : 'Nueva venta P2P'}
   309→   309→          </DialogTitle>
   310→   310→          <DialogDescription>
   311→   311→            {watchedType === 'compra'
   312→   312→              ? 'Registra una compra: tú envías moneda fiat y recibes el activo.'
   313→   313→              : 'Registra una venta: tú envías el activo y recibes moneda fiat.'}
   314→   314→          </DialogDescription>
   315→   315→        </DialogHeader>
   316→   316→
   317→   317→        <Form {...form}>
   318→   318→          <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-4">
   319→   319→            {/* Tipo de operación */}
   320→   320→            <FormField
   321→   321→              control={form.control}
   322→   322→              name="type"
   323→   323→              render={({ field }) => (
   324→   324→                <FormItem>
   325→   325→                  <FormLabel>Tipo de operación</FormLabel>
   326→   326→                  <div className="grid grid-cols-2 gap-2">
   327→   327→                    <button
   328→   328→                      type="button"
   329→   329→                      onClick={() => {
   330→   330→                        field.onChange('compra')
   331→   331→                        setFeeLocked(false) // recalcular fee con nuevo tipo
   332→   332→                      }}
   333→   333→                      className={`rounded-lg border-2 p-3 text-sm font-medium transition ${
   334→   334→                        field.value === 'compra'
   335→   335→                          ? 'border-emerald-500 bg-emerald-50 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300'
   336→   336→                          : 'border-border hover:border-emerald-300'
   337→   337→                      }`}
   338→   338→                    >
   339→   339→                      Compra
   340→   340→                    </button>
   341→   341→                    <button
   342→   342→                      type="button"
   343→   343→                      onClick={() => {
   344→   344→                        field.onChange('venta')
   345→   345→                        setFeeLocked(false)
   346→   346→                      }}
   347→   347→                      className={`rounded-lg border-2 p-3 text-sm font-medium transition ${
   348→   348→                        field.value === 'venta'
   349→   349→                          ? 'border-rose-500 bg-rose-50 text-rose-700 dark:bg-rose-950/40 dark:text-rose-300'
   350→   350→                          : 'border-border hover:border-rose-300'
   351→   351→                      }`}
   352→   352→                    >
   353→   353→                      Venta
   354→   354→                    </button>
   355→   355→                  </div>
   356→   356→                  <FormMessage />
   357→   357→                </FormItem>
   358→   358→              )}
   359→   359→            />
   360→   360→
   361→   361→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   362→   362→              <FormField
   363→   363→                control={form.control}
   364→   364→                name="counterparty"
   365→   365→                render={({ field }) => (
   366→   366→                  <FormItem className="sm:col-span-2">
   367→   367→                    <FormLabel>Contraparte *</FormLabel>
   368→   368→                    <FormControl>
   369→   369→                      <Input
   370→   370→                        placeholder="Persona o plataforma (ej: @usuario, Binance P2P, etc.)"
   371→   371→                        {...field}
   372→   372→                      />
   373→   373→                    </FormControl>
   374→   374→                    <FormMessage />
   375→   375→                  </FormItem>
   376→   376→                )}
   377→   377→              />
   378→   378→
   379→   379→              {/* EXCHANGE */}
   380→   380→              <FormField
   381→   381→                control={form.control}
   382→   382→                name="exchangeId"
   383→   383→                render={({ field }) => (
   384→   384→                  <FormItem className="sm:col-span-2">
   385→   385→                    <FormLabel className="flex items-center gap-1.5">
   386→   386→                      <Sparkles className="h-3.5 w-3.5 text-blue-500" />
   387→   387→                      Exchange / Plataforma
   388→   388→                    </FormLabel>
   389→   389→                    <Select
   390→   390→                      onValueChange={handleExchangeChange}
   391→   391→                      value={field.value ?? '__none'}
   392→   392→                    >
   393→   393→                      <FormControl>
   394→   394→                        <SelectTrigger>
   395→   395→                          <SelectValue placeholder="Sin exchange específico" />
   396→   396→                        </SelectTrigger>
   397→   397→                      </FormControl>
   398→   398→                      <SelectContent>
   399→   399→                        <SelectItem value="__none">Sin exchange (manual)</SelectItem>
   400→   400→                        {activeExchanges.map((e) => (
   401→   401→                          <SelectItem key={e.id} value={e.id}>
   402→   402→                            <span className="flex items-center gap-2">
   403→   403→                              <span
   404→   404→                                className="inline-block w-2 h-2 rounded-full"
   405→   405→                                style={{ backgroundColor: e.color }}
   406→   406→                              />
   407→   407→                              {e.name}
   408→   408→                              <span className="text-xs text-muted-foreground ml-1">
   409→   409→                                {watchedType === 'compra'
   410→   410→                                  ? e.buyFeeType === 'percent'
   411→   411→                                    ? `${e.buyFeeValue}%`
   412→   412→                                    : `fijo ${e.buyFeeValue}`
   413→   413→                                  : e.sellFeeType === 'percent'
   414→   414→                                    ? `${e.sellFeeValue}%`
   415→   415→                                    : `fijo ${e.sellFeeValue}`}
   416→   416→                              </span>
   417→   417→                            </span>
   418→   418→                          </SelectItem>
   419→   419→                        ))}
   420→   420→                      </SelectContent>
   421→   421→                    </Select>
   422→   422→                    <FormDescription>
   423→   423→                      Selecciona el exchange para calcular la comisión automáticamente. Puedes
   424→   424→                      ajustarla después.
   425→   425→                    </FormDescription>
   426→   426→                    <FormMessage />
   427→   427→                  </FormItem>
   428→   428→                )}
   429→   429→              />
   430→   430→
   431→   431→              <FormField
   432→   432→                control={form.control}
   433→   433→                name="asset"
   434→   434→                render={({ field }) => (
   435→   435→                  <FormItem>
   436→   436→                    <FormLabel>Activo</FormLabel>
   437→   437→                    <Select onValueChange={field.onChange} value={field.value}>
   438→   438→                      <FormControl>
   439→   439→                        <SelectTrigger>
   440→   440→                          <SelectValue />
   441→   441→                        </SelectTrigger>
   442→   442→                      </FormControl>
   443→   443→                      <SelectContent>
   444→   444→                        {ASSETS.map((a) => (
   445→   445→                          <SelectItem key={a} value={a}>
   446→   446→                            {a}
   447→   447→                          </SelectItem>
   448→   448→                        ))}
   449→   449→                      </SelectContent>
   450→   450→                    </Select>
   451→   451→                    <FormMessage />
   452→   452→                  </FormItem>
   453→   453→                )}
   454→   454→              />
   455→   455→
   456→   456→              <FormField
   457→   457→                control={form.control}
   458→   458→                name="currency"
   459→   459→                render={({ field }) => (
   460→   460→                  <FormItem>
   461→   461→                    <FormLabel>Moneda fiat</FormLabel>
   462→   462→                    <Select onValueChange={field.onChange} value={field.value}>
   463→   463→                      <FormControl>
   464→   464→                        <SelectTrigger>
   465→   465→                          <SelectValue />
   466→   466→                        </SelectTrigger>
   467→   467→                      </FormControl>
   468→   468→                      <SelectContent>
   469→   469→                        {CURRENCIES.map((c) => (
   470→   470→                          <SelectItem key={c} value={c}>
   471→   471→                            {c}
   472→   472→                          </SelectItem>
   473→   473→                        ))}
   474→   474→                      </SelectContent>
   475→   475→                    </Select>
   476→   476→                    <FormMessage />
   477→   477→                  </FormItem>
   478→   478→                )}
   479→   479→              />
   480→   480→
   481→   481→              <FormField
   482→   482→                control={form.control}
   483→   483→                name="amount"
   484→   484→                render={({ field }) => (
   485→   485→                  <FormItem>
   486→   486→                    <FormLabel>Cantidad de {watchedAsset || 'activo'} *</FormLabel>
   487→   487→                    <FormControl>
   488→   488→                      <Input
   489→   489→                        type="number"
   490→   490→                        step="0.000001"
   491→   491→                        placeholder="0.00"
   492→   492→                        {...field}
   493→   493→                      />
   494→   494→                    </FormControl>
   495→   495→                    <FormMessage />
   496→   496→                  </FormItem>
   497→   497→                )}
   498→   498→              />
   499→   499→
