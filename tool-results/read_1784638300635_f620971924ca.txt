     1→'use client'

     4→import { useQuery, useQueryClient } from '@tanstack/react-query'
     5→import {
     6→  Card,
     7→  CardContent,
     8→  CardDescription,
     9→  CardHeader,
    10→  CardTitle,
    11→} from '@/components/ui/card'
    12→import { Button } from '@/components/ui/button'
    13→import { Input } from '@/components/ui/input'
    14→import {
    15→  Select,

    17→  SelectItem,
    18→  SelectTrigger,
    19→  SelectValue,
    20→} from '@/components/ui/select'
    21→import {
    22→  Table,
    23→  TableBody,
    24→  TableCell,
    25→  TableHead,
    26→  TableHeader,
    27→  TableRow,
    28→} from '@/components/ui/table'
    29→import {
    30→  AlertDialog,
    31→  AlertDialogAction,
    32→  AlertDialogCancel,
    33→  AlertDialogContent,
    34→  AlertDialogDescription,
    35→  AlertDialogFooter,
    36→  AlertDialogHeader,
    37→  AlertDialogTitle,
    38→} from '@/components/ui/alert-dialog'
    39→import {
    40→  Plus,
    41→  Pencil,
    42→  Trash2,
    43→  Search,
    44→  ShoppingCart,
    45→  Tag,
    46→  AlertCircle,
    47→  Filter,
    48→  Inbox,
    49→  Building2,
    50→} from 'lucide-react'
    51→import type {
    52→  Bank,
    53→  Exchange,
    54→  Transaction,
    55→  TransactionType,
    56→  TransactionStatus,
    57→} from '@/lib/types'
    58→import { TransactionForm } from './TransactionForm'
    59→import { formatCurrency, formatNumber, formatDate } from '@/lib/format'
    60→import { toast } from '@/hooks/use-toast'
    61→import { Badge } from '@/components/ui/badge'
    62→import * as storage from '@/lib/storage'
    63→
    64→const STATUS_LABELS: Record<TransactionStatus, string> = {
    65→  completada: 'Completada',
    66→  pendiente: 'Pendiente',
    67→  cancelada: 'Cancelada',
    68→}
    69→
    70→const STATUS_COLORS: Record<TransactionStatus, string> = {
    71→  completada: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300',
    72→  pendiente: 'bg-amber-100 text-amber-700 dark:bg-amber-950/40 dark:text-amber-300',
    73→  cancelada: 'bg-rose-100 text-rose-700 dark:bg-rose-950/40 dark:text-rose-300',
    74→}
    75→
    76→interface TransactionsViewProps {
    77→  mode: 'compra' | 'venta' | 'all'
    78→}
    79→
    80→export function TransactionsView({ mode }: TransactionsViewProps) {
    81→  const queryClient = useQueryClient()
    82→  const [search, setSearch] = useState('')
    83→  const [statusFilter, setStatusFilter] = useState<string>('all')
    84→  const [bankFilter, setBankFilter] = useState<string>('all')
    85→  const [exchangeFilter, setExchangeFilter] = useState<string>('all')
    86→  const [formOpen, setFormOpen] = useState(false)
    87→  const [editing, setEditing] = useState<Transaction | null>(null)
    88→  const [deleting, setDeleting] = useState<Transaction | null>(null)
    89→
    90→  // Cargar bancos y exchanges
    91→  const { data: banks = [] } = useQuery<Bank[]>({
    92→    queryKey: ['banks'],
    93→    queryFn: () => storage.listBanks(),
    94→  })
    95→
    96→  const { data: exchanges = [] } = useQuery<Exchange[]>({
    97→    queryKey: ['exchanges'],
    98→    queryFn: () => storage.listExchanges(),
    99→  })
   100→
   101→  const filters = useMemo(
   102→    () => ({
   103→      type: mode !== 'all' ? (mode as TransactionType) : undefined,
   104→      status: statusFilter !== 'all' ? (statusFilter as TransactionStatus) : undefined,
   105→      bankId: bankFilter !== 'all' ? bankFilter : undefined,
   106→      exchangeId: exchangeFilter !== 'all' ? exchangeFilter : undefined,
   107→    }),
   108→    [mode, statusFilter, bankFilter, exchangeFilter]
   109→  )
   110→
   111→  const { data: transactions = [], isLoading } = useQuery<Transaction[]>({
   112→    queryKey: ['transactions', filters],
   113→    queryFn: () => storage.listTransactions(filters),
   114→  })
   115→
   116→  const filtered = transactions.filter((t) => {
   117→    const q = search.toLowerCase().trim()
   118→    if (!q) return true
   119→    return (
   120→      t.counterparty.toLowerCase().includes(q) ||
   121→      t.asset.toLowerCase().includes(q) ||
   122→      (t.reference ?? '').toLowerCase().includes(q) ||
   123→      (t.notes ?? '').toLowerCase().includes(q)
   124→    )
   125→  })
   126→
   127→  // KPIs
   128→  const completadas = filtered.filter((t) => t.status === 'completada')
   129→  const totalMonto = completadas.reduce((s, t) => s + t.amount, 0)
   130→  const totalFiat = completadas.reduce((s, t) => s + t.total, 0)
   131→  const totalFees = completadas.reduce((s, t) => s + t.fee, 0)
   132→  const totalNeto = completadas.reduce((s, t) => s + (t.netTotal ?? t.total - t.fee), 0)
   133→  const tasaPromedio = totalMonto > 0 ? totalFiat / totalMonto : 0
   134→
   135→  async function refetch() {
   136→    await queryClient.invalidateQueries({ queryKey: ['transactions'] })
   137→    await queryClient.invalidateQueries({ queryKey: ['banks'] })
   138→    await queryClient.invalidateQueries({ queryKey: ['exchanges'] })
   139→    await queryClient.invalidateQueries({ queryKey: ['stats'] })
   140→  }
   141→
   142→  function openNew() {
   143→    setEditing(null)
   144→    setFormOpen(true)
   145→  }
   146→
   147→  function openEdit(t: Transaction) {
   148→    setEditing(t)
   149→    setFormOpen(true)
   150→  }
   151→
   152→  async function confirmDelete() {
   153→    if (!deleting) return
   154→    try {
   155→      await storage.deleteTransaction(deleting.id)
   156→      toast({
   157→        title: 'Transacción eliminada',
   158→        description: `Operación con ${deleting.counterparty} fue eliminada.`,
   159→      })
   160→      await refetch()
   161→    } catch (e) {
   162→      toast({
   163→        title: 'Error',
   164→        description: e instanceof Error ? e.message : 'Error desconocido',
   165→        variant: 'destructive',
   166→      })
   167→    } finally {
   168→      setDeleting(null)
   169→    }
   170→  }
   171→
   172→  const title =
   173→    mode === 'compra'
   174→      ? 'Compras P2P'
   175→      : mode === 'venta'
   176→        ? 'Ventas P2P'
   177→        : 'Todas las transacciones'
   178→
   179→  const description =
   180→    mode === 'compra'
   181→      ? 'Operaciones donde recibiste un activo a cambio de moneda fiat.'
   182→      : mode === 'venta'
   183→        ? 'Operaciones donde entregaste un activo a cambio de moneda fiat.'
   184→        : 'Historial completo de compras y ventas P2P.'
   185→
   186→  const defaultType: TransactionType =
   187→    mode === 'venta' ? 'venta' : 'compra'
   188→
   189→  return (
   190→    <div className="space-y-2.5 sm:space-y-4">
   191→      {/* KPIs */}
   192→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-1.5 sm:gap-3">
   193→        <Card className="p-0">
   194→          <CardHeader className="p-2 sm:p-3 pb-1 sm:pb-2">
   195→            <CardDescription className="text-[9px] sm:text-xs">Operaciones</CardDescription>
   196→            <CardTitle className="text-sm sm:text-2xl tabular-nums leading-tight">
   197→              {filtered.length}
   198→            </CardTitle>
   199→          </CardHeader>
   200→        </Card>
   201→        <Card className="p-0">
   202→          <CardHeader className="p-2 sm:p-3 pb-1 sm:pb-2">
   203→            <CardDescription className="text-[9px] sm:text-xs">Volumen bruto</CardDescription>
   204→            <CardTitle className="text-sm sm:text-2xl tabular-nums leading-tight">
   205→              {formatCurrency(totalFiat)}
   206→            </CardTitle>
   207→          </CardHeader>
   208→        </Card>
   209→        <Card className="p-0">
   210→          <CardHeader className="p-2 sm:p-3 pb-1 sm:pb-2">
   211→            <CardDescription className="flex items-center gap-1 text-[9px] sm:text-xs">
   212→              <Building2 className="h-2.5 w-2.5 sm:h-3 sm:w-3 text-amber-500" /> Comisiones
   213→            </CardDescription>
   214→            <CardTitle className="text-sm sm:text-2xl tabular-nums text-amber-600 dark:text-amber-400 leading-tight">
   215→              {formatCurrency(totalFees)}
   216→            </CardTitle>
   217→          </CardHeader>
   218→        </Card>
   219→        <Card className="p-0">
   220→          <CardHeader className="p-2 sm:p-3 pb-1 sm:pb-2">
   221→            <CardDescription className="text-[9px] sm:text-xs">Volumen neto</CardDescription>
   222→            <CardTitle className="text-sm sm:text-2xl tabular-nums text-emerald-600 dark:text-emerald-400 leading-tight">
   223→              {formatCurrency(totalNeto)}
   224→            </CardTitle>
   225→          </CardHeader>
   226→        </Card>
   227→      </div>
   228→
   229→      {/* Header con acciones */}
   230→      <div className="flex flex-col sm:flex-row gap-2 sm:gap-3 sm:items-center sm:justify-between">
   231→        <div>
   232→          <h2 className="text-base sm:text-xl font-semibold flex items-center gap-2">
   233→            {mode === 'compra' && <ShoppingCart className="h-4 w-4 sm:h-5 sm:w-5 text-emerald-500" />}
   234→            {mode === 'venta' && <Tag className="h-4 w-4 sm:h-5 sm:w-5 text-rose-500" />}
   235→            {title}
   236→          </h2>
   237→          <p className="text-[11px] sm:text-sm text-muted-foreground">{description}</p>
   238→        </div>
   239→        <Button onClick={openNew} size="sm" className="self-start">
   240→          <Plus className="mr-1 h-4 w-4" />
   241→          {mode === 'venta' ? 'Nueva venta' : 'Nueva compra'}
   242→        </Button>
   243→      </div>
   244→
   245→      {/* Filtros */}
   246→      <div className="grid grid-cols-2 sm:flex sm:flex-row gap-1.5 sm:gap-2">
   247→        <div className="relative col-span-2 sm:flex-1">
   248→          <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />
   249→          <Input
   250→            placeholder="Buscar contraparte, activo..."
   251→            value={search}
   252→            onChange={(e) => setSearch(e.target.value)}
   253→            className="pl-8 h-9 text-sm"
   254→          />
   255→        </div>
   256→        <Select value={statusFilter} onValueChange={setStatusFilter}>
   257→          <SelectTrigger className="w-full sm:w-36 h-9 text-sm">
   258→            <Filter className="mr-1 h-3.5 w-3.5" />
   259→            <SelectValue placeholder="Estado" />
   260→          </SelectTrigger>
   261→          <SelectContent>
   262→            <SelectItem value="all">Todos</SelectItem>
   263→            <SelectItem value="completada">Completadas</SelectItem>
   264→            <SelectItem value="pendiente">Pendientes</SelectItem>
   265→            <SelectItem value="cancelada">Canceladas</SelectItem>
   266→          </SelectContent>
   267→        </Select>
   268→        <Select value={bankFilter} onValueChange={setBankFilter}>
   269→          <SelectTrigger className="w-full sm:w-36 h-9 text-sm">
   270→            <SelectValue placeholder="Banco" />
   271→          </SelectTrigger>
   272→          <SelectContent>
   273→            <SelectItem value="all">Todos los bancos</SelectItem>
   274→            {banks.map((b) => (
   275→              <SelectItem key={b.id} value={b.id}>
   276→                {b.name}
   277→              </SelectItem>
   278→            ))}
   279→          </SelectContent>
   280→        </Select>
   281→        <Select value={exchangeFilter} onValueChange={setExchangeFilter}>
   282→          <SelectTrigger className="w-full sm:w-36 h-9 text-sm col-span-2 sm:col-span-1">
   283→            <SelectValue placeholder="Exchange" />
   284→          </SelectTrigger>
   285→          <SelectContent>
   286→            <SelectItem value="all">Todos</SelectItem>
   287→            <SelectItem value="__none__">Sin exchange</SelectItem>
   288→            {exchanges.map((e) => (
   289→              <SelectItem key={e.id} value={e.id}>
   290→                {e.name}
   291→              </SelectItem>
   292→            ))}
   293→          </SelectContent>
   294→        </Select>
   295→      </div>
   296→
   297→      {/* Tabla en desktop / cards en móvil */}
   298→      <Card>
   299→        <CardContent className="p-0">
   300→          {isLoading ? (
   301→            <div className="p-4 sm:p-8 space-y-2">
   302→              {[...Array(4)].map((_, i) => (
   303→                <div key={i} className="h-12 rounded bg-muted/40 animate-pulse" />
   304→              ))}
   305→            </div>
   306→          ) : filtered.length === 0 ? (
   307→            <div className="flex flex-col items-center justify-center py-12 sm:py-16 text-center">
   308→              <Inbox className="h-10 w-10 sm:h-12 sm:w-12 text-muted-foreground/50 mb-3" />
   309→              <p className="font-medium text-sm">No hay transacciones</p>
   310→              <p className="text-xs sm:text-sm text-muted-foreground mt-1 max-w-sm px-4">
   311→                {search || statusFilter !== 'all' || bankFilter !== 'all' || exchangeFilter !== 'all'
   312→                  ? 'No se encontraron resultados con los filtros actuales.'
   313→                  : `Registra tu primera ${mode === 'venta' ? 'venta' : mode === 'compra' ? 'compra' : 'operación'} para empezar a llevar el control.`}
   314→              </p>
   315→              {!search && statusFilter === 'all' && bankFilter === 'all' && exchangeFilter === 'all' && (
   316→                <Button onClick={openNew} size="sm" className="mt-4">
   317→                  <Plus className="mr-1 h-4 w-4" />
   318→                  {mode === 'venta' ? 'Registrar venta' : 'Registrar compra'}
   319→                </Button>
   320→              )}
   321→            </div>
   322→          ) : (
   323→            <>
   324→              {/* MOBILE: cards verticales */}
   325→              <div className="sm:hidden divide-y">
   326→                {filtered.map((t) => (
   327→                  <div key={t.id} className="p-2.5 space-y-1.5">
   328→                    <div className="flex items-center justify-between gap-2">
   329→                      <div className="flex items-center gap-1.5 min-w-0">
   330→                        <Badge
   331→                          variant="outline"
   332→                          className={
   333→                            t.type === 'compra'
   334→                              ? 'border-emerald-300 text-emerald-700 dark:border-emerald-700 dark:text-emerald-300 text-[9px] px-1.5 py-0'
   335→                              : 'border-rose-300 text-rose-700 dark:border-rose-700 dark:text-rose-300 text-[9px] px-1.5 py-0'
   336→                          }
   337→                        >
   338→                          {t.type === 'compra' ? 'Compra' : 'Venta'}
   339→                        </Badge>
   340→                        <span className="text-[10px] text-muted-foreground whitespace-nowrap">
   341→                          {formatDate(t.date, true)}
   342→                        </span>
   343→                      </div>
   344→                      <div className="flex gap-0.5 flex-shrink-0">
   345→                        <Button
   346→                          size="icon"
   347→                          variant="ghost"
   348→                          className="h-7 w-7"
   349→                          onClick={() => openEdit(t)}
   350→                          title="Editar"
   351→                        >
   352→                          <Pencil className="h-3 w-3" />
   353→                        </Button>
   354→                        <Button
   355→                          size="icon"
   356→                          variant="ghost"
   357→                          className="h-7 w-7 hover:text-rose-600"
   358→                          onClick={() => setDeleting(t)}
   359→                          title="Eliminar"
   360→                        >
   361→                          <Trash2 className="h-3 w-3" />
   362→                        </Button>
   363→                      </div>
   364→                    </div>
   365→                    <div className="flex items-center justify-between gap-2">
   366→                      <span className="font-medium text-sm truncate">{t.counterparty}</span>
   367→                      {t.exchange && (
   368→                        <span
   369→                          className="inline-flex items-center gap-1 px-1.5 py-0.5 rounded text-[9px] flex-shrink-0"
   370→                          style={{ backgroundColor: `${t.exchange.color}22` }}
   371→                          title={t.exchange.name}
   372→                        >
   373→                          <span
   374→                            className="w-1.5 h-1.5 rounded-full"
   375→                            style={{ backgroundColor: t.exchange.color }}
   376→                          />
   377→                          {t.exchange.shortName ?? t.exchange.name}
   378→                        </span>
   379→                      )}
   380→                    </div>
   381→                    {t.reference && (
   382→                      <p className="text-[10px] text-muted-foreground font-mono truncate">
   383→                        Ref: {t.reference}
   384→                      </p>
   385→                    )}
   386→                    <div className="flex items-center justify-between text-xs">
   387→                      <span className="tabular-nums">
   388→                        {formatNumber(t.amount, 2)} <span className="text-muted-foreground">{t.asset}</span>
   389→                      </span>
   390→                      <span className="tabular-nums text-muted-foreground">
   391→                        @{formatNumber(t.rate, 2)}
   392→                      </span>
   393→                    </div>
   394→                    <div className="flex items-center justify-between text-xs">
   395→                      <span className="text-muted-foreground">Bruto</span>
   396→                      <span className="font-medium tabular-nums">
   397→                        {formatCurrency(t.total, t.currency)}
   398→                      </span>
   399→                    </div>
   400→                    {t.fee > 0 && (
   401→                      <div className="flex items-center justify-between text-xs">
   402→                        <span className="text-muted-foreground">Comisión</span>
   403→                        <span className="text-amber-600 dark:text-amber-400 tabular-nums">
   404→                          −{formatCurrency(t.fee, t.currency)}
   405→                        </span>
   406→                      </div>
   407→                    )}
   408→                    <div className="flex items-center justify-between text-xs">
   409→                      <span className="text-muted-foreground">Neto</span>
   410→                      <span className="font-bold tabular-nums text-emerald-600 dark:text-emerald-400">
   411→                        {formatCurrency(t.netTotal ?? t.total - t.fee, t.currency)}
   412→                      </span>
   413→                    </div>
   414→                    <div className="flex items-center justify-between gap-2 pt-0.5">
   415→                      <div className="flex items-center gap-1 text-[10px] min-w-0">
   416→                        {t.fromBank && (
   417→                          <span
   418→                            className="inline-flex items-center gap-1 px-1.5 py-0.5 rounded truncate"
   419→                            style={{ backgroundColor: `${t.fromBank.color}22` }}
   420→                            title={`Origen: ${t.fromBank.name}`}
   421→                          >
   422→                            <span
   423→                              className="w-1.5 h-1.5 rounded-full flex-shrink-0"
   424→                              style={{ backgroundColor: t.fromBank.color }}
   425→                            />
   426→                            <span className="truncate">{t.fromBank.name}</span>
   427→                          </span>
   428→                        )}
   429→                        {t.toBank && (
   430→                          <>
   431→                            <span className="text-muted-foreground flex-shrink-0">→</span>
   432→                            <span
   433→                              className="inline-flex items-center gap-1 px-1.5 py-0.5 rounded truncate"
   434→                              style={{ backgroundColor: `${t.toBank.color}22` }}
   435→                              title={`Destino: ${t.toBank.name}`}
   436→                            >
   437→                              <span
   438→                                className="w-1.5 h-1.5 rounded-full flex-shrink-0"
   439→                                style={{ backgroundColor: t.toBank.color }}
   440→                              />
   441→                              <span className="truncate">{t.toBank.name}</span>
   442→                            </span>
   443→                          </>
   444→                        )}
   445→                        {!t.fromBank && !t.toBank && (
   446→                          <span className="text-muted-foreground">Sin bancos</span>
   447→                        )}
   448→                      </div>
   449→                      <span
   450→                        className={`inline-block px-1.5 py-0.5 rounded text-[9px] flex-shrink-0 ${STATUS_COLORS[t.status]}`}
   451→                      >
   452→                        {STATUS_LABELS[t.status]}
   453→                      </span>
   454→                    </div>
   455→                  </div>
   456→                ))}
   457→              </div>
   458→
   459→              {/* DESKTOP: tabla */}
   460→              <div className="hidden sm:block overflow-x-auto">
   461→                <Table>
   462→                  <TableHeader>
   463→                    <TableRow>
   464→                      <TableHead className="w-[100px]">Tipo</TableHead>
   465→                      <TableHead>Fecha</TableHead>
   466→                      <TableHead>Contraparte</TableHead>
   467→                      <TableHead>Exchange</TableHead>
   468→                      <TableHead className="text-right">Activo</TableHead>
   469→                      <TableHead className="text-right">Tasa</TableHead>
   470→                      <TableHead className="text-right">Bruto</TableHead>
   471→                      <TableHead className="text-right">Comisión</TableHead>
   472→                      <TableHead className="text-right">Neto</TableHead>
   473→                      <TableHead>Bancos</TableHead>
   474→                      <TableHead>Estado</TableHead>
   475→                      <TableHead className="w-[80px]"></TableHead>
   476→                    </TableRow>
   477→                  </TableHeader>
   478→                  <TableBody>
   479→                    {filtered.map((t) => (
   480→                      <TableRow key={t.id} className="hover:bg-muted/40">
   481→                        <TableCell>
   482→                          <Badge
   483→                            variant="outline"
   484→                            className={
   485→                              t.type === 'compra'
   486→                                ? 'border-emerald-300 text-emerald-700 dark:border-emerald-700 dark:text-emerald-300'
   487→                                : 'border-rose-300 text-rose-700 dark:border-rose-700 dark:text-rose-300'
   488→                            }
   489→                          >
   490→                            {t.type === 'compra' ? 'Compra' : 'Venta'}
   491→                          </Badge>
   492→                        </TableCell>
   493→                        <TableCell className="text-xs whitespace-nowrap">
   494→                          {formatDate(t.date, true)}
   495→                        </TableCell>
   496→                        <TableCell>
   497→                          <div className="font-medium truncate max-w-[180px]">
   498→                            {t.counterparty}
   499→                          </div>
   500→                          {t.reference && (
   501→                            <div className="text-xs text-muted-foreground font-mono">
   502→                              Ref: {t.reference}
   503→                            </div>
   504→                          )}
   505→                        </TableCell>
   506→                        <TableCell>
   507→                          {t.exchange ? (
   508→                            <span
   509→                              className="inline-flex items-center gap-1 px-1.5 py-0.5 rounded text-xs"
   510→                              style={{ backgroundColor: `${t.exchange.color}22` }}
   511→                              title={t.exchange.name}
   512→                            >
   513→                              <span
   514→                                className="w-1.5 h-1.5 rounded-full"
   515→                                style={{ backgroundColor: t.exchange.color }}
   516→                              />
   517→                              {t.exchange.shortName ?? t.exchange.name}
