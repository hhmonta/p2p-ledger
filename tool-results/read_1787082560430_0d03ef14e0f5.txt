    40→// Schema de un tier
    41→const tierSchema = z.object({
    42→  minAmount: z.coerce.number().min(0),
    43→  feeType: z.enum(['percent', 'fixed']),
    44→  feeValue: z.coerce.number().min(0),
    45→})
    46→
    47→const exchangeSchema = z.object({
    48→  name: z.string().min(1, 'El nombre es obligatorio'),
    49→  shortName: z.string().optional(),
    50→  color: z.string().min(1),
    51→  buyFeeType: z.enum(['percent', 'fixed']),
    52→  buyFeeValue: z.coerce.number().min(0),
    53→  buyTiers: z.array(tierSchema).optional(),
    54→  sellFeeType: z.enum(['percent', 'fixed']),
    55→  sellFeeValue: z.coerce.number().min(0),
    56→  sellTiers: z.array(tierSchema).optional(),
    57→  fixedFee: z.coerce.number().min(0).default(0),
    58→  fixedFeeCurrency: z.string().min(1),
    59→  discountPercent: z.coerce.number().min(0).max(100).default(0),
    60→  discountLabel: z.string().optional(),
    61→  isActive: z.boolean().default(true),
    62→  notes: z.string().optional(),
    63→})
    64→
    65→type FormValues = z.infer<typeof exchangeSchema>
    66→type TierField = { minAmount: number; feeType: 'percent' | 'fixed'; feeValue: number }

    93→
    94→export function ExchangeForm({
    95→  open,
    96→  onOpenChange,
    97→  exchange,
    98→  onSaved,
    99→}: ExchangeFormProps) {
   100→  const [submitting, setSubmitting] = useState(false)
   101→
   102→  const form = useForm<FormValues>({
   103→    resolver: zodResolver(exchangeSchema),
   104→    defaultValues: {
   105→      name: '',
   106→      shortName: '',
   107→      color: '#3b82f6',
   108→      buyFeeType: 'percent',
   109→      buyFeeValue: 0,
   110→      buyTiers: [],
   111→      sellFeeType: 'percent',
   112→      sellFeeValue: 0,
   113→      sellTiers: [],
   114→      fixedFee: 0,
   115→      fixedFeeCurrency: 'USDT',
   116→      discountPercent: 0,
   117→      discountLabel: '',
   118→      isActive: true,
   119→      notes: '',
   120→    },
   121→  })
   122→
   123→  // Field arrays para tiers dinámicos
   124→  const buyTiersField = useFieldArray({ control: form.control, name: 'buyTiers' })
   125→  const sellTiersField = useFieldArray({ control: form.control, name: 'sellTiers' })
   126→
   127→  useEffect(() => {
   128→    if (open) {
   129→      if (exchange) {
   130→        form.reset({
   131→          name: exchange.name,
   132→          shortName: exchange.shortName ?? '',
   133→          color: exchange.color,
   134→          buyFeeType: exchange.buyFeeType as FeeType,
   135→          buyFeeValue: exchange.buyFeeValue,
   136→          buyTiers: (exchange.buyTiers ?? []).map((t) => ({
   137→            minAmount: t.minAmount,
   138→            feeType: t.feeType,
   139→            feeValue: t.feeValue,
   140→          })),
   141→          sellFeeType: exchange.sellFeeType as FeeType,
   142→          sellFeeValue: exchange.sellFeeValue,
   143→          sellTiers: (exchange.sellTiers ?? []).map((t) => ({
   144→            minAmount: t.minAmount,
   145→            feeType: t.feeType,
   146→            feeValue: t.feeValue,
   147→          })),
   148→          fixedFee: exchange.fixedFee,
   149→          fixedFeeCurrency: exchange.fixedFeeCurrency,
   150→          discountPercent: exchange.discountPercent,
   151→          discountLabel: exchange.discountLabel ?? '',
   152→          isActive: exchange.isActive,
   153→          notes: exchange.notes ?? '',
   154→        })
   155→      } else {
   156→        form.reset({
   157→          name: '',
   158→          shortName: '',
   159→          color: '#3b82f6',
   160→          buyFeeType: 'percent',
   161→          buyFeeValue: 0,
   162→          buyTiers: [],
   163→          sellFeeType: 'percent',
   164→          sellFeeValue: 0,
   165→          sellTiers: [],
   166→          fixedFee: 0,
   167→          fixedFeeCurrency: 'USDT',
   168→          discountPercent: 0,
   169→          discountLabel: '',
   170→          isActive: true,
   171→          notes: '',
   172→        })
   173→      }
   174→    }
   175→  }, [open, exchange, form])
   176→
   177→  async function onSubmit(values: FormValues) {
   178→    setSubmitting(true)
   179→    try {

   182→        if (!tiers || tiers.length === 0) return []

   184→          .filter((t) => t.minAmount !== undefined && t.feeValue !== undefined)
   185→          .map((t) => ({
   186→            minAmount: Number(t.minAmount),
   187→            feeType: t.feeType as FeeType,
   188→            feeValue: Number(t.feeValue),
   189→          }))
   190→          .sort((a, b) => a.minAmount - b.minAmount)
   191→      }
   192→
   193→      const payload: ExchangeInput = {
   194→        name: values.name,
   195→        shortName: values.shortName || null,
   196→        color: values.color,
   197→        buyFeeType: values.buyFeeType,
   198→        buyFeeValue: Number(values.buyFeeValue),
   199→        buyTiers: cleanTiers(values.buyTiers as TierField[] | undefined),
   200→        sellFeeType: values.sellFeeType,
   201→        sellFeeValue: Number(values.sellFeeValue),
   202→        sellTiers: cleanTiers(values.sellTiers as TierField[] | undefined),
   203→        fixedFee: Number(values.fixedFee),
   204→        fixedFeeCurrency: values.fixedFeeCurrency,
   205→        discountPercent: Number(values.discountPercent),
   206→        discountLabel: values.discountLabel || null,
   207→        isActive: values.isActive,
   208→        notes: values.notes || null,
   209→      }

   212→        await storage.updateExchange(exchange.id, payload)

   214→        await storage.createExchange(payload)

   218→        title: exchange ? 'Exchange actualizado' : 'Exchange creado',
   219→        description: `«${values.name}» se guardó correctamente.`,
   220→      })
   221→      onSaved()

   223→    } catch (e) {
   224→      toast({
   225→        title: 'Error',
   226→        description: e instanceof Error ? e.message : 'Error desconocido',
   227→        variant: 'destructive',
   228→      })

   230→      setSubmitting(false)
   231→    }
   232→  }
   233→
   234→  const buyFeeType = form.watch('buyFeeType')
   235→  const sellFeeType = form.watch('sellFeeType')
   236→  const buyTiers = form.watch('buyTiers') ?? []

   238→  const discountPercent = form.watch('discountPercent')

   242→      <DialogContent className="sm:max-w-[640px] max-h-[94vh] overflow-y-auto">
   243→        <DialogHeader>
   244→          <DialogTitle>
   245→            {exchange ? 'Editar exchange' : 'Nuevo exchange'}
   246→          </DialogTitle>
   247→          <DialogDescription>
   248→            Configura la plataforma P2P y sus comisiones. Puedes definir comisiones escalonadas
   249→            por monto y descuentos VIP/BNB que se aplican automáticamente.
   250→          </DialogDescription>
   251→        </DialogHeader>
   252→
   253→        <Form {...form}>
   254→          <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-4">
   255→            <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   256→              <FormField
   257→                control={form.control}
   258→                name="name"
   259→                render={({ field }) => (
   260→                  <FormItem className="sm:col-span-2">
   261→                    <FormLabel>Nombre del exchange *</FormLabel>
   262→                    <FormControl>
   263→                      <Input
   264→                        placeholder="Ej: Binance P2P, OKX, P2p.me, Mercado Libre..."
   265→                        {...field}
   266→                      />
   267→                    </FormControl>
   268→                    <FormMessage />
   269→                  </FormItem>
   270→                )}
   271→              />
   272→
   273→              <FormField
   274→                control={form.control}
   275→                name="shortName"
   276→                render={({ field }) => (
   277→                  <FormItem>
   278→                    <FormLabel>Nombre corto (opcional)</FormLabel>
   279→                    <FormControl>
   280→                      <Input
   281→                        placeholder="Ej: BIN, OKX, ML"
   282→                        maxLength={6}
   283→                        {...field}
   284→                      />
   285→                    </FormControl>
   286→                    <FormDescription>
   287→                      Etiqueta corta para badges.
   288→                    </FormDescription>
   289→                    <FormMessage />
   290→                  </FormItem>
   291→                )}
   292→              />
   293→
   294→              <FormField
   295→                control={form.control}
   296→                name="isActive"
   297→                render={({ field }) => (
   298→                  <FormItem className="flex flex-row items-center justify-between rounded-lg border p-3 h-[60px]">
   299→                    <div className="space-y-0.5">
   300→                      <FormLabel>Activo</FormLabel>

   302→                        Inactivo oculta el exchange en selects.
   303→                      </FormDescription>
   304→                    </div>
   305→                    <FormControl>
   306→                      <Switch
   307→                        checked={field.value}
   308→                        onCheckedChange={field.onChange}
   309→                      />
   310→                    </FormControl>
   311→                  </FormItem>
   312→                )}
   313→              />
   314→            </div>
   315→
   316→            {/* =========== COMISIONES DE COMPRA =========== */}
   317→            <div className="rounded-lg border p-3 space-y-3 bg-emerald-50/30 dark:bg-emerald-950/10">
   318→              <div className="flex items-center justify-between">
   319→                <div className="flex items-center gap-2">
   320→                  <div className="h-2 w-2 rounded-full bg-emerald-500" />
   321→                  <h4 className="text-sm font-medium">Comisión al comprar</h4>
   322→                </div>
   323→                {buyTiers.length > 0 && (
   324→                  <span className="text-xs px-2 py-0.5 rounded bg-emerald-500/20 text-emerald-700 dark:text-emerald-300">
   325→                    {buyTiers.length} {buyTiers.length === 1 ? 'tier' : 'tiers'} activos
   326→                  </span>
   327→                )}
   328→              </div>
   329→
   330→              {/* Comisión base (se usa si no hay tiers) */}
   331→              <div className="grid grid-cols-2 gap-3">
   332→                <FormField
   333→                  control={form.control}
   334→                  name="buyFeeType"
   335→                  render={({ field }) => (
   336→                    <FormItem>
   337→                      <FormLabel className="text-xs">Tipo base</FormLabel>
   338→                      <Select onValueChange={field.onChange} value={field.value}>
   339→                        <FormControl>
   340→                          <SelectTrigger className="h-9">
   341→                            <SelectValue />
   342→                          </SelectTrigger>
   343→                        </FormControl>
   344→                        <SelectContent>
   345→                          <SelectItem value="percent">Porcentual (%)</SelectItem>
   346→                          <SelectItem value="fixed">Fijo (monto)</SelectItem>
   347→                        </SelectContent>
   348→                      </Select>
   349→                      <FormMessage />
   350→                    </FormItem>
   351→                  )}
   352→                />

   358→                      <FormLabel className="text-xs">
   359→                        {buyFeeType === 'percent' ? 'Valor (%) base' : 'Monto base'}

   361→                      <FormControl>
   362→                        <Input
   363→                          type="number"
   364→                          step="0.0001"
   365→                          placeholder="0"
   366→                          disabled={buyTiers.length > 0}
   367→                          {...field}
   368→                        />
   369→                      </FormControl>

   371→                    </FormItem>
   372→                  )}

   380→                    <Layers className="h-3 w-3" />
   381→                    Tiers escalonados (opcional)

   385→                    variant="outline"
   386→                    size="sm"

   388→                    onClick={() =>
   389→                      buyTiersField.append({
   390→                        minAmount: buyTiers.length > 0 ? Math.max(...buyTiers.map((t) => Number(t.minAmount))) + 1000 : 0,
   391→                        feeType: 'percent',
   392→                        feeValue: 0.1,
   393→                      })
   394→                    }
   395→                  >
   396→                    <Plus className="h-3 w-3 mr-1" /> Añadir tier
   397→                  </Button>
   398→                </div>
   399→                {buyTiers.length === 0 ? (
   400→                  <p className="text-xs text-muted-foreground italic">
   401→                    Sin tiers. Se usará la comisión base para cualquier monto.
   402→                  </p>
   403→                ) : (
   404→                  <div className="space-y-2">
   405→                    {buyTiersField.fields.map((t, i) => (
   406→                      <div
   407→                        key={t.id}
   408→                        className="grid grid-cols-[1fr_1fr_1fr_auto] gap-2 items-center p-2 rounded border bg-background"
   409→                      >
   410→                        <div>
   411→                          <label className="text-[10px] text-muted-foreground">Monto mín.</label>
   412→                          <Input
   413→                            type="number"
   414→                            step="0.01"
   415→                            className="h-8 text-xs"
   416→                            {...form.register(`buyTiers.${i}.minAmount` as const)}
   417→                          />

   420→                          <label className="text-[10px] text-muted-foreground">Tipo</label>
   421→                          <Select
   422→                            onValueChange={(v) =>
   423→                              form.setValue(`buyTiers.${i}.feeType` as const, v as 'percent' | 'fixed')
   424→                            }
   425→                            value={form.watch(`buyTiers.${i}.feeType` as const)}
   426→                          >
   427→                            <SelectTrigger className="h-8 text-xs">
   428→                              <SelectValue />
   429→                            </SelectTrigger>
   430→                            <SelectContent>
   431→                              <SelectItem value="percent">%</SelectItem>
   432→                              <SelectItem value="fixed">Fijo</SelectItem>
   433→                            </SelectContent>
   434→                          </Select>
   435→                        </div>
   436→                        <div>

   470→                </div>
   471→                {sellTiers.length > 0 && (
   472→                  <span className="text-xs px-2 py-0.5 rounded bg-rose-500/20 text-rose-700 dark:text-rose-300">
   473→                    {sellTiers.length} {sellTiers.length === 1 ? 'tier' : 'tiers'} activos
   474→                  </span>
   475→                )}
   476→              </div>
   477→
   478→              <div className="grid grid-cols-2 gap-3">
   479→                <FormField
   480→                  control={form.control}
   481→                  name="sellFeeType"
   482→                  render={({ field }) => (
   483→                    <FormItem>
   484→                      <FormLabel className="text-xs">Tipo base</FormLabel>
   485→                      <Select onValueChange={field.onChange} value={field.value}>
   486→                        <FormControl>
   487→                          <SelectTrigger className="h-9">
   488→                            <SelectValue />
   489→                          </SelectTrigger>
   490→                        </FormControl>
   491→                        <SelectContent>
   492→                          <SelectItem value="percent">Porcentual (%)</SelectItem>
   493→                          <SelectItem value="fixed">Fijo (monto)</SelectItem>
   494→                        </SelectContent>
   495→                      </Select>
   496→                      <FormMessage />
   497→                    </FormItem>
   498→                  )}
   499→                />
   500→                <FormField
   501→                  control={form.control}
   502→                  name="sellFeeValue"
   503→                  render={({ field }) => (
   504→                    <FormItem>
   505→                      <FormLabel className="text-xs">
   506→                        {sellFeeType === 'percent' ? 'Valor (%) base' : 'Monto base'}
   507→                      </FormLabel>
   508→                      <FormControl>
   509→                        <Input
   510→                          type="number"
   511→                          step="0.0001"
   512→                          placeholder="0"

   519→                  )}
   520→                />
   521→              </div>
   522→
   523→              <div className="space-y-2">
   524→                <div className="flex items-center justify-between">
   525→                  <FormLabel className="text-xs flex items-center gap-1">
   526→                    <Layers className="h-3 w-3" />

   529→                  <Button
   530→                    type="button"
   531→                    variant="outline"

   534→                    onClick={() =>
   535→                      sellTiersField.append({

   548→                  </p>
   549→                ) : (
   550→                  <div className="space-y-2">
   551→                    {sellTiersField.fields.map((t, i) => (
